\section{Mathematical Background}

\subsection{Taylor series}
\begin{definition}[Taylor series]
    A function $f$ that is smooth near $a$ (it can be differentiated any number of times there) equals the power series
    \begin{equation}
        f(x) = \sum_{n=0}^{\infty}\frac{f^{(n)}(a)}{n!}(x-a)^n,
    \end{equation}
    where $f^{(n)}$ is the $n$th derivative. The special case $a = 0$ is called the Maclaurin series.
\end{definition}
\begin{result}[Series used in this course]
    \begin{equation}
        e^{x} = \sum_{n=0}^{\infty}\frac{x^n}{n!}, \qquad
        \cos x = \sum_{n=0}^{\infty}\frac{(-1)^n x^{2n}}{(2n)!}, \qquad
        \sin x = \sum_{n=0}^{\infty}\frac{(-1)^n x^{2n+1}}{(2n+1)!},
    \end{equation}
    \begin{equation}
        \frac{1}{1-x} = \sum_{n=0}^{\infty}x^n \quad (|x|<1), \qquad (1+u)^N = \sum_{n=0}^{N}\binom{N}{n}u^n \quad (N \in \mathbb{N}).
    \end{equation}
\end{result}
\begin{result}[Euler's formula]
    Comparing the series for $e^{i\theta}$, $\cos\theta$ and $\sin\theta$ gives $e^{i\theta} = \cos\theta + i\sin\theta$, hence
    \begin{equation}
        \cos\theta = \frac{e^{i\theta}+e^{-i\theta}}{2}, \qquad \sin\theta = \frac{e^{i\theta}-e^{-i\theta}}{2i}.
    \end{equation}
    These are used constantly when rotation matrices are written out.
\end{result}
\begin{problem}[The exponential as a limit]
    Show that $\lim_{N\to\infty}\left(1+x/N\right)^N = e^x$.
\end{problem}
\begin{remark}
    This limit is what lets a rotation be written as an exponential (Section~\ref{sec:generators}).
\end{remark}

\subsection{Bases}
\begin{definition}[Basis]
    A set of vectors is \emph{linearly independent} if none of them can be written as a combination of the others, and it \emph{spans} a space if every vector in the space is some combination of them. A \emph{basis} is a set that does both, so every vector is a unique combination of the basis vectors. The number of basis vectors is the dimension of the space.
\end{definition}
\begin{definition}[Kronecker delta]
    \begin{equation}
        \delta_{ij} = \begin{cases} 1 & i = j, \\ 0 & i \neq j. \end{cases}
    \end{equation}
\end{definition}

\subsection{The classical picture}

\subsubsection{Angular momentum}
\begin{definition}[Orbital angular momentum]
    The orbital angular momentum of a particle is
    \begin{equation}
        \vec{L} = \vec{r} \times \vec{p},
    \end{equation}
    where $\vec{r}$ is the position vector and $\vec{p} = m\vec{v}$ is the linear momentum.
\end{definition}
For motion confined to the $x$-$y$ plane, only the $z$ component survives. On a circular path of radius $r$ with speed $v$,
\[L_{z} = mvr.\]
\begin{definition}[Angular momentum of a rigid body]
    For a rigid body\footnote{A rigid body is one that does not deform under the action of external forces.} rotating about a fixed axis,
    \[L = I_{0}\omega,\]
    with $\omega$ the angular velocity and $I_{0}$ the moment of inertia, which measures how the mass is distributed relative to the rotation axis.
\end{definition}

\subsubsection{Magnetic moment of a circulating charge}
\begin{definition}[Magnetic moment]
    A vector $\vec{\mu}$ measuring the net circulation of electric charge. It sets the torque, and in a nonuniform field the force, that a magnet experiences in an external magnetic field.
\end{definition}
A point particle of mass $m$ and charge $q$ on a circular orbit of radius $r$ and period $T$ passes any given point once per period, so it acts as a steady current $I = q/T = qv/2\pi r$. Its moment is
\[\mu = \frac{IA}{c}\footnotemark = \frac{1}{c}\left(\frac{qv}{2\pi r}\right)\left(\pi r^{2}\right) = \frac{q}{2mc}\,mvr = \frac{q}{2mc}\,L_{z},\]
where $A$ is the area enclosed by the orbit.
\footnotetext{Gaussian (CGS) units, where a current loop has moment $IA/c$. In SI units it is $IA$.}

\subsubsection{Gyromagnetic ratio}
\begin{definition}[Gyromagnetic ratio]
    The constant $\gamma$ relating magnetic moment to angular momentum,
    \begin{equation}
        \vec{\mu} = \gamma \vec{L}, \qquad \gamma = \frac{q}{2mc} \text{ for a circulating charge.}
    \end{equation}
\end{definition}
\begin{remark}
    This is purely classical: a point charge on a definite orbit, nothing quantized. It matters because measuring $\vec{\mu}$ is how we measure angular momentum, which is what the Stern-Gerlach apparatus does.
\end{remark}

\subsection{Spin}
\begin{definition}[Spin]
    Spin $\vec{S}$ is an \emph{intrinsic} angular momentum: a fixed property of the particle type, like charge or mass, not caused by any motion through space. A stationary electron alone in empty space still has spin. It comes in addition to any orbital $\vec{L}$, not in place of it.
\end{definition}

\subsubsection{Intrinsic moment and the g-factor}
Spin also carries a magnetic moment proportional to it, but with an extra dimensionless factor,
\begin{equation}
    \vec{\mu} = \gamma\vec{S}, \qquad \gamma = \frac{gq}{2mc}.
\end{equation}
\begin{definition}[g-factor]
    The dimensionless number $g$ above. It says how much stronger or weaker the moment is than the classical value ($g = 1$). It must be measured or derived from relativistic quantum theory.
\end{definition}

\subsubsection{Magnetons}
\begin{definition}[Planck constant]
    The constant that sets the scale at which quantization matters. It relates photon energy to frequency, $E = h\nu$, and
    \[h \approx 6.626 \times 10^{-34}\ \mathrm{J\,s}, \qquad \hbar = \frac{h}{2\pi} = 1.055 \times 10^{-27}\ \mathrm{erg\,s} = 6.582 \times 10^{-16}\ \mathrm{eV\,s}.\]
    Both have units of angular momentum, $[\hbar] = [L_z] = \mathrm{kg\,m^2/s}$.
\end{definition}
Writing $\vec{S} = \hbar(\vec{S}/\hbar)$ with $\vec{S}/\hbar$ dimensionless, the $\hbar$ joins the gyromagnetic ratio. The combination $|e|\hbar/2mc$ then carries all the units and sets the natural size of the moment for a given mass.
\begin{definition}[Magnetons]
    The natural units of magnetic moment for the electron (Bohr magneton) and for nucleons, i.e.\ protons and neutrons (nuclear magneton):
    \begin{equation}
        \mu_{B} = \frac{|e|\hbar}{2m_{e}c}, \qquad \mu_{N} = \frac{|e|\hbar}{2m_{p}c} \approx \frac{\mu_B}{1836},
    \end{equation}
    since $m_{p}/m_{e} \approx 1836$.
\end{definition}
The electron moment is then
\begin{equation}
    \vec{\mu}_{e} = g_{e}\,\mu_{B}\,\frac{\vec{S}}{\hbar},
\end{equation}
and similarly for the proton and neutron with $\mu_{N}$ in place of $\mu_{B}$.

\begin{remark}
    Classically $\mu = (q/2mc)L_z$ vanishes when $q = 0$, yet the neutral neutron has a moment. So its $\vec{\mu}$ cannot come from a circulating point charge: a first sign the classical picture fails.
\end{remark}
\begin{remark}
    Since $m_{p} \gg m_{e}$, $\mu_{N} \ll \mu_{B}$, so nuclei are deflected far less than electrons in a Stern-Gerlach device.
\end{remark}

\subsection{The Stern-Gerlach apparatus}
\begin{definition}[Stern-Gerlach (SG) apparatus]
    A device that measures one component of a particle's magnetic moment. A beam of neutral atoms passes through a magnet shaped so that the field strength changes along a chosen axis, here $z$ (a field \emph{gradient}). The gradient pushes each atom by an amount proportional to $\mu_z$, and a screen records where it lands. Since $\vec{\mu} = \gamma\vec{S}$, the landing position measures $S_z$.
\end{definition}

    \centering

    \caption{Illustration of the Stern-Gerlach apparatus.}
    \label{fig:stern-gerlach-apparatus}
\end{figure}

\subsubsection{Force in a nonuniform field}
A magnetic moment in a field has potential energy $U = -\vec{\mu}\cdot\vec{B}$, so
\[\vec{F} = -\vec{\nabla}U = \vec{\nabla}(\vec{\mu}\cdot\vec{B}), \qquad F_{z} = \mu_{z}\frac{\partial B_{z}}{\partial z}.\]
\begin{remark}
    A uniform field gives no net force, only a torque (a twist that turns $\vec{\mu}$ without moving the atom). That is why the magnet needs a gradient.
\end{remark}
\begin{definition}[Deflection]
    The sideways displacement of an atom on the screen caused by $F_z$. It is proportional to $\mu_z$, so the screen pattern is a record of the moments in the beam.
\end{definition}

\subsubsection{Silver atoms}
Silver has zero net orbital angular momentum, $\vec{L} = 0$, and its whole moment comes from one unpaired outermost (valence) electron, so the atom behaves like a single electron spin. With
\[\mu_{z} = -\frac{g_{e}|e|}{2m_{e}c}\,S_{z}\]
and a gradient $\partial B_{z}/\partial z < 0$,
\[F_{z} = g_{e}\frac{|e|}{2m_{e}c}\left|\frac{\partial B_{z}}{\partial z}\right| S_{z}.\]
\begin{result}
    $F_{z} \propto S_{z}$: the deflection measures the $z$ component of spin directly.
\end{result}

\subsubsection{Classical prediction versus observation}
Atoms leave the oven with spins pointing in random directions, so $S_z = |\vec{S}|\cos\theta$ takes every value in $[-|\vec{S}|,|\vec{S}|]$. Classically the screen should show a continuous smear. It shows two spots.
\begin{result}[Quantization of spin]
    The electron's spin component is quantized:
    \begin{equation}
        S_{z} = \pm\frac{\hbar}{2}.
    \end{equation}
\end{result}
\begin{definition}[Spin quantum number]
    The number $s$, fixed by the particle type, taking values $s = 0, \tfrac12, 1, \tfrac32, \dots$ A measurement of $S_z$ gives one of $n = 2s+1$ outcomes,
    \begin{equation}
        S_z = m\hbar, \qquad m = -s, -s+1, \dots, +s,
    \end{equation}
    so $s$ is found by counting beams. Silver gives 2 beams, so the electron has $s = \tfrac12$. The deuteron (a proton and a neutron with aligned spins) gives 3 beams, $S_z = +\hbar, 0, -\hbar$, so $s = 1$.
\end{definition}
\begin{remark}
    $S_z$ is a component. The magnitude is $|\vec{S}| = \hbar\sqrt{s(s+1)}$, which for the electron is $\hbar\sqrt{3}/2$, not $\hbar/2$.
\end{remark}
\begin{definition}[Ket]
    \label{Ket-def}
    A ket $\ket{\,\cdot\,}$ is a label for a quantum state, treated as a vector. For electron spin, the two outcomes of an SG$z$ give
    \[\ket{+\hat{z}} \equiv \ket{\uparrow} \; \text{for } S_{z} = +\frac{\hbar}{2}, \qquad \ket{-\hat{z}} \equiv \ket{\downarrow} \; \text{for } S_{z} = -\frac{\hbar}{2},\]
    and likewise $\ket{\pm\hat{x}}$ for $S_x = \pm\hbar/2$, and so on for any axis. We follow Townsend and use the $\ket{\pm\hat{n}}$ labels.
\end{definition}

\subsubsection{\texorpdfstring{Experiment 1: SG$z$ then SG$z$}{Experiment 1: SGz then SGz}}
Filter the $+\hat{z}$ beam and send it into a second SG$z$. Every atom comes out $+\hat{z}$. The measurement is repeatable: an atom found with $S_z = +\hbar/2$ gives the same answer again, so it left the first device in $\ket{+\hat{z}}$. We end with $N_0/2$ atoms in $\ket{+\hat{z}}$.

\subsubsection{\texorpdfstring{Experiment 2: SG$z$ then SG$x$}{Experiment 2: SGz then SGx}}
Filter $+\hat{z}$ and send it into an SG$x$. Half come out $+\hat{x}$, half $-\hat{x}$: knowing $S_z$ tells us nothing about $S_x$. We end with $N_0/4$ in each of $\ket{\pm\hat{x}}$.

\subsubsection{\texorpdfstring{Experiment 3: SG$z$, SG$x$, then SG$z$}{Experiment 3: SGz, SGx, then SGz}}
Filter $+\hat{z}$, filter $+\hat{x}$, then measure $S_z$ again. The beam splits 50/50 even though every atom was $+\hat{z}$ at the start. Blocking $+\hat{x}$ instead gives the same result. We end with $N_0/8$ in each of $\ket{\pm\hat{z}}$.
\begin{result}
    No state has definite values of both $S_x$ and $S_z$. Measuring $S_x$ destroys what we knew about $S_z$.
\end{result}

\subsubsection{\texorpdfstring{Experiment 4: SG$z$, modified SG$x$, then SG$z$}{Experiment 4: SGz, modified SGx, then SGz}}
\begin{definition}[Modified SG device]
    An SG device that splits the beam according to $S_n$ and then recombines the two beams, so both outcomes exit at the same place without being recorded.
\end{definition}
Filter $+\hat{z}$, pass it through a modified SG$x$ with neither path blocked, then measure $S_z$. Every atom comes out $+\hat{z}$: all $N_0/2$ end in $\ket{+\hat{z}}$.
\begin{result}
    The atoms cannot be going half through $+\hat{x}$ and half through $-\hat{x}$, since Experiment 3 showed each half alone gives 50/50. Instead the two paths \emph{interfere}: as with water or light waves, contributions from the two paths add and can cancel, here cancelling the $-\hat{z}$ outcome. No definite path can be assigned to an atom. Section~\ref{sec:states} makes this precise.
\end{result}

\section{Quantum States}
\label{sec:states}
\marginpar{September 11-18, 2026}

Throughout, an \emph{$n$-level system} is one whose measurements have $n$ possible outcomes. For spin $s$, $n = 2s+1$, so spin-$\tfrac12$ has $n = 2$.

\subsection{Kets, bras and bases}
\begin{definition}[Bra and inner product]
    Every ket (Def.~\ref{Ket-def}) $\ket{\psi}$ has a matching bra $\bra{\psi}$. A bra followed by a ket gives a complex number, the inner product $\braket{\phi|\psi} \in \mathbb{C}$.
\end{definition}
\begin{definition}[Orthonormal basis]
    The outcome states $\{\ket{j} : j = 1,\dots,n\}$ of one measurement form a basis that is orthonormal:
    \begin{equation}
        \braket{i|j} = \delta_{ij}.
    \end{equation}
    For spin-$\tfrac12$ this is $\braket{\pm\hat{z}|\pm\hat{z}} = 1$ and $\braket{\pm\hat{z}|\mp\hat{z}} = 0$, and the same holds for $\{\ket{\pm\hat{x}}\}$ or any other axis.
\end{definition}
\begin{remark}
    The label $j$ counts \emph{outcomes}, not axes. Choosing to measure $S_x$ gives $\ket{1} = \ket{+\hat{x}}$, $\ket{2} = \ket{-\hat{x}}$; choosing $S_z$ gives $\ket{1} = \ket{+\hat{z}}$, $\ket{2} = \ket{-\hat{z}}$. Either way $n = 2$. A ket may also carry several labels, usually the quantum numbers that pin it down: $\ket{s,m}$ for spin, so $\ket{\pm\hat{z}} = \ket{\tfrac12,\pm\tfrac12}$, and for the deuteron $\ket{1,+1}, \ket{1,0}, \ket{1,-1}$.
\end{remark}
\begin{definition}[Superposition]
    Any state is a linear combination of basis kets,
    \begin{equation}
        \ket{\psi} = \sum_{j=1}^{n} c_j\ket{j}, \qquad c_j \in \mathbb{C}; \qquad \text{spin-}\tfrac12: \ \ket{\psi} = c_{+}\ket{+\hat{z}} + c_{-}\ket{-\hat{z}}.
    \end{equation}
\end{definition}
\begin{proposition}[Coefficients are inner products]
    \begin{equation}
        c_j = \braket{j|\psi}.
    \end{equation}
\end{proposition}
\begin{proof}
    $\braket{i|\psi} = \sum_j c_j\braket{i|j} = \sum_j c_j\delta_{ij} = c_i$.
\end{proof}
\begin{definition}[Projector]
    The outer product (a ket times a bra) $P_j = \ket{j}\bra{j}$. It is an operator that keeps only the $\ket{j}$ part of a state,
    \[P_j\ket{\psi} = \ket{j}\braket{j|\psi} = c_j\ket{j},\]
    and projecting twice changes nothing: $P_j^2 = \ket{j}\braket{j|j}\bra{j} = P_j$.
\end{definition}
\begin{proposition}[Completeness relation]
    Because every state can be expanded in the basis, the projectors sum to the identity:
    \begin{equation}
        \id = \sum_{j=1}^{n}\ket{j}\bra{j}; \qquad \text{spin-}\tfrac12: \ \id = \ket{+\hat{n}}\bra{+\hat{n}} + \ket{-\hat{n}}\bra{-\hat{n}} \text{ for any axis } \hat{n}.
    \end{equation}
\end{proposition}
\begin{remark}[Inserting the identity]
    Writing $\id$ in this form in front of a state reproduces its expansion, $\ket{\psi} = \sum_j\ket{j}\braket{j|\psi}$. This trick is used in almost every derivation below.
\end{remark}

\subsection{Measurement and probability}
\begin{definition}[Observable]
    A measurable quantity $A$. Its possible outcomes $a_j$ each come with a state $\ket{a_j}$, and these states form an orthonormal basis. For spin-$\tfrac12$, $A$ can be $S_x$, $S_y$ or $S_z$, with outcomes $\pm\hbar/2$.
\end{definition}
\begin{principle}[Born rule]
    The probability that a system in $\ket{\psi}$ is found in $\ket{\phi}$ is
    \begin{equation}
        P(\phi\,|\,\psi) = \left|\braket{\phi|\psi}\right|^2 \in [0,1].
    \end{equation}
    In particular $P(A = a_j) = |\braket{a_j|\psi}|^2$, e.g.\ $P(S_z = \pm\hbar/2) = |c_\pm|^2$.
\end{principle}
\begin{definition}[Probability amplitude]
    The complex number $\braket{\phi|\psi}$ whose modulus squared is the probability.
\end{definition}
\begin{remark}
    The Born rule has no deeper derivation: it is a postulate that agrees with every experiment. As David Mermin put it, "shut up and calculate."
\end{remark}

\subsubsection{Bras of superpositions}
Write the bra with unknown coefficients, $\bra{\psi} = c_{+}'\bra{+\hat{z}} + c_{-}'\bra{-\hat{z}}$. By orthonormality,
\[\braket{\psi|\psi} = c_{+}'c_{+} + c_{-}'c_{-}.\]
A system in $\ket{\psi}$ is found in $\ket{\psi}$ with certainty, and it is always found in one of the two beams, so this must equal $|c_+|^2 + |c_-|^2 = 1$. That forces $c_{\pm}' = c_{\pm}^{*}$.
\begin{result}
    Going from a ket to its bra, conjugate the coefficients:
    \begin{equation}
        \bra{\psi} = \sum_j c_j^*\bra{j}, \qquad \braket{\phi|\psi} = \braket{\psi|\phi}^{*}.
    \end{equation}
\end{result}

\subsubsection{Normalization}
\begin{proposition}[Normalization]
    $\braket{\psi|\psi} = 1$ is the statement that the probabilities of all outcomes add to 1.
\end{proposition}
\begin{proof}
    \[\braket{\psi|\psi} = \sum_{i,j}c_i^*c_j\braket{a_i|a_j} = \sum_j|c_j|^2 = \sum_j P(A = a_j) = 1.\]
\end{proof}
\begin{example}
    For $\ket{\psi} = \frac{1}{\sqrt{3}}\ket{+\hat{z}} + i\sqrt{\frac{2}{3}}\ket{-\hat{z}}$, $c_+ = 1/\sqrt{3}$ and $c_- = i\sqrt{2/3}$, so $P(S_z = +\hbar/2) = 1/3$ and $P(S_z = -\hbar/2) = |i|^2\cdot 2/3 = 2/3$. They add to 1, so the state is normalized.
\end{example}

\subsubsection{Global and relative phase}
\begin{definition}[Global and relative phase]
    A \emph{global} phase multiplies the whole ket: $e^{i\alpha}\ket{\psi}$ with $\alpha \in \mathbb{R}$. A \emph{relative} phase sits between the terms of a superposition, like the $e^{i\delta}$ in $c_+\ket{+\hat{z}} + e^{i\delta}c_-\ket{-\hat{z}}$, and cannot be pulled out front.
\end{definition}
\begin{proposition}
    A global phase changes no probability: $\left|\braket{\phi|e^{i\alpha}\psi}\right|^2 = |e^{i\alpha}|^2\left|\braket{\phi|\psi}\right|^2 = \left|\braket{\phi|\psi}\right|^2$ for every $\ket{\phi}$.
\end{proposition}
\begin{result}
    $\ket{\psi}$ and $e^{i\alpha}\ket{\psi}$ are the same physical state, so global phases can be dropped. Relative phases cannot: they leave the $S_z$ probabilities alone but change the $S_x$ and $S_y$ ones (Section~\ref{sec:xy-states}).
\end{result}

\subsection{Expectation value and uncertainty}
\begin{definition}[Expectation value]
    The average outcome, weighted by the probabilities:
    \begin{equation}
        \braket{A} = \sum_j a_j\,P(A = a_j) = \sum_j a_j\left|\braket{a_j|\psi}\right|^2; \qquad \braket{S_z} = \frac{\hbar}{2}|c_+|^2 - \frac{\hbar}{2}|c_-|^2.
    \end{equation}
\end{definition}
\begin{remark}
    For $N$ measurements on identically prepared atoms, with $N_\pm$ giving $\pm\hbar/2$, the measured average is $\braket{S_z}_N = \frac{\hbar}{2}\frac{N_+ - N_-}{N}$, which tends to $\braket{S_z}$ as $N \to \infty$.
\end{remark}
\begin{definition}[Uncertainty]
    The standard deviation (typical spread of outcomes around the average):
    \begin{equation}
        (\Delta A)^2 = \braket{(A - \braket{A})^2} = \braket{A^2} - \braket{A}^2, \qquad \braket{A^2} = \sum_j a_j^2\left|\braket{a_j|\psi}\right|^2.
    \end{equation}
\end{definition}
\begin{remark}
    For spin-$\tfrac12$ both outcomes square to $\hbar^2/4$, so $\braket{S_z^2} = \hbar^2/4$ for every state.
\end{remark}
\begin{example}
    For $\ket{+\hat{x}}$, Experiment 3 gives both $S_z$ probabilities $1/2$, so $\braket{S_z} = 0$ and $\Delta S_z = \hbar/2$. The average is zero but every single measurement gives $\pm\hbar/2$: maximal spread.
\end{example}
\begin{example}
    For $\ket{\psi} = \sqrt{\tfrac{1}{3}}\ket{+\hat{z}} + i\sqrt{\tfrac{2}{3}}\ket{-\hat{z}}$,
    \[\braket{S_z} = \frac{\hbar}{2}\cdot\frac{1}{3} - \frac{\hbar}{2}\cdot\frac{2}{3} = -\frac{\hbar}{6}, \qquad (\Delta S_z)^2 = \frac{\hbar^2}{4} - \frac{\hbar^2}{36} = \frac{2}{9}\hbar^2, \qquad \Delta S_z = \frac{\sqrt{2}}{3}\hbar.\]
\end{example}
\begin{remark}[Shot noise]
    The scatter of a finite-sample average around the true expectation value,
    \[\left|\braket{S_z}_N - \braket{S_z}\right| \sim \frac{\Delta S_z}{\sqrt{N}}.\]
    It comes from the randomness of individual quantum outcomes, not from a faulty apparatus, and is absent only when $\Delta S_z = 0$, i.e.\ when the state has a definite $S_z$.
\end{remark}

\subsection{Spin states along \texorpdfstring{$x$}{x} and \texorpdfstring{$y$}{y}}
\label{sec:xy-states}

\subsubsection{The state \texorpdfstring{$\ket{+\hat{x}}$}{|+x>}}
Experiment 3 gives $|\braket{\pm\hat{z}|+\hat{x}}|^2 = \tfrac12$, so both amplitudes have modulus $1/\sqrt{2}$ with their own phases:
\[\ket{+\hat{x}} = \frac{e^{i\delta_{+}}}{\sqrt{2}}\left(\ket{+\hat{z}} + e^{i\delta}\ket{-\hat{z}}\right), \qquad \delta = \delta_{-} - \delta_{+}.\]
Dropping the global phase $e^{i\delta_+}$,
\[\ket{+\hat{x}} = \frac{1}{\sqrt{2}}\left(\ket{+\hat{z}} + e^{i\delta}\ket{-\hat{z}}\right).\]
Experiment 3 alone does not fix the relative phase $\delta$.

\subsubsection{Experiment 5: measuring \texorpdfstring{$S_y$}{Sy}}
\begin{definition}[Cyclic permutation]
    Relabelling the axes $x \to y \to z \to x$ (or the reverse). It is a rotation of the coordinate system, so no physics changes.
\end{definition}
Experiment 5 is Experiment 3 cyclically permuted: filter $+\hat{z}$, then measure $S_y$. By symmetry it also splits 50/50, so every cross-axis overlap has modulus squared $\tfrac12$:
\[\left|\braket{+\hat{x}|+\hat{z}}\right|^2 = \left|\braket{+\hat{y}|+\hat{z}}\right|^2 = \left|\braket{-\hat{x}|+\hat{y}}\right|^2 = \frac{1}{2}.\]
The same argument as for $\ket{+\hat{x}}$ gives, after dropping global phases,
\[\ket{\pm\hat{x}} = \frac{1}{\sqrt{2}}\left(\ket{+\hat{z}} \pm e^{i\delta}\ket{-\hat{z}}\right), \qquad \ket{\pm\hat{y}} = \frac{1}{\sqrt{2}}\left(\ket{+\hat{z}} \pm e^{i\gamma}\ket{-\hat{z}}\right).\]

\subsubsection{Fixing the phases}
Using orthonormality of the $\hat{z}$ basis,
\[\left|\braket{+\hat{x}|+\hat{y}}\right|^2 = \left|\frac{1}{2}\left(1 + e^{i(\gamma - \delta)}\right)\right|^2 = \frac{1}{2}\left(1 + \cos(\gamma - \delta)\right).\]
Setting this to $\tfrac12$ forces $\gamma - \delta = \pm\pi/2$. The rest is convention.
\begin{result}[Phase convention]
    With $\delta = 0$ and $\gamma = +\pi/2$,
    \begin{equation}
        \ket{\pm\hat{x}} = \frac{1}{\sqrt{2}}\left(\ket{+\hat{z}} \pm \ket{-\hat{z}}\right), \qquad
        \ket{\pm\hat{y}} = \frac{1}{\sqrt{2}}\left(\ket{+\hat{z}} \pm i\ket{-\hat{z}}\right),
    \end{equation}
    \begin{equation}
        \bra{\pm\hat{x}} = \frac{1}{\sqrt{2}}\left(\bra{+\hat{z}} \pm \bra{-\hat{z}}\right), \qquad \bra{\pm\hat{y}} = \frac{1}{\sqrt{2}}\left(\bra{+\hat{z}} \mp i\bra{-\hat{z}}\right).
    \end{equation}
\end{result}
\begin{remark}
    Complex coefficients are required: with real ones the relative phases could only be $0$ or $\pi$, and $\gamma - \delta = \pm\pi/2$ would be impossible.
\end{remark}
\begin{example}
    For $\ket{\psi} = \sqrt{\tfrac{1}{3}}\ket{+\hat{z}} + i\sqrt{\tfrac{2}{3}}\ket{-\hat{z}}$, compute $\braket{S_y}$. Using $\braket{\pm\hat{y}|+\hat{z}} = \tfrac{1}{\sqrt{2}}$ and $\braket{\pm\hat{y}|-\hat{z}} = \mp\tfrac{i}{\sqrt{2}}$,
    \[\braket{\pm\hat{y}|\psi} = \frac{1}{\sqrt{2}}\left(\sqrt{\tfrac{1}{3}} \pm \sqrt{\tfrac{2}{3}}\right), \qquad P\!\left(S_y = \pm\tfrac{\hbar}{2}\right) = \frac{1}{2} \pm \frac{\sqrt{2}}{3},\]
    which add to 1. Then
    \[\braket{S_y} = \frac{\hbar}{2}\left[\left(\frac{1}{2} + \frac{\sqrt{2}}{3}\right) - \left(\frac{1}{2} - \frac{\sqrt{2}}{3}\right)\right] = \frac{\sqrt{2}}{3}\hbar.\]
    The factor $i$ did nothing for $S_z$, but as a relative phase it is exactly what tilts the spin toward $+\hat{y}$.
\end{example}

\subsection{States as column vectors}
\begin{definition}[Representation of a state]
    In a basis, a ket is represented by the column of its coefficients and its bra by the conjugate row (the adjoint, $\dagger$). In the $S_z$ basis:
    \begin{equation}
        \ket{\psi} \xrightarrow{S_z} \psi = \begin{pmatrix} c_+ \\ c_- \end{pmatrix}, \qquad
        \bra{\psi} \xrightarrow{S_z} \psi^\dagger = \begin{pmatrix} c_+^* & c_-^* \end{pmatrix}, \qquad c_\pm = \braket{\pm\hat{z}|\psi}.
    \end{equation}
\end{definition}
\begin{remark}
    Write $\ket{\psi} \xrightarrow{S_z} \psi$, not $\ket{\psi} = \psi$: the ket is basis-independent, the column is not.
\end{remark}
\begin{example}
    \[\ket{\pm\hat{x}} \xrightarrow{S_z} \frac{1}{\sqrt{2}}\begin{pmatrix} 1 \\ \pm 1 \end{pmatrix}, \qquad \ket{\pm\hat{y}} \xrightarrow{S_z} \frac{1}{\sqrt{2}}\begin{pmatrix} 1 \\ \pm i \end{pmatrix}.\]
\end{example}
\begin{problem}[Other bases]
    In the $S_x$ basis, $\ket{\psi} \xrightarrow{S_x} \begin{pmatrix} \braket{+\hat{x}|\psi} \\ \braket{-\hat{x}|\psi} \end{pmatrix}$, and similarly for $S_y$. Write the column representing $\ket{+\hat{z}}$ in the $S_z$, $S_x$ and $S_y$ bases.
\end{problem}
\begin{proposition}[Inner product as a matrix product]
    If $\ket{\phi} \xrightarrow{S_z} \phi$ and $\ket{\psi} \xrightarrow{S_z} \psi$, then $\braket{\phi|\psi} = \phi^\dagger\psi$ (row times column), so $P(\phi\,|\,\psi) = \left|\phi^\dagger\psi\right|^2$ and $\braket{\psi|\psi} = \psi^\dagger\psi$.
\end{proposition}
\begin{remark}
    For a state given in another basis, replace each basis ket by its $S_z$ column and add. For $\ket{\psi} = a\ket{+\hat{y}} + b\ket{-\hat{y}}$,
    \[\psi \xrightarrow{S_z} \frac{a}{\sqrt{2}}\begin{pmatrix} 1 \\ i \end{pmatrix} + \frac{b}{\sqrt{2}}\begin{pmatrix} 1 \\ -i \end{pmatrix} = \frac{1}{\sqrt{2}}\begin{pmatrix} a+b \\ i(a-b) \end{pmatrix}.\]
    Checking $\psi^\dagger\psi = 1$ before going further catches most mistakes. A cleaner systematic method is in Section~\ref{sec:change-basis}.
\end{remark}
\begin{problem}[A probability by matrix mechanics]
    Using matrix mechanics, find $P(S_x = +\hbar/2)$ for the state $\ket{\psi} = \sqrt{\tfrac{1}{5}}\ket{+\hat{z}} + i\sqrt{\tfrac{4}{5}}\ket{-\hat{z}}$.
\end{problem}

\subsection{Operators as matrices}
\begin{definition}[Linear operator]
    A map $A$ from kets to kets that respects superpositions, $A\left(a\ket{\psi} + b\ket{\phi}\right) = aA\ket{\psi} + bA\ket{\phi}$. Every operator in this course is linear.
\end{definition}
\begin{definition}[Matrix of an operator]
    In a basis $\{\ket{j}\}$, $A$ is represented by the matrix of \emph{matrix elements}
    \begin{equation}
        A_{ij} = \bra{i}A\ket{j}.
    \end{equation}
\end{definition}
\begin{proposition}[Operator acting on a state]
    $A\ket{\psi}$ is represented by the matrix times the column, $A\psi$.
\end{proposition}
\begin{proof}
    The $k$th entry of the column for $A\ket{\psi}$ is $\bra{k}A\ket{\psi}$. Inserting the identity,
    \[\bra{k}A\ket{\psi} = \sum_j\bra{k}A\ket{j}\braket{j|\psi} = \sum_j A_{kj}\,c_j,\]
    which is row $k$ of $A$ times the column $\psi$.
\end{proof}
\begin{example}
    The projectors and identity in the $S_z$ basis:
    \[P_{+\hat{z}} \xrightarrow{S_z} \begin{pmatrix} 1 & 0 \\ 0 & 0 \end{pmatrix}, \qquad P_{-\hat{z}} \xrightarrow{S_z} \begin{pmatrix} 0 & 0 \\ 0 & 1 \end{pmatrix}, \qquad \id \xrightarrow{S_z} \begin{pmatrix} 1 & 0 \\ 0 & 1 \end{pmatrix},\]
    and indeed $P_{+\hat{z}}\psi = (c_+, 0)^T$.
\end{example}
\begin{definition}[Adjoint]
    The adjoint $A^\dagger$ is the conjugate transpose: $(A^\dagger)_{ij} = A_{ji}^*$. In operator form, writing $A = \sum_{i,j}A_{ij}\ket{i}\bra{j}$,
    \begin{equation}
        A^\dagger = \sum_{i,j}A_{ij}^*\ket{j}\bra{i}.
    \end{equation}
    For real entries it is just the transpose.
\end{definition}
\begin{proposition}[Bra of a transformed state]
    If $\ket{\psi'} = A\ket{\psi}$, then $\bra{\psi'} = \bra{\psi}A^\dagger$.
\end{proposition}
\begin{proof}
    The column of $\ket{\psi'}$ is $A\psi$, and its bra is the conjugate transpose, $(A\psi)^\dagger = \psi^\dagger A^\dagger$.
\end{proof}
\begin{proposition}[Adjoint of a product]
    \begin{equation}
        \left(AB\right)^\dagger = B^\dagger A^\dagger.
    \end{equation}
    The order reverses; $A^\dagger B^\dagger$ is a different matrix in general.
\end{proposition}

\subsection{Observables as operators}
\begin{definition}[Eigenstate and eigenvalue]
    $\ket{a}$ is an eigenstate of $A$ with eigenvalue $a$ if
    \begin{equation}
        A\ket{a} = a\ket{a},
    \end{equation}
    the familiar $Mv = \lambda v$ in matrix form. A general operator turns a state into a different state; eigenstates come back as themselves times a number. A basis made of eigenstates of $A$ is called an \emph{eigenbasis} of $A$.
\end{definition}
\begin{proposition}[Operator of an observable]
    For an observable with outcomes $a_j$ and states $\ket{a_j}$, define
    \begin{equation}
        \hat{A} = \sum_j a_j\ket{a_j}\bra{a_j}.
    \end{equation}
    Then $\hat{A}\ket{a_j} = a_j\ket{a_j}$ (the outcomes are its eigenvalues, the outcome states its eigenstates), and
    \begin{equation}
        \braket{A} = \bra{\psi}\hat{A}\ket{\psi}, \qquad \braket{A^2} = \bra{\psi}\hat{A}^2\ket{\psi}.
    \end{equation}
\end{proposition}
\begin{proof}
    $\hat{A}\ket{a_k} = \sum_j a_j\ket{a_j}\delta_{jk} = a_k\ket{a_k}$, and
    $\bra{\psi}\hat{A}\ket{\psi} = \sum_j a_j\braket{\psi|a_j}\braket{a_j|\psi} = \sum_j a_j\left|\braket{a_j|\psi}\right|^2 = \braket{A}$. Likewise $\hat{A}^2 = \sum_j a_j^2\ket{a_j}\bra{a_j}$.
\end{proof}
\begin{proposition}[An operator is diagonal in its own eigenbasis]
    \begin{equation}
        \bra{a_i}\hat{A}\ket{a_j} = a_j\delta_{ij}, \qquad \hat{A} \xrightarrow{\ket{a_j}} D = \begin{pmatrix} a_1 & & 0 \\ & \ddots & \\ 0 & & a_n \end{pmatrix}.
    \end{equation}
\end{proposition}
\begin{definition}[Spin operators]
    Applying this to spin, for $\hat{n} \in \{\hat{x},\hat{y},\hat{z}\}$,
    \begin{equation}
        \hat{S}_n = \frac{\hbar}{2}\left(\ket{+\hat{n}}\bra{+\hat{n}} - \ket{-\hat{n}}\bra{-\hat{n}}\right), \qquad \hat{S}_n\ket{\pm\hat{n}} = \pm\frac{\hbar}{2}\ket{\pm\hat{n}}.
    \end{equation}
    In the $S_z$ basis its matrix elements are
    \[\bra{i}\hat{S}_n\ket{j} = \frac{\hbar}{2}\Big(\braket{i|+\hat{n}}\braket{+\hat{n}|j} - \braket{i|-\hat{n}}\braket{-\hat{n}|j}\Big), \qquad i,j \in \{+\hat{z},-\hat{z}\}.\]
\end{definition}
\begin{problem}[The matrix of $\hat{S}_z$]
    Find the matrix representation of $\hat{S}_z$ in the $S_z$ basis.
\end{problem}
\begin{problem}[The matrix of $\hat{S}_x$]
    Find the matrix representation of $\hat{S}_x$ in the $S_z$ basis.
\end{problem}
\begin{problem}[The matrix of $\hat{S}_y$]
    Find the matrix representation of $\hat{S}_y$ in the $S_z$ basis.
\end{problem}
\begin{remark}[Shortcut]
    Write each projector as a column times its conjugate row, e.g.
    \[\ket{\pm\hat{x}}\bra{\pm\hat{x}} \xrightarrow{S_z} \frac{1}{2}\begin{pmatrix} 1 \\ \pm1 \end{pmatrix}\begin{pmatrix} 1 & \pm1 \end{pmatrix} = \frac{1}{2}\begin{pmatrix} 1 & \pm1 \\ \pm1 & 1 \end{pmatrix},\]
    then subtract and multiply by $\hbar/2$.
\end{remark}
\begin{definition}[Pauli matrices]
    The results, all in the $S_z$ basis:
    \begin{equation}
        \hat{S}_j = \frac{\hbar}{2}\sigma_j, \qquad
        \sigma_x = \begin{pmatrix} 0 & 1 \\ 1 & 0 \end{pmatrix}, \quad
        \sigma_y = \begin{pmatrix} 0 & -i \\ i & 0 \end{pmatrix}, \quad
        \sigma_z = \begin{pmatrix} 1 & 0 \\ 0 & -1 \end{pmatrix}.
    \end{equation}
    Each equals its own adjoint and squares to $\id$, so $\hat{S}_j^2 = \tfrac{\hbar^2}{4}\id$.
\end{definition}
\begin{problem}[Expectation value and uncertainty by matrix mechanics]
    For $\ket{\psi} = \sqrt{\tfrac{1}{5}}\ket{+\hat{y}} + i\sqrt{\tfrac{4}{5}}\ket{-\hat{y}}$, find $\braket{S_z}$ and $\Delta S_z$ using matrix mechanics.
\end{problem}

\subsection{Change of basis}
\label{sec:change-basis}
Take an $n$-level system with two orthonormal bases, $\{\ket{j}\}$ and $\{\ket{\alpha}\}$, $j,\alpha = 1,\dots,n$. Within each basis the overlaps are Kronecker deltas, but across bases they are not: $\braket{+\hat{x}|+\hat{z}} = \tfrac{1}{\sqrt{2}}$ is neither $0$ nor $1$. The same ket then has two columns,
\[\ket{\psi} \xrightarrow{\ket{j}} \psi^{j} = \begin{pmatrix} \braket{1|\psi} \\ \vdots \\ \braket{n|\psi} \end{pmatrix}, \qquad
\ket{\psi} \xrightarrow{\ket{\alpha}} \psi^{\alpha} = \begin{pmatrix} \braket{\alpha{=}1|\psi} \\ \vdots \\ \braket{\alpha{=}n|\psi} \end{pmatrix}.\]
Inserting $\id = \sum_\alpha\ket{\alpha}\bra{\alpha}$ into each $\ket{j}$ amplitude,
\[\braket{j|\psi} = \sum_{\alpha}\braket{j|\alpha}\braket{\alpha|\psi}.\]
\begin{result}[Change-of-basis matrix]
    \begin{equation}
        \psi^{j} = U\psi^{\alpha}, \qquad \psi^{\alpha} = U^\dagger\psi^{j}, \qquad U_{j\alpha} = \braket{j|\alpha}.
    \end{equation}
    The columns of $U$ are the new basis kets $\ket{\alpha}$ written in the old basis $\{\ket{j}\}$.
\end{result}
\begin{definition}[Unitary operator]
    $U$ is unitary if $U^\dagger U = UU^\dagger = \id$, i.e.\ its inverse is its adjoint.
\end{definition}
\begin{theorem}
    The change-of-basis matrix $U$ is unitary.
\end{theorem}
\begin{proof}
    With $(U^\dagger)_{\alpha j} = \braket{j|\alpha}^* = \braket{\alpha|j}$ and completeness of $\{\ket{j}\}$,
    \[(U^\dagger U)_{\alpha\beta} = \sum_j\braket{\alpha|j}\braket{j|\beta} = \braket{\alpha|\beta} = \delta_{\alpha\beta}.\]
    Completeness of $\{\ket{\alpha}\}$ gives $UU^\dagger = \id$ the same way.
\end{proof}
\begin{remark}
    So normalization does not depend on the basis: $\psi^{j\dagger}\psi^{j} = \psi^{\alpha\dagger}U^\dagger U\psi^{\alpha} = \psi^{\alpha\dagger}\psi^{\alpha}$.
\end{remark}
\begin{example}[From $S_x$ to $S_z$]
    With $\ket{j} = \ket{\pm\hat{z}}$ and $\ket{\alpha} = \ket{\pm\hat{x}}$, the columns of $U$ are the $S_z$ columns of $\ket{\pm\hat{x}}$:
    \begin{equation}
        \label{eq:Uzx}
        U = \begin{pmatrix} \braket{+\hat{z}|+\hat{x}} & \braket{+\hat{z}|-\hat{x}} \\ \braket{-\hat{z}|+\hat{x}} & \braket{-\hat{z}|-\hat{x}} \end{pmatrix} = \frac{1}{\sqrt{2}}\begin{pmatrix} 1 & 1 \\ 1 & -1 \end{pmatrix}, \qquad \psi^z = U\psi^x.
    \end{equation}
    It is real and symmetric, so $U^\dagger = U$. The matrix taking the $S_y$ basis to $S_z$ has columns $\frac{1}{\sqrt{2}}(1, \pm i)^T$.
\end{example}
\begin{remark}[Lecture notation]
    Bill calls this matrix $R\left(\tfrac{\pi}{2}\hat{y}\right)$ and writes
    \[\psi^z = R\left(\tfrac{\pi}{2}\hat{y}\right)\psi^x, \qquad \psi^x = R^\dagger\left(\tfrac{\pi}{2}\hat{y}\right)\psi^z,\]
    because rotating by $90^\circ$ about $\hat{y}$ takes $\ket{\pm\hat{z}}$ to $\ket{\pm\hat{x}}$. It agrees with the true rotation matrix up to the phase of one basis ket (see the phase caveat in Section~\ref{sec:any-axis}).
\end{remark}
\begin{problem}[Expectation value from the $S_x$ basis]
    For $\ket{\psi} = \frac{1}{\sqrt{3}}\ket{+\hat{x}} + i\sqrt{\frac{2}{3}}\ket{-\hat{x}}$, find $\braket{S_z} = \psi^{z\dagger}\hat{S}_z\psi^z$ using matrix mechanics.
\end{problem}

\subsubsection{Operators in another basis}
Let $\{\ket{j}\} = \{\ket{a_j}\}$ be the eigenbasis of $\hat{A}$, so $\hat{A} \xrightarrow{\ket{a_j}} D$ (diagonal), and $U_{j\alpha} = \braket{a_j|\alpha}$.
\begin{proposition}[Similarity transformation]
    \begin{equation}
        \hat{A} \xrightarrow{\ket{\alpha}} A^{\alpha} = U^\dagger D\,U.
    \end{equation}
    Rewriting the same operator in a new basis this way is called a similarity transformation.
\end{proposition}
\begin{proof}
    Insert the identity twice:
    \[\bra{\alpha}\hat{A}\ket{\beta} = \sum_{i,j}\braket{\alpha|a_i}\bra{a_i}\hat{A}\ket{a_j}\braket{a_j|\beta} = \sum_{i,j}(U^\dagger)_{\alpha i}D_{ij}U_{j\beta} = (U^\dagger D\,U)_{\alpha\beta}.\]
\end{proof}
\begin{result}[Expectation value in any basis]
    \begin{equation}
        \braket{A} = \psi^{j\dagger}D\,\psi^{j} = \psi^{\alpha\dagger}\left[U^\dagger D\,U\right]\psi^{\alpha}.
    \end{equation}
    Either move the state to the eigenbasis, or move the operator to the state's basis. Both give the same number.
\end{result}
\begin{problem}[$\hat{S}_z$ in the $S_x$ basis]

\section{Rotations}
\marginpar{September 21-28, 2026}

\subsection{The rotation operator}
\begin{definition}[Rotation operator]
    $R(\theta\hat{n})$ is the linear operator that rotates a state by angle $\theta$ about the axis $\hat{n}$: $\ket{\psi'} = R(\theta\hat{n})\ket{\psi}$.
\end{definition}
\begin{example}
    $R\left(\tfrac{\pi}{2}\hat{y}\right)$ turns $\hat{z}$ into $\hat{x}$, so it takes $\ket{\pm\hat{z}}$ to $\ket{\pm\hat{x}}$ up to phase, and by linearity
    \[R\left(\tfrac{\pi}{2}\hat{y}\right)\left(c_+\ket{+\hat{z}} + c_-\ket{-\hat{z}}\right) = c_+\ket{+\hat{x}} + c_-\ket{-\hat{x}}.\]
    Up to a phase on one column, its $S_z$ matrix is the change-of-basis matrix $U$ of Eq.~\eqref{eq:Uzx} (exact form in Section~\ref{sec:any-axis}).
\end{example}
\begin{remark}[Why the bra carries $R^\dagger$]
    By the bra-of-a-transformed-state proposition, $\bra{\psi'} = \bra{\psi}R^\dagger$. Using $\bra{\psi}R$ instead fails: with $\ket{+\hat{x}} = R\left(\tfrac{\pi}{2}\hat{y}\right)\ket{+\hat{z}}$ it would give
    \[\braket{+\hat{x}|+\hat{x}} = \bra{+\hat{z}}R\left(\tfrac{\pi}{2}\hat{y}\right)R\left(\tfrac{\pi}{2}\hat{y}\right)\ket{+\hat{z}} = \bra{+\hat{z}}R(\pi\hat{y})\ket{+\hat{z}} \propto \braket{+\hat{z}|-\hat{z}} = 0,\]
    since a $\pi$ rotation about $\hat{y}$ flips $+\hat{z}$ to $-\hat{z}$. That contradicts $\braket{+\hat{x}|+\hat{x}} = 1$.
\end{remark}
\begin{theorem}
    Rotation operators are unitary: $R^\dagger R = RR^\dagger = \id$.
\end{theorem}
\begin{proof}
    A rotation must preserve normalization: $\braket{\psi'|\psi'} = \bra{\psi}R^\dagger R\ket{\psi} = 1$ for every $\ket{\psi}$, which forces $R^\dagger R = \id$.
\end{proof}
\begin{remark}
    If a unitary $U$ maps $\ket{\psi}$ to $c\ket{\phi}$ with $\ket{\phi}$ normalized, then $1 = \bra{\psi}U^\dagger U\ket{\psi} = |c|^2$, so $c = e^{i\varphi}$. In particular, eigenvalues of a unitary are pure phases, and $U$ leaves a physical state unchanged only when that state is an eigenstate of $U$.
\end{remark}

\subsection{Generators of rotations}
\label{sec:generators}
Rotations about the same axis add, $R(\theta_1\hat{z})R(\theta_2\hat{z}) = R\big((\theta_1+\theta_2)\hat{z}\big)$, so a finite rotation is $N \gg 1$ small ones,
\begin{equation}
    R(\theta\hat{z}) = \left[R\left(\tfrac{\theta}{N}\hat{z}\right)\right]^{N}.
\end{equation}
Taylor expanding one small step about the identity and naming the first-order operator $-iJ_z/\hbar$,
\begin{equation}
    R\left(\tfrac{\theta}{N}\hat{z}\right) = \id - \frac{i}{\hbar}J_z\frac{\theta}{N} + \mathcal{O}\!\left(\frac{1}{N^2}\right).
\end{equation}
\begin{result}
    By the problem ``The exponential as a limit'',
    \begin{equation}
        R(\theta\hat{z}) = \lim_{N\to\infty}\left(\id - \frac{i}{\hbar}J_z\frac{\theta}{N}\right)^{N} = e^{-iJ_z\theta/\hbar},
    \end{equation}
    and likewise $R(\theta\hat{n}) = e^{-iJ_n\theta/\hbar}$ for any axis.
\end{result}
\begin{definition}[Generator of rotations]
    The operator $J_z$ above: the thing that, exponentiated, produces rotations about $\hat{z}$. It has units of $\hbar$, i.e.\ of angular momentum ($[\hbar] = [\vec{r}\times\vec{p}]$), so $\vec{J}$ is identified as the angular momentum operator. For spin-$\tfrac12$, $\vec{J} = \vec{S}$ and $R(\theta\hat{n}) = e^{-i\theta\hat{S}_n/\hbar}$.
\end{definition}
\begin{definition}[Hermitian operator]
    $A$ is Hermitian if it equals its own adjoint, $A = A^\dagger$ (e.g.\ the Pauli matrices).
\end{definition}
\begin{proposition}
    The generator $J_z$ is Hermitian.
\end{proposition}
\begin{proof}
    To first order in $\theta$, $R = \id - i\theta J_z/\hbar$ and $R^\dagger = \id + i\theta J_z^\dagger/\hbar$. Unitarity gives
    \[\id = R^\dagger R = \id + \frac{i\theta}{\hbar}\left(J_z^\dagger - J_z\right) + \mathcal{O}(\theta^2),\]
    so $J_z^\dagger = J_z$.
\end{proof}
\begin{principle}
    Every observable is represented by a Hermitian operator.
\end{principle}
\begin{proposition}
    The eigenvalues of a Hermitian operator are real, so measurement outcomes are real.
\end{proposition}
\begin{proof}
    If $A\ket{a} = a\ket{a}$, the bra is $\bra{a}A^\dagger = a^*\bra{a}$, so $a\braket{a|a} = \bra{a}A\ket{a} = \bra{a}A^\dagger\ket{a} = a^*\braket{a|a}$, and $a = a^*$.
\end{proof}

\subsection{Functions of an operator}
\begin{proposition}[Function of an operator on an eigenstate]
    \label{prop:function-eigenstate}
    If $\hat{A}\ket{a} = a\ket{a}$ and $f$ has a Taylor series, then
    \begin{equation}
        f(\hat{A})\ket{a} = f(a)\ket{a}.
    \end{equation}
\end{proposition}
\begin{proof}
    $f(\hat{A})$ is defined by its Taylor series, and $\hat{A}^n\ket{a} = a^n\ket{a}$, so term by term
    \[f(\hat{A})\ket{a} = \sum_{n}\frac{f^{(n)}(0)}{n!}\hat{A}^n\ket{a} = \sum_{n}\frac{f^{(n)}(0)}{n!}a^n\ket{a} = f(a)\ket{a}.\]
\end{proof}
\begin{proposition}[Spectral form]
    Writing an operator as a sum of its eigenvalues times the projectors onto its eigenstates is its spectral form. Inserting $\id = \sum_j\ket{a_j}\bra{a_j}$ and using Proposition~\ref{prop:function-eigenstate},
    \begin{equation}
        f(\hat{A}) = \sum_j f(a_j)\ket{a_j}\bra{a_j}.
    \end{equation}
    With $f(x) = x$ this is the observable operator $\hat{A} = \sum_j a_j\ket{a_j}\bra{a_j}$.
\end{proposition}

\subsubsection{Rotations about \texorpdfstring{$\hat{z}$}{z}}
\begin{result}
    With $f(x) = e^{-i\phi x/\hbar}$ and $\hat{S}_z\ket{\pm\hat{z}} = \pm\frac{\hbar}{2}\ket{\pm\hat{z}}$,
    \begin{equation}
        R(\phi\hat{z})\ket{\pm\hat{z}} = e^{\mp i\phi/2}\ket{\pm\hat{z}}, \qquad R(\phi\hat{z}) \xrightarrow{S_z} \begin{pmatrix} e^{-i\phi/2} & 0 \\ 0 & e^{i\phi/2} \end{pmatrix}.
    \end{equation}
    This makes sense: a rotation about $\hat{z}$ cannot change $S_z$, so it can only attach phases to the $S_z$ states. In other bases the matrix is not diagonal.
\end{result}
\begin{remark}
    A full turn, $\phi = 2\pi$, gives $R = -\id$: a spin-$\tfrac12$ state needs $4\pi$ to return to itself. The sign is a global phase, so it is invisible on its own.
\end{remark}

\subsection{Rotations about any axis}
\label{sec:any-axis}
\begin{proposition}[Spectral form of a rotation]
    For $\hat{n} \in \{\hat{x},\hat{y},\hat{z}\}$,
    \begin{equation}
        R(\phi\hat{n}) = e^{-i\phi/2}\ket{+\hat{n}}\bra{+\hat{n}} + e^{i\phi/2}\ket{-\hat{n}}\bra{-\hat{n}}.
    \end{equation}
\end{proposition}
\begin{proof}
    Spectral form with $f(x) = e^{-i\phi x/\hbar}$, $\hat{A} = \hat{S}_n$, eigenvalues $\pm\hbar/2$.
\end{proof}
\begin{remark}[Recipe]
    In any basis $\{\ket{i}\}$ this reads $V D V^\dagger$, with $D = \mathrm{diag}\left(e^{-i\phi/2}, e^{i\phi/2}\right)$ and $V$ the matrix whose columns are $\ket{\pm\hat{n}}$ written in $\{\ket{i}\}$. It is the similarity transformation with the roles of the two bases swapped.
\end{remark}
\begin{proposition}[$R(\phi\hat{x})$ in the $S_z$ basis]
    \begin{equation}
        R(\phi\hat{x}) \xrightarrow{S_z} \begin{pmatrix} \cos\frac{\phi}{2} & -i\sin\frac{\phi}{2} \\ -i\sin\frac{\phi}{2} & \cos\frac{\phi}{2} \end{pmatrix}.
    \end{equation}
\end{proposition}
\begin{proof}
    Here $V = \frac{1}{\sqrt{2}}\begin{pmatrix} 1 & 1 \\ 1 & -1 \end{pmatrix} = V^\dagger$, so
    \[R(\phi\hat{x}) \xrightarrow{S_z} \frac{1}{2}\begin{pmatrix} 1 & 1 \\ 1 & -1 \end{pmatrix}\begin{pmatrix} e^{-i\phi/2} & 0 \\ 0 & e^{i\phi/2} \end{pmatrix}\begin{pmatrix} 1 & 1 \\ 1 & -1 \end{pmatrix} = \frac{1}{2}\begin{pmatrix} e^{-i\phi/2} + e^{i\phi/2} & e^{-i\phi/2} - e^{i\phi/2} \\ e^{-i\phi/2} - e^{i\phi/2} & e^{-i\phi/2} + e^{i\phi/2} \end{pmatrix},\]
    and Euler's formula gives $2\cos\frac{\phi}{2}$ and $-2i\sin\frac{\phi}{2}$.
\end{proof}
\begin{remark}
    This is the same matrix as $R(\phi\hat{z})$ in the $S_x$ basis. It is also the same as $R(\phi\hat{z})$ in the $S_y$ basis (PS2 Q8): with $\ket{\pm\hat{y}} = \frac{1}{\sqrt{2}}\left(\ket{+\hat{z}} \pm i\ket{-\hat{z}}\right)$,
    \[\bra{\pm\hat{y}}R(\phi\hat{z})\ket{\pm\hat{y}} = \cos\frac{\phi}{2}, \qquad \bra{\pm\hat{y}}R(\phi\hat{z})\ket{\mp\hat{y}} = -i\sin\frac{\phi}{2},\]
    so in both bases
    \[R(\phi\hat{z}) \longrightarrow \begin{pmatrix} \cos\frac{\phi}{2} & -i\sin\frac{\phi}{2} \\ -i\sin\frac{\phi}{2} & \cos\frac{\phi}{2} \end{pmatrix}.\]
\end{remark}
\begin{proposition}[$R(\phi\hat{y})$ in the $S_z$ basis]
    \begin{equation}
        R(\phi\hat{y}) \xrightarrow{S_z} \begin{pmatrix} \cos\frac{\phi}{2} & -\sin\frac{\phi}{2} \\ \sin\frac{\phi}{2} & \cos\frac{\phi}{2} \end{pmatrix}.
    \end{equation}
\end{proposition}
\begin{proof}
    Now $V = \frac{1}{\sqrt{2}}\begin{pmatrix} 1 & 1 \\ i & -i \end{pmatrix}$ and $V^\dagger = \frac{1}{\sqrt{2}}\begin{pmatrix} 1 & -i \\ 1 & i \end{pmatrix} \neq V$. Then
    \[R(\phi\hat{y}) \xrightarrow{S_z} \frac{1}{2}\begin{pmatrix} e^{-i\phi/2} + e^{i\phi/2} & -i\left(e^{-i\phi/2} - e^{i\phi/2}\right) \\ i\left(e^{-i\phi/2} - e^{i\phi/2}\right) & e^{-i\phi/2} + e^{i\phi/2} \end{pmatrix},\]
    and with $e^{-i\phi/2} - e^{i\phi/2} = -2i\sin\frac{\phi}{2}$ the off-diagonal entries are $\mp\sin\frac{\phi}{2}$.
\end{proof}
\begin{remark}
    This matrix is real, so $R(\phi\hat{y})$ never introduces factors of $i$ into an $S_z$ column. At $\phi = \pi$ it gives $R(\pi\hat{y})\ket{+\hat{z}} = \ket{-\hat{z}}$, the flip used above to show bras carry $R^\dagger$.
\end{remark}
\begin{remark}[Phase caveat]
    At $\phi = \pi/2$,
    \[R\left(\tfrac{\pi}{2}\hat{y}\right) \xrightarrow{S_z} \frac{1}{\sqrt{2}}\begin{pmatrix} 1 & -1 \\ 1 & 1 \end{pmatrix},\]
    whose second column is $-\ket{-\hat{x}}$, while $U$ in Eq.~\eqref{eq:Uzx} has $+\ket{-\hat{x}}$. They differ only by the phase of one basis ket, which is a free choice, so both are valid.
\end{remark}

\section{Angular Momentum}
\marginpar{September 28, 2026}

\subsection{Rotations about different axes}
\begin{definition}[Commutator]
    \begin{equation}
        [A,B] = AB - BA.
    \end{equation}
    $A$ and $B$ commute if $[A,B] = 0$, i.e.\ the order in which they act does not matter.
\end{definition}
Rotations about different axes do not commute: $R(\theta\hat{x})R(\theta\hat{y}) \neq R(\theta\hat{y})R(\theta\hat{x})$.
\begin{remark}
    Rotate a textbook by $90^\circ$ about $\hat{x}$ then about $\hat{y}$ (right-hand rule). Repeat from the same start with $\hat{y}$ first. The spine ends up pointing in different directions.
\end{remark}
\begin{result}[Angular momentum commutation relations]
    \begin{equation}
        [J_x,J_y] = i\hbar J_z, \qquad [J_y,J_z] = i\hbar J_x, \qquad [J_z,J_x] = i\hbar J_y,
    \end{equation}
    or compactly $\vec{J}\times\vec{J} = i\hbar\vec{J}$, with $\vec{J} = J_x\hat{x} + J_y\hat{y} + J_z\hat{z}$.
\end{result}

\subsection{Compatible observables}
\begin{definition}[Degenerate eigenvalue]
    An eigenvalue is degenerate if more than one independent eigenstate has it. Those eigenstates span a \emph{subspace}: a smaller vector space inside the full one, closed under linear combinations.
\end{definition}
\begin{theorem}
    If $[A,B] = 0$, then $A$ and $B$ have a common eigenbasis.
\end{theorem}
\begin{proof}
    Let $A\ket{a} = a\ket{a}$. Since $AB = BA$,
    \[A\big(B\ket{a}\big) = BA\ket{a} = a\big(B\ket{a}\big),\]
    so $B\ket{a}$ is also an eigenstate of $A$ with eigenvalue $a$.

    \emph{Nondegenerate case.} If $\ket{a}$ is the only state with eigenvalue $a$, then $B\ket{a} = b\ket{a}$, so $\ket{a}$ is an eigenstate of both.

    \emph{Degenerate case.} If $n$ states $\ket{a,i}$ share eigenvalue $a$, then $B$ maps their subspace into itself, so there it is an $n\times n$ matrix $B_{ij} = \bra{a,i}B\ket{a,j}$. Diagonalizing it (finding the basis in which it is diagonal) gives eigenstates
    \[\ket{a,b_j} = \sum_{i}\ket{a,i}\braket{a,i|a,b_j}, \qquad A\ket{a,b_j} = a\ket{a,b_j}, \qquad B\ket{a,b_j} = b_j\ket{a,b_j},\]
    which are still eigenstates of $A$ because they are combinations of states with the same eigenvalue $a$.
\end{proof}
\begin{definition}[Compatible observables]
    Observables that commute, and so can have definite values at the same time. Otherwise they are incompatible.
\end{definition}
\begin{remark}
    $[J_x,J_y] = i\hbar J_z \neq 0$, so the components of angular momentum are incompatible. This is the operator version of Experiment 3.
\end{remark}

\subsection{Total angular momentum}
\begin{definition}[Square of the angular momentum]
    \begin{equation}
        \vec{J}^2 = J_x^2 + J_y^2 + J_z^2.
    \end{equation}
\end{definition}
\begin{proposition}
    $\vec{J}^2$ commutes with every component of $\vec{J}$; in particular $[\vec{J}^2, J_z] = 0$.
\end{proposition}
\begin{proof}
    Using $[A,BC] = B[A,C] + [A,B]C$:
    \begin{align*}
        [J_z, J_x^2] &= J_x[J_z,J_x] + [J_z,J_x]J_x = i\hbar\left(J_xJ_y + J_yJ_x\right), \\
        [J_z, J_y^2] &= J_y[J_z,J_y] + [J_z,J_y]J_y = -i\hbar\left(J_yJ_x + J_xJ_y\right), \\
        [J_z, J_z^2] &= 0,
    \end{align*}
    using $[J_z,J_y] = -i\hbar J_x$. The first two cancel, so $[J_z,\vec{J}^2] = 0$. The same steps work for $x$ and $y$.
\end{proof}
\begin{remark}
    $\vec{J}^2$ and $J_z$ are compatible, so they share an eigenbasis.
\end{remark}

\section{Solutions}

\begin{solution}[The exponential as a limit]
    Take the logarithm:
    \begin{equation*}
        \ln\left(1+\frac{x}{N}\right)^{N} = N\ln\left(1+\frac{x}{N}\right).
    \end{equation*}
    With $\ln(1+u) = u - \frac{u^2}{2} + \frac{u^3}{3} - \cdots$ and $u = x/N$,
    \begin{equation*}
        N\ln\left(1+\frac{x}{N}\right) = x - \frac{x^2}{2N} + \frac{x^3}{3N^2} - \cdots.
    \end{equation*}
    Every term after the first vanishes as $N \to \infty$, so the logarithm tends to $x$ and
    \begin{equation*}
        \lim_{N\to\infty}\left(1+\frac{x}{N}\right)^{N} = e^{x}.
    \end{equation*}
\end{solution}

\begin{solution}[Other bases]
    \begin{align*}
        \ket{+\hat{z}} &\xrightarrow{S_z} \begin{pmatrix} 1 \\ 0 \end{pmatrix}, \\
        \ket{+\hat{z}} &\xrightarrow{S_x} \begin{pmatrix} \braket{+\hat{x}|+\hat{z}} \\ \braket{-\hat{x}|+\hat{z}} \end{pmatrix} = \frac{1}{\sqrt{2}}\begin{pmatrix} 1 \\ 1 \end{pmatrix}, \\
        \ket{+\hat{z}} &\xrightarrow{S_y} \begin{pmatrix} \braket{+\hat{y}|+\hat{z}} \\ \braket{-\hat{y}|+\hat{z}} \end{pmatrix} = \frac{1}{\sqrt{2}}\begin{pmatrix} 1 \\ 1 \end{pmatrix}.
    \end{align*}
\end{solution}

\begin{solution}[A probability by matrix mechanics]
    \[\psi \xrightarrow{S_z} \frac{1}{\sqrt{5}}\begin{pmatrix} 1 \\ 2i \end{pmatrix}, \qquad \bra{+\hat{x}} \xrightarrow{S_z} \frac{1}{\sqrt{2}}\begin{pmatrix} 1 & 1 \end{pmatrix},\]
    \[\braket{+\hat{x}|\psi} = \frac{1+2i}{\sqrt{10}}, \qquad P\!\left(S_x = +\tfrac{\hbar}{2}\right) = \frac{|1+2i|^2}{10} = \frac{1}{2}.\]
    Take the modulus of the whole sum, not of each term separately.
\end{solution}

\begin{solution}[The matrix of $\hat{S}_z$]
    Every overlap is $0$ or $1$, so only the diagonal survives:
    \[\bra{\pm\hat{z}}\hat{S}_z\ket{\pm\hat{z}} = \pm\frac{\hbar}{2}, \qquad \hat{S}_z \xrightarrow{S_z} \frac{\hbar}{2}\begin{pmatrix} 1 & 0 \\ 0 & -1 \end{pmatrix}.\]
    An operator is diagonal in its own eigenbasis, with its outcomes on the diagonal.
\end{solution}

\begin{solution}[The matrix of $\hat{S}_x$]
    The overlaps are $\braket{+\hat{z}|\pm\hat{x}} = \braket{-\hat{z}|+\hat{x}} = \tfrac{1}{\sqrt{2}}$ and $\braket{-\hat{z}|-\hat{x}} = -\tfrac{1}{\sqrt{2}}$, all real. Then
    \begin{align*}
        \bra{+\hat{z}}\hat{S}_x\ket{+\hat{z}} &= \frac{\hbar}{2}\left(\tfrac{1}{2} - \tfrac{1}{2}\right) = 0, &
        \bra{+\hat{z}}\hat{S}_x\ket{-\hat{z}} &= \frac{\hbar}{2}\left(\tfrac{1}{2} + \tfrac{1}{2}\right) = \frac{\hbar}{2}, \\
        \bra{-\hat{z}}\hat{S}_x\ket{+\hat{z}} &= \frac{\hbar}{2}\left(\tfrac{1}{2} + \tfrac{1}{2}\right) = \frac{\hbar}{2}, &
        \bra{-\hat{z}}\hat{S}_x\ket{-\hat{z}} &= \frac{\hbar}{2}\left(\tfrac{1}{2} - \tfrac{1}{2}\right) = 0,
    \end{align*}
    where the $+$ signs come from the minus in $\braket{-\hat{z}|-\hat{x}}$ meeting the minus in front of the second projector. So
    \[\hat{S}_x \xrightarrow{S_z} \frac{\hbar}{2}\begin{pmatrix} 0 & 1 \\ 1 & 0 \end{pmatrix}.\]
\end{solution}

\begin{solution}[The matrix of $\hat{S}_y$]
    The overlaps are complex and the bra side is conjugated: $\braket{-\hat{z}|\pm\hat{y}} = \pm\tfrac{i}{\sqrt{2}}$, $\braket{\pm\hat{y}|-\hat{z}} = \mp\tfrac{i}{\sqrt{2}}$, and $\braket{+\hat{z}|\pm\hat{y}} = \braket{\pm\hat{y}|+\hat{z}} = \tfrac{1}{\sqrt{2}}$. Then
    \begin{align*}
        \bra{+\hat{z}}\hat{S}_y\ket{-\hat{z}} &= \frac{\hbar}{2}\left(\tfrac{1}{\sqrt{2}}\cdot\tfrac{-i}{\sqrt{2}} - \tfrac{1}{\sqrt{2}}\cdot\tfrac{i}{\sqrt{2}}\right) = -\frac{i\hbar}{2}, \\
        \bra{-\hat{z}}\hat{S}_y\ket{+\hat{z}} &= \frac{\hbar}{2}\left(\tfrac{i}{\sqrt{2}}\cdot\tfrac{1}{\sqrt{2}} - \tfrac{-i}{\sqrt{2}}\cdot\tfrac{1}{\sqrt{2}}\right) = +\frac{i\hbar}{2},
    \end{align*}
    and the diagonal entries vanish because $i\cdot(-i) = 1$ in both terms. So
    \[\hat{S}_y \xrightarrow{S_z} \frac{\hbar}{2}\begin{pmatrix} 0 & -i \\ i & 0 \end{pmatrix}.\]
\end{solution}

\begin{solution}[Expectation value and uncertainty by matrix mechanics]
    In the $S_z$ basis,
    \[\psi \xrightarrow{S_z} \frac{1}{\sqrt{10}}\left[\begin{pmatrix} 1 \\ i \end{pmatrix} + \begin{pmatrix} 2i \\ 2 \end{pmatrix}\right] = \frac{1}{\sqrt{10}}\begin{pmatrix} 1+2i \\ 2+i \end{pmatrix}.\]
    Then
    \[\braket{S_z} = \psi^\dagger\hat{S}_z\psi = \frac{\hbar}{2}\cdot\frac{1}{10}\begin{pmatrix} 1-2i & 2-i \end{pmatrix}\begin{pmatrix} 1+2i \\ -(2+i) \end{pmatrix} = \frac{\hbar}{2}\cdot\frac{5-5}{10} = 0.\]
    Since $\hat{S}_z^2 = \tfrac{\hbar^2}{4}\id$, $\braket{S_z^2} = \tfrac{\hbar^2}{4}$, so $\Delta S_z = \hbar/2$: both outcomes have probability $\tfrac12$ and the spread is maximal.
\end{solution}

\begin{solution}[Expectation value from the $S_x$ basis]
    \[\ket{\psi} \xrightarrow{S_x} \psi^x = \frac{1}{\sqrt{3}}\begin{pmatrix} 1 \\ i\sqrt{2} \end{pmatrix}.\]
    Change to the $S_z$ basis with $U$ from Eq.~\eqref{eq:Uzx}:
    \[\psi^z = U\psi^x = \frac{1}{\sqrt{6}}\begin{pmatrix} 1+i\sqrt{2} \\ 1-i\sqrt{2} \end{pmatrix}, \qquad \psi^{z\dagger} = \frac{1}{\sqrt{6}}\begin{pmatrix} 1-i\sqrt{2} & 1+i\sqrt{2} \end{pmatrix}.\]
    Then
    \[\braket{S_z} = \psi^{z\dagger}\hat{S}_z\psi^z = \frac{\hbar}{2}\cdot\frac{1}{6}\left(|1+i\sqrt{2}|^2 - |1-i\sqrt{2}|^2\right) = \frac{\hbar}{2}\left(\frac{3}{6} - \frac{3}{6}\right) = 0.\]
    Both components have equal modulus, so both $S_z$ outcomes have probability $\tfrac12$.
\end{solution}

\begin{solution}[$\hat{S}_z$ in the $S_x$ basis]
    $U$ from Eq.~\eqref{eq:Uzx} is real and symmetric, so $U^\dagger = U$ and
    \begin{align*}
        \hat{S}_z \xrightarrow{S_x} U^\dagger D\,U &= \frac{\hbar}{2}\cdot\frac{1}{2}\begin{pmatrix} 1 & 1 \\ 1 & -1 \end{pmatrix}\begin{pmatrix} 1 & 0 \\ 0 & -1 \end{pmatrix}\begin{pmatrix} 1 & 1 \\ 1 & -1 \end{pmatrix} = \frac{\hbar}{2}\begin{pmatrix} 0 & 1 \\ 1 & 0 \end{pmatrix}.
    \end{align*}
    This is $\frac{\hbar}{2}\sigma_x$: in the $S_x$ basis, $\hat{S}_z$ looks the way $\hat{S}_x$ looks in the $S_z$ basis.

    Keeping the state in the $S_x$ basis,
    \[\braket{S_z} = \psi^{x\dagger}\hat{S}_z^{x}\psi^x = \frac{\hbar}{2}\cdot\frac{1}{3}\begin{pmatrix} 1 & -i\sqrt{2} \end{pmatrix}\begin{pmatrix} i\sqrt{2} \\ 1 \end{pmatrix} = \frac{\hbar}{2}\cdot\frac{i\sqrt{2} - i\sqrt{2}}{3} = 0,\]
    matching the previous solution. In general, for $\psi^x = (c_+, c_-)^T$ this gives $\braket{S_z} = \hbar\,\mathrm{Re}\left(c_+^*c_-\right)$, which vanishes here because the relative phase is $i$.
\end{solution}

\begin{solution}[$R(\phi\hat{z})$ in the $S_x$ basis]
    Expanding the $S_x$ states in $\hat{z}$,
    \[R\ket{\pm\hat{x}} = \frac{1}{\sqrt{2}}\left(e^{-i\phi/2}\ket{+\hat{z}} \pm e^{i\phi/2}\ket{-\hat{z}}\right).\]
    Hitting these with $\bra{\pm\hat{x}} = \tfrac{1}{\sqrt{2}}\left(\bra{+\hat{z}} \pm \bra{-\hat{z}}\right)$ and using Euler's formula,
    \begin{align*}
        \bra{+\hat{x}}R\ket{+\hat{x}} = \bra{-\hat{x}}R\ket{-\hat{x}} &= \frac{1}{2}\left(e^{-i\phi/2} + e^{i\phi/2}\right) = \cos\frac{\phi}{2}, \\
        \bra{+\hat{x}}R\ket{-\hat{x}} = \bra{-\hat{x}}R\ket{+\hat{x}} &= \frac{1}{2}\left(e^{-i\phi/2} - e^{i\phi/2}\right) = -i\sin\frac{\phi}{2},
    \end{align*}
    so
    \[R(\phi\hat{z}) \xrightarrow{S_x} \begin{pmatrix} \cos\frac{\phi}{2} & -i\sin\frac{\phi}{2} \\ -i\sin\frac{\phi}{2} & \cos\frac{\phi}{2} \end{pmatrix}.\]
    Check: at $\phi = \pi/2$, $R\ket{+\hat{x}} \xrightarrow{S_x} \tfrac{1}{\sqrt{2}}(1,-i)^{T}$, which is $\ket{+\hat{y}}$ up to a global phase, as rotating $\hat{x}$ by $90^\circ$ about $\hat{z}$ should give.
\end{solution}
