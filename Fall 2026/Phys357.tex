\subsection{Angular Momentum}
\begin{definition}[Orbital Angular Momentum]
    The orbital angular momentum of a particle is
    \begin{equation}
        \vec{L} = \vec{r} \times \vec{p},
    \end{equation}
    where $\vec{r}$ is the position vector and $\vec{p} = m\vec{v}$ is the linear momentum.
\end{definition}

For motion confined to the $x$-$y$ plane, only the $z$ component survives. On a circular path of radius $r$ with speed $v$,
\begin{equation}
    L_{z} = mvr.
\end{equation}

\begin{definition}[Angular Momentum of a Rigid Body]
    For a rigid body rotating about a fixed axis this 
    \begin{equation}
        L = I_{0}\omega,
    \end{equation}
    with $I_{0}$ the moment of inertia and $\omega$ the angular velocity.
\end{definition}

\subsection{Magnetic moment of a circulating charge}
\begin{definition}[Magnetic moment]
    A magnetic moment is a vector quantity that determines the torque a magnet will experience in an external magnetic field.
\end{definition}

Consider a point particle of mass $m$ and charge $q$ in a circular orbit of radius $r$ and period $T$. Three steps connect this to a magnetic moment.

\begin{enumerate}
    \item The charge passes any point on the orbit once per period, so it is equivalent to a steady current
    \begin{equation*}
        I = \frac{q}{T} = \frac{qv}{2\pi r}.
    \end{equation*}

    \item A current loop of area $A = \pi r^{2}$ has magnetic moment
    \begin{equation*}
        \mu = \frac{IA}{c}.\footnotemark
    \end{equation*}

    \item Substituting gives
    \begin{equation}
        \mu = \frac{1}{c}\left(\frac{qv}{2\pi r}\right)\left(\pi r^{2}\right)
            = \frac{q}{2mc}\,mvr
            = \frac{q}{2mc}\,L_{z}.
    \end{equation}
\end{enumerate}
\footnotetext{Here $c$ is in Gaussian units.}
\begin{result}[Gyromagnetic Ratio]
    The magnetic moment is proportional to the angular momentum,
    \begin{equation}
        \vec{\mu} = \gamma \vec{L}, \qquad \gamma = \frac{q}{2mc}.
    \end{equation}
    The constant $\gamma$ is the gyromagnetic ratio.
\end{result}

\begin{remark}
    This is a purely classical result. It matters here because measuring $\vec{\mu}$ is how we measure $\vec{L}$, and it is what the Stern-Gerlach apparatus exploits.
\end{remark}

\subsection{Spin and the Magnetic Moment}

\begin{remark}
    The moment of inertia $I_0$ measures how mass is distributed relative to the rotation axis.
\end{remark}

In quantum mechanics the intrinsic angular momentum $\vec{S}$, called spin, replaces the orbital angular momentum $\vec{L}$. The magnetic moment is still proportional to it, but with an extra dimensionless factor,
\begin{equation}
    \vec{\mu} = \frac{qg}{2mc}\,\vec{S},
\end{equation}
where $g$ is the g-factor of the particle. It has no classical analogue and must be measured or derived from relativistic quantum theory.

\begin{definition}[Gyromagnetic ratio]
    The gyromagnetic ratio is the ratio of the magnetic moment to the angular momentum,
    \begin{equation}
        \gamma = \frac{qg}{2mc},
    \end{equation}
    where $q$ is the charge, $g$ is the g-factor, $m$ is the mass, and $c$ is the speed of light.
\end{definition}

\begin{definition}[Reduced Planck constant]
    \begin{equation}
        \hbar = \frac{h}{2\pi}.
    \end{equation}
    It carries units of angular momentum, so $[\hbar] = [L_z]$.
\end{definition}

\subsubsection{Magnetons}

Factoring $\hbar$ out of the gyromagnetic ratio gives a natural unit of magnetic moment for each mass scale.

\begin{definition}[Bohr magneton]
    \begin{equation}
        \mu_{B} = \frac{|e|\hbar}{2m_{e}c}.
    \end{equation}
\end{definition}

\begin{definition}[Nuclear magneton]
    \begin{equation}
        \mu_{N} = \frac{|e|\hbar}{2m_{p}c}.
    \end{equation}
\end{definition}

The electron moment is then written as
\begin{equation}
    \vec{\mu}_{e} = g_{e}\,\mu_{B}\,\frac{\vec{S}}{\hbar},
\end{equation}
and similarly for the proton and neutron with $\mu_{N}$ in place of $\mu_{B}$.

\begin{remark}
    The neutron is neutral but still has a magnetic moment, so $\vec{\mu}$ cannot come from a single circulating point charge. This is a first sign that the classical picture fails.
\end{remark}

\begin{remark}
    Since $m_{p} \gg m_{e}$, we get $\mu_{N} \ll \mu_{B}$ and therefore $|\vec{\mu}_{p}| \ll |\vec{\mu}_{e}|$. Stern-Gerlach deflection of nuclei is much weaker than for electrons.
\end{remark}

\subsection{Force in an Inhomogeneous Field}

A magnetic moment in a field has potential energy
\begin{equation}
    U = -\vec{\mu}\cdot\vec{B},
\end{equation}
so the force on it is
\begin{equation}
    \vec{F} = -\vec{\nabla}U, \qquad F_{z} = -\frac{\partial U}{\partial z}.
\end{equation}

\begin{remark}
    A uniform field gives no force, only a torque. The Stern-Gerlach apparatus needs a field \emph{gradient} to deflect anything.
\end{remark}

\subsubsection{Silver atoms}

Silver is used because its net orbital angular momentum vanishes, $\vec{L} = 0$, while $\vec{S} \neq 0$. The entire magnetic moment comes from a single unpaired valence electron, so the atom behaves as one electron spin.

Taking the field along $z$,
\begin{equation}
    U = -\mu_{z}B_{z} = g_{e}\frac{|e|}{2m_{e}c}\,S_{z}B_{z},
\end{equation}
and therefore
\begin{equation}
    F_{z} = -g_{e}\frac{|e|}{2m_{e}c}\,S_{z}\,\frac{\partial B_{z}}{\partial z}
          = g_{e}\frac{|e|}{2m_{e}c}\left|\frac{\partial B_{z}}{\partial z}\right| S_{z},
\end{equation}
where the last equality assumes $\partial B_{z}/\partial z < 0$.

\begin{result}
    \begin{equation}
        F_{z} \propto S_{z}.
    \end{equation}
    The deflection of an atom measures the $z$ component of its spin directly. Classically $S_{z}$ varies continuously over $[-|\vec{S}|, |\vec{S}|]$, so the beam should smear into a band. It does not.
\end{result}
