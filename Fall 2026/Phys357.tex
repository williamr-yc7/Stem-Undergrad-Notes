\section{Stern-Gerlach Experiments}

\subsection{The classical picture}
\marginpar{August 31, 2026}

\subsubsection{Angular momentum}

\begin{definition}[Orbital Angular Momentum]
    The orbital angular momentum of a particle is
    \begin{equation}
        \vec{L} = \vec{r} \times \vec{p},
    \end{equation}
    where $\vec{r}$ is the position vector and $\vec{p} = m\vec{v}$ is the linear momentum.
\end{definition}

For motion confined to the $x$-$y$ plane, only the $z$ component survives. On a circular path of radius $r$ with speed $v$,
\[L_{z} = mvr.\]

\begin{definition}[Angular Momentum of a Rigid Body]
    For a rigid body\footnote{A rigid body is one that does not deform under the action of external forces.} rotating about a fixed axis this is
    \begin{equation}
        L = I_{0}\omega,
    \end{equation}
    with $I_{0}$ the moment of inertia and $\omega$ the angular velocity.
\end{definition}

\begin{remark}
    The moment of inertia $I_0$ measures how mass is distributed relative to the rotation axis.
\end{remark}

\subsubsection{Magnetic moment}

\begin{definition}[Magnetic moment]
    A magnetic moment is a measure of the net circulation of electric charge. It is a vector quantity that determines the torque a magnet will experience in an external magnetic field.
\end{definition}

The magnetic moment of a point particle of mass $m$ and charge $q$ in a circular orbit of radius $r$ and period $T$ is
\begin{equation}
    \mu = \frac{IA}{c}\footnotemark = \frac{1}{c}\left(\frac{qv}{2\pi r}\right)\left(\pi r^{2}\right) = \frac{q}{2mc}\,mvr = \frac{q}{2mc}\,L_{z},
\end{equation}
where $I$ is the current and $A$ is the area enclosed by the orbit. A single circulating charge counts as a steady current $I = \frac{q}{T} = \frac{qv}{2\pi r}$ because it passes any given point on the loop once per period.

\footnotetext{Here $c$ is in Gaussian units.}

\begin{result}[Gyromagnetic Ratio]
    The magnetic moment is proportional to the angular momentum,
    \begin{equation}
        \vec{\mu} = \gamma \vec{L}, \qquad \gamma = \frac{q}{2mc}.
    \end{equation}
    The constant $\gamma$ is the gyromagnetic ratio.
\end{result}

\begin{remark}
    This is a purely classical result, in the sense that it follows from Newtonian mechanics and classical electromagnetism alone. The charge is treated as a point particle on a definite orbit with a definite radius and speed, and nothing is quantized. It matters here because measuring $\vec{\mu}$ is how we measure $\vec{L}$, and it is what the Stern-Gerlach apparatus exploits.
\end{remark}

\begin{definition}[Stern-Gerlach Apparatus]
    A device that measures one component of a particle's magnetic moment. A beam of neutral atoms passes through a magnet shaped to produce a strong field gradient along a chosen axis, here $z$. The gradient exerts a force $F_z = \mu_z \, \partial B_z / \partial z$, so each atom is deflected by an amount proportional to $\mu_z$, and a screen records where it lands. Since $\vec{\mu} = \gamma \vec{S}$, the position on the screen is a direct measurement of $S_z$.
\end{definition}

\subsection{Spin and the magnetic moment}
\marginpar{September 2, 2026}

In quantum mechanics the intrinsic angular momentum $\vec{S}$, called spin, replaces the orbital angular momentum $\vec{L}$. The magnetic moment is still proportional to it, but with an extra dimensionless factor,
\begin{equation}
    \vec{\mu} = \frac{qg}{2mc}\,\vec{S},
\end{equation}
where $g$ is the g-factor of the particle. It has no classical analogue and must be measured or derived from relativistic quantum theory.

\begin{definition}[Gyromagnetic ratio]
    The gyromagnetic ratio is the ratio of the magnetic moment to the angular momentum,
    \begin{equation}
        \gamma = \frac{qg}{2mc},
    \end{equation}
    where $q$ is the charge, $g$ is the g-factor, $m$ is the mass, and $c$ is the speed of light.
\end{definition}

\subsubsection{Magnetons}

\begin{definition}[Reduced Planck constant]
    \begin{equation}
        \hbar = \frac{h}{2\pi}.
    \end{equation}
    It carries units of angular momentum, so $[\hbar] = [L_z]$.
\end{definition}

Factoring $\hbar$ out of the gyromagnetic ratio gives a natural unit of magnetic moment for each mass scale.

\begin{definition}[Bohr magneton]
    \begin{equation}
        \mu_{B} = \frac{|e|\hbar}{2m_{e}c}.
    \end{equation}
\end{definition}

\begin{definition}[Nuclear magneton]
    \begin{equation}
        \mu_{N} = \frac{|e|\hbar}{2m_{p}c}.
    \end{equation}
\end{definition}

The electron moment is then written as
\begin{equation}
    \vec{\mu}_{e} = g_{e}\,\mu_{B}\,\frac{\vec{S}}{\hbar},
\end{equation}
and similarly for the proton and neutron with $\mu_{N}$ in place of $\mu_{B}$.

\subsubsection{Values for the three particles}

\begin{table}[H]

Proton   & $+|e|$ & $m_{p}$ & $g_{p} \approx 5.59$ & $g_{p}\mu_{N}/\hbar$ \\
Neutron  & $0$    & $m_{p}$ & $g_{n} \approx -3.83$ & $g_{n}\mu_{N}/\hbar$ \\
\hline
\end{tabular}
\caption{Charges, masses, g-factors and gyromagnetic ratios.}
\end{table}

\begin{remark}
    The neutron is neutral but still has a magnetic moment, so $\vec{\mu}$ cannot come from a single circulating point charge. This is a first sign that the classical picture fails.
\end{remark}

\begin{remark}
    Since $m_{p} \gg m_{e}$, we get $\mu_{N} \ll \mu_{B}$ and therefore $|\vec{\mu}_{p}| \ll |\vec{\mu}_{e}|$. Stern-Gerlach deflection of nuclei is much weaker than for electrons.
\end{remark}

\subsection{The apparatus: force in an inhomogeneous field}

A magnetic moment in a field has potential energy
\begin{equation}
    U = -\vec{\mu}\cdot\vec{B},
\end{equation}
so the force on it is
\begin{equation}
    \vec{F} = -\vec{\nabla}U, \qquad F_{z} = -\frac{\partial U}{\partial z} = \mu_{z}\frac{\partial B_{z}}{\partial z}.
\end{equation}

\begin{remark}
    A uniform field gives no force, only a torque. The Stern-Gerlach apparatus needs a field \emph{gradient} to deflect anything.
\end{remark}

\subsubsection{Silver atoms}

Silver is used because its net orbital angular momentum vanishes, $\vec{L} = 0$, while $\vec{S} \neq 0$. The entire magnetic moment comes from a single unpaired valence electron, so the atom behaves as one electron spin. For the electron,
\begin{equation}
    \mu_{z} = -\frac{g_{e}|e|}{2m_{e}c}\,S_{z}.
\end{equation}

Taking the field along $z$,
\begin{equation}
    U = -\mu_{z}B_{z} = g_{e}\frac{|e|}{2m_{e}c}\,S_{z}B_{z},
\end{equation}
and therefore
\begin{equation}
    F_{z} = -g_{e}\frac{|e|}{2m_{e}c}\,S_{z}\,\frac{\partial B_{z}}{\partial z}
          = g_{e}\frac{|e|}{2m_{e}c}\left|\frac{\partial B_{z}}{\partial z}\right| S_{z},
\end{equation}
where the last equality assumes $\partial B_{z}/\partial z < 0$.

\begin{result}
    \begin{equation}
        F_{z} \propto S_{z}.
    \end{equation}
    The deflection of an atom measures the $z$ component of its spin directly.
\end{result}

\subsection{The result: quantization of \texorpdfstring{$S_z$}{Sz}}
\marginpar{September 4, 2026}

Spin is a vector,
\begin{equation}
    \vec{S} = (S_{x}, S_{y}, S_{z})^{T}, \qquad S_{z} = |\vec{S}|\cos\theta.
\end{equation}
Classically $\theta \in [0,\pi]$, so $\cos\theta \in [-1,1]$ and $S_{z}$ varies continuously over $[-|\vec{S}|, |\vec{S}|]$. Since $F_z \propto S_z$, the beam should smear into a continuous band. It does not.

\begin{result}[Quantization of spin]
    The electron spin angular momentum is quantized:
    \begin{equation}
        S_{z} = \pm\frac{\hbar}{2}.
    \end{equation}
    The beam splits into two spots.
\end{result}

\subsubsection{Ket notation}

New notation, a ``ket'':
\begin{equation*}
    \ket{\uparrow} \; \text{for } S_{z} = +\frac{\hbar}{2},
    \qquad
    \ket{\downarrow} \; \text{for } S_{z} = -\frac{\hbar}{2}.
\end{equation*}
