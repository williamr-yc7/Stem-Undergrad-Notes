\subsection{Partial Differential Equations}
\begin{definition}[Partial Differential Equation]
    A partial differential equation (PDE) is a differential equation whose unknown function depends on more than one independent variable.
\end{definition}
\begin{remark}
    Physics rarely hands us a function $u$ directly. It gives a law for how $u$ changes. Example: heat flows from hot to cold, so the temperature $u(x,t)$ in a room changes in time wherever it varies in space. Recovering $u$ from such a relation is the whole subject.
\end{remark}

\subsection{Classification}
\subsubsection{Order}
\begin{definition}[Order]
    The order of a PDE is the order of the highest derivative appearing in it.
\end{definition}
\begin{example}
    $u_{xx} = 0$ is second order, and $u_t + u u_x = 0$ is first order.
\end{example}

\subsubsection{Linearity}
\begin{definition}[Linear PDE]
    A PDE is linear if it can be written as $L(u) = f(x)$, where $f$ does not involve $u$ and $L$ satisfies
    \begin{equation}
        L(u_1 + c\,u_2) = L(u_1) + c\,L(u_2).
    \end{equation}
\end{definition}
\begin{remark}[Practical test]
    Every term containing $u$ or a derivative of $u$ must satisfy:
    \begin{itemize}
        \item $u$ and its derivatives appear only to the first power (no $u^2$, $(u_x)^2$),
        \item they are never multiplied together (no $u\,u_x$, $u_x u_y$),
        \item they never sit inside a function (no $\sin u$, $e^u$, $\ln u$),
        \item their coefficients depend only on the independent variables, never on $u$.
    \end{itemize}
    Each violation creates cross terms, so the superposition identity above fails.
\end{remark}
\begin{example}
    $u_{xx} + xy\,u_y + e^x u = \cos x$ is linear. Nonlinear: $u\,u_x + u_y = 0$ (product), $u_{xx} + (u_y)^2 = 0$ (power), $u_{xx} + \sin u = 0$ (inside a function).
\end{example}
\begin{definition}[Degrees of nonlinearity]
    Let $k$ be the order of the PDE.
    \begin{itemize}
        \item \textbf{Semilinear:} linear in the order-$k$ derivatives, with coefficients depending only on $x$; nonlinearity only in lower-order terms. Example: $u_{xx} + \sin u = 0$.
        \item \textbf{Quasilinear:} linear in the order-$k$ derivatives, but their coefficients may depend on $u$ and its lower-order derivatives. Example: Burgers, $u_t + u\,u_x = 0$.
        \item \textbf{Fully nonlinear:} the order-$k$ derivatives themselves appear nonlinearly. Example: $(u_x)^2 + (u_y)^2 = 1$.
    \end{itemize}
\end{definition}

\subsubsection{Homogeneity}
\begin{definition}[Homogeneous]
    A linear PDE $L(u) = f(x)$ is homogeneous if $f \equiv 0$, and inhomogeneous otherwise.
\end{definition}
\begin{example}
    $u_t + a u_x = x^3$ is linear with $L(u) = u_t + a u_x$ and $f(x) = x^3$, so it is inhomogeneous.
\end{example}
\begin{example}
    $u_t + u u_x = f(x)$ is nonlinear, because the term $u u_x$ is not linear in $u$.
\end{example}

\subsection{Solutions}
\begin{definition}[Classical solution]
    Let the PDE be written as $F(u) = 0$ and let $k$ be its order. A function $u$ is a (classical) solution if
    \begin{enumerate}[label=(\arabic*)]
        \item $u \in C^k$, and
        \item $F(u) = 0$ at every point of the domain.
    \end{enumerate}
\end{definition}
\begin{example}[Transport equation]
    Let $u = u(x,t)$ and $F(u_t, u_x) = u_t + c u_x = 0$ with $c \in \mathbb{R}$. Let $\Phi$ be any differentiable function and set $u(x,t) = \Phi(x - ct)$. Then
    \begin{equation}
        u_x = \Phi'(x-ct), \qquad u_t = -c\,\Phi'(x-ct),
    \end{equation}
    so $u_t + c u_x = 0$. The whole family $\Phi$ solves the equation. Meaning: the initial profile $\Phi$ slides rigidly to the right at speed $c$.
\end{example}
\begin{example}[Heat equation]
    Let $u_t = c^2 u_{xx}$ and take the ansatz $u(x,t) = e^{-c^2k^2 t}\sin(kx)$ for an arbitrary constant $k$. Then
    \begin{equation}
        u_t = -c^2k^2 u, \qquad
        u_x = k\,e^{-c^2k^2t}\cos(kx), \qquad
        u_{xx} = -k^2 e^{-c^2k^2t}\sin(kx) = -k^2 u,
    \end{equation}
    so $u_t = c^2 u_{xx}$. Meaning: the sine keeps its shape and decays, and high frequencies (large $k$) decay faster, at rate $c^2k^2$. Diffusion smooths.
\end{example}
\begin{example}[Wave equation]
    Let $u_{tt} = c^2 u_{xx}$ and take $u(x,t) = \cos(kct)\sin(kx)$. Then
    \begin{equation}
        u_{xx} = -k^2 u, \qquad u_{tt} = -c^2k^2 u,
    \end{equation}
    so $u_{tt} = c^2 u_{xx}$. Meaning: a standing wave that oscillates forever without decaying. Same spatial profile as the heat example, completely different time behaviour.
\end{example}


\subsection{Principle of Superposition}
\begin{principle}[Superposition]
    Let $L$ be linear.
    \begin{enumerate}[label=(\arabic*)]
        \item If $L(u_1) = L(u_2) = 0$, then $L(u_1 + c\,u_2) = 0$ for every $c \in \mathbb{R}$. Solutions of a homogeneous linear PDE form a vector space.
        \item If $L(u_p) = f$ and $L(u_h) = 0$, then $L(u_p + u_h) = f$: one particular solution plus any homogeneous solution solves the inhomogeneous equation.
    \end{enumerate}
\end{principle}
\begin{proof}
    By linearity of $L$,
    \begin{equation}
        L(u_1 + c u_2) = L(u_1) + c\,L(u_2) = 0 + c \cdot 0 = 0,
        \qquad
        L(u_p + u_h) = f + 0 = f. \qedhere
    \end{equation}
\end{proof}
\begin{example}
    For $u_t = c^2 u_{xx}$, both
    \begin{equation}
        u_1(x,t) = e^{-c^2k^2t}\sin(kx)
        \qquad \text{and} \qquad
        u_2(x,t) = e^{-c^2 t}\sin(x)
    \end{equation}
    are solutions, so $u(x,t) = a_1 u_1(x,t) + a_2 u_2(x,t)$ is a solution for any $a_1, a_2 \in \mathbb{R}$. This is how we will later build series solutions.
\end{example}

\section{General Solutions}

\subsection{Solving by Direct Integration}
\begin{example}
    Solve $u_x = 0$ for $u(x,y)$ on $\mathbb{R}^2$. The solution is $u(x,y) = f(y)$ for an arbitrary $f$.
\end{example}
\begin{example}
    Solve $u_{xx} = 0$. Set $v(x,y) = u_x$. Then $v_x = 0$, so by the previous example $v(x,y) = f(y)$, hence $u_x = f(y)$. Fixing $y$ and integrating in $x$ (the ``constant'' of integration only has to be constant in $x$) gives
    \begin{equation}
        u(x,y) = f(y)\,x + g(y).
    \end{equation}
    Check: $u_x = f(y)$ and $u_{xx} = 0$.
\end{example}
\begin{example}
    Solve $u_{xy} = 0$. Set $v(x,y) = u_x$. Then $v_y = 0$, so $v(x,y) = f(x)$, hence $u_x = f(x)$. Fixing $y$ and integrating in $x$ gives
    \begin{equation}
        u(x,y) = F(x) + g(y), \qquad F' = f.
    \end{equation}
    Check: $u_x = F'(x)$ depends on $x$ only, so $u_{xy} = 0$.
\end{example}
\begin{remark}
    In each case the general solution carries arbitrary \emph{functions} rather than arbitrary constants. An ODE of order $k$ has $k$ free constants; a PDE has free functions of the remaining variables. Second order in one variable, or first order in each of two, gives two free functions.
\end{remark}

\subsection{Auxiliary Conditions}
Auxiliary conditions (initial or boundary conditions) prescribe $u$, and possibly its derivatives, on a lower dimensional set. They cut the family of general solutions down to one. A typical problem on a domain $\Omega \subset \mathbb{R}^d$ reads
\begin{equation}
    \Delta u = f(x) \quad \text{in } \Omega,
    \qquad
    u(x) = g(x) \quad \text{on } \partial\Omega .
\end{equation}
The big three problems of the course:
\begin{align}
    &\text{Heat IVP:} && u_t = \alpha u_{xx},\ u(x,0) = g(x), \\
    &\text{Wave IVP:} && u_{tt} = c^2 u_{xx},\ u(x,0) = g(x),\ u_t(x,0) = h(x), \\
    &\text{Dirichlet problem:} && \Delta u = 0 \text{ in the unit disk } D,\ u = g \text{ on } \partial D.
\end{align}
\begin{definition}[Well-posed, Hadamard]
    A PDE with auxiliary conditions is well-posed if
    \begin{enumerate}[label=(\arabic*)]
        \item \textbf{Existence:} a solution exists,
        \item \textbf{Uniqueness:} there is only one,
        \item \textbf{Stability:} small changes in the data give small changes in the solution.
    \end{enumerate}
    Stability matters because data comes from measurements (with errors) and computers discretize.
\end{definition}
\begin{remark}[Counting conditions]
    For ODEs, number of conditions $=$ order. For PDEs this fails: all three problems above are second order, but the wave IVP needs two conditions while heat and Laplace need one.
\end{remark}

    \hfill \emph{Solution: p.~\pageref{sol:a1-110}.}

\subsection{Operator Viewpoint}
Write a linear PDE as
\begin{equation}
    Lu = f,
\end{equation}
where $L$ is the operator, $u$ is the unknown and $f$ is the input. The linear algebra analogue is
\begin{equation}
    Mx = b, \qquad x, b \in \mathbb{R}^d, \qquad x = M^{-1}b,
\end{equation}
so solving a linear PDE is the infinite-dimensional version of inverting $M$. Well-posedness already shows up here:
\begin{itemize}
    \item $M$ invertible: exactly one solution for every $b$.
    \item $M$ not invertible, $b$ in the range of $M$: solutions exist but are not unique (add anything in the null space).
    \item $M$ not invertible, $b$ not in the range: no solution.
\end{itemize}
\begin{example}
    $M = \begin{pmatrix} 1 & 1 \\ 1 & 1 \end{pmatrix}$. For $b = (2,2)$ every $x = (2-t, t)$ works: existence without uniqueness. For $b = (1,2)$ there is no solution: the data is incompatible with the equation.
\end{example}

\subsection{How We Will Solve PDEs}
\begin{enumerate}[label=(\arabic*)]
    \item Assume a smooth solution exists.
    \item Work out what the equation and conditions force on it.
    \item In good cases this gives an explicit formula.
    \item Verify the formula really solves the problem. The verification is what makes it a proof.
\end{enumerate}


\section{Potential Theory}

\subsection{Newton's Problem}
Place a point mass $M$ at the origin and a test mass $m$ at $x \in \mathbb{R}^3 \setminus \{0\}$. The gravitational force on $m$ has magnitude $GmM/\|x\|^2$ and points back toward the origin:
\begin{equation}
    \vec{F}(x) = -\frac{GmM}{\|x\|^2} \cdot \frac{x}{\|x\|} = -\frac{GmM}{\|x\|^3}\, x.
\end{equation}

\subsection{Toolkit}
\begin{lemma}[Derivative of distance is direction]
    Let $r(x) = \|x\|$ with $x = (x_1,x_2,x_3) \in \mathbb{R}^3$. Then for $x \neq 0$,
    \begin{equation}
        \frac{\partial r}{\partial x_i} = \frac{x_i}{r},
        \qquad
        \nabla r(x) = \frac{x}{r} = \hat{x}.
    \end{equation}
\end{lemma}
\begin{proof}
    Start from $r^2 = x_1^2 + x_2^2 + x_3^2$ and differentiate both sides with respect to $x_i$:
    \begin{equation}
        2r \frac{\partial r}{\partial x_i} = 2x_i
        \quad \Longrightarrow \quad
        \frac{\partial r}{\partial x_i} = \frac{x_i}{r}.
    \end{equation}
    This requires $r \neq 0$, so the identity fails at the origin.
\end{proof}
\begin{remark}
    $\nabla r$ is the unit vector pointing radially outward, so $\|\nabla r\| = 1$ everywhere it is defined.
\end{remark}

\subsection{Radial Chain Rule}
\begin{proposition}
    Let $f : (0,\infty) \to \mathbb{R}$ be differentiable and set $g(x) = f(r(x)) = f(\|x\|)$. Then for $x \neq 0$,
    \begin{equation}
        \nabla g(x) = f'(r)\, \nabla r(x) = f'(r)\, \frac{x}{r}.
    \end{equation}
\end{proposition}
\begin{proof}
    Componentwise, by the chain rule,
    \begin{equation}
        \frac{\partial g}{\partial x_i} = f'(r(x)) \frac{\partial r}{\partial x_i} = f'(r) \frac{x_i}{r}.
    \end{equation}
\end{proof}
So the gradient of any radial function points radially, and $f'$ alone controls the magnitude.
\begin{example}[Gradient refresher, non-radial]
    For $g(x) = x_1 x_2 + x_3^2$,
    \begin{equation}
        \nabla g(x) = \left( \frac{\partial g}{\partial x_1}, \frac{\partial g}{\partial x_2}, \frac{\partial g}{\partial x_3} \right) = (x_2,\; x_1,\; 2x_3).
    \end{equation}
\end{example}

\subsection{The Laplacian}
Everything below works in $\mathbb{R}^d$ for any $d \geq 1$. Write $x = (x_1, \dots, x_d)$.

\subsubsection*{First form: sum of second partials}
\begin{definition}[Laplacian]
    \begin{equation}
        \Delta = \sum_{i=1}^{d} \frac{\partial^2}{\partial x_i^2},
        \qquad
        \Delta f(x) = \sum_{i=1}^{d} \frac{\partial^2 f}{\partial x_i^2}(x).
    \end{equation}
\end{definition}
\begin{example}
    Let $f(x_1,x_2,x_3) = x_1^2 - x_2^2$. Then
    \begin{equation}
        \frac{\partial^2 f}{\partial x_1^2} = 2, \qquad
        \frac{\partial^2 f}{\partial x_2^2} = -2, \qquad
        \frac{\partial^2 f}{\partial x_3^2} = 0,
    \end{equation}
    so $\Delta f = 2 - 2 + 0 = 0$. A function with $\Delta f = 0$ is called \emph{harmonic}.
\end{example}

\subsubsection*{Second form: divergence of the gradient}
\begin{definition}[Divergence]
    For a vector field $\vec{v}(x) = (v^1(x), \dots, v^d(x))$, where the superscripts label components and are not powers,
    \begin{equation}
        \nabla \cdot \vec{v} = \operatorname{div} \vec{v} = \sum_{i=1}^{d} \frac{\partial v^i}{\partial x_i}(x).
    \end{equation}
\end{definition}
Applying this to the gradient field $\vec{v} = \nabla f$, whose $i$-th component is $\partial f / \partial x_i$, gives
\begin{equation}
    \nabla \cdot (\nabla f) = \sum_{i=1}^{d} \frac{\partial}{\partial x_i}\left( \frac{\partial f}{\partial x_i} \right) = \sum_{i=1}^{d} \frac{\partial^2 f}{\partial x_i^2} = \Delta f.
\end{equation}
\begin{proposition}
    \begin{equation}
        \Delta = \nabla \cdot \nabla = \operatorname{div} \circ \operatorname{grad}.
    \end{equation}
\end{proposition}
\begin{remark}
    This second form is the useful one for radial functions: combining it with the radial chain rule turns $\Delta$ into an ODE in $r$.
\end{remark}

\subsection{Harmonic Radial Functions}
\begin{proposition}[Radial Laplacian]
    For $g(x) = f(r)$, $r = \|x\|$ in $\mathbb{R}^d$, $x \neq 0$:
    \begin{equation}
        \Delta g = f''(r) + \frac{d-1}{r} f'(r).
    \end{equation}
\end{proposition}
Solve $\Delta f = 0$ in $d = 3$ with the ansatz $f(r) = r^a$:
\begin{equation}
    \Delta f = a(a-1)r^{a-2} + \frac{2}{r}\, a r^{a-1} = a(a+1)\, r^{a-2},
\end{equation}
which vanishes for $r > 0$ iff $a = 0$ or $a = -1$. So
\begin{equation}
    g(x) = A + \frac{B}{\|x\|}.
\end{equation}
\begin{remark}
    $1/\|x\|$ is the Newtonian potential: $\nabla(1/r) = -x/\|x\|^3$, the inverse-square field. Harmonic away from the origin only. Its divergence is $0$ for $x \neq 0$, yet its flux through every sphere around the origin is the same nonzero number ($-4\pi$). So all the ``source'' sits at one point, and no function can describe it. This is the motivation for distributions (Section \ref{sec:distA}).
\end{remark}


\section{First Order PDEs and the Solution Operator}
\marginpar{September 9, 2026}

\subsection{The Underlying ODE}
Start from the ODE
\begin{equation}
    x'(s) = v(x(s), s),
\end{equation}
where $v$ is a velocity field: a particle at $x$ at time $s$ moves with velocity $v(x,s)$. Assume $v(x,s)$ is smooth. By the existence and uniqueness theorem there is a unique trajectory through every point, forwards and backwards in time.

\subsection{The Solution Operator}
\begin{definition}[Solution operator]
    Given times $t_0$ and $t_1$, the solution operator is the map
    \begin{equation}
        S_{t_0 \to t_1}(x_0) = x(t_1),
    \end{equation}
    where $x(\cdot)$ solves the ODE with initial condition $x(t_0) = x_0$. In words: where is the particle at time $t_1$ if it was at $x_0$ at time $t_0$?
\end{definition}
\begin{proposition}[Identities]
    \begin{enumerate}
        \item $S_{t \to t}(x) = x$, that is $S_{t \to t} = \mathrm{Id}$.
        \item $S_{t_1 \to t_2} \circ S_{t_0 \to t_1} = S_{t_0 \to t_2}$ (two legs or one, same endpoint).
        \item $S_{t_1 \to t_0} = \left( S_{t_0 \to t_1} \right)^{-1}$: running time backwards inverts the operator.
    \end{enumerate}
\end{proposition}
\begin{example}[Constant velocity]
    $v = c$. Trajectories are lines $x(t) = x_0 + c(t - t_0)$, so $S_{t_0\to t_1}(x) = x + c(t_1 - t_0)$.
\end{example}
\begin{example}[Linear velocity]
    $v(x) = -x$. Trajectories are $x(t) = x_0 e^{-(t-t_0)}$, so $S_{t_0\to t_1}(x) = x\,e^{-(t_1-t_0)}$. In the $(x,t)$ plane these are exponential curves collapsing toward $x = 0$ (the constant velocity ones are parallel lines).
\end{example}

\subsection{Transporting Labels: Where the PDE Comes From}
Stamp each particle at time $0$ with a label $g(x_0)$ that never changes. Let $u(x,t)$ be the label of the particle sitting at $x$ at time $t$. That particle started at $S_{t\to 0}(x)$ (run the ODE backwards), so
\begin{equation}\label{eq:label}
    u(x,t) = g\big(S_{t \to 0}(x)\big).
\end{equation}
Labels are carried forward by the flow; to evaluate $u$ we track backward to the data.
\begin{example}
    Constant velocity: $u(x,t) = g(x - ct)$, the signal translated at speed $c$. Linear velocity $v = -x$: $u(x,t) = g(x e^{t})$.
\end{example}
\textbf{Deriving the PDE.} $u$ is constant along every trajectory, so by the chain rule
\begin{equation}
    0 = \frac{d}{dt} u(x(t), t) = u_t + x'(t)\, u_x = u_t + v(x,t)\, u_x .
\end{equation}
Since a trajectory passes through every point,
\begin{equation}
    u_t + v(x,t)\, u_x = 0 \qquad \text{(linear transport equation)}.
\end{equation}
\begin{remark}
    We wrote the solution first and got the PDE by differentiating. The method of characteristics is the reverse trip: read $v$ off the PDE, rebuild the flow, then ``flow backwards, read the label''. The trajectories of $v$ are exactly the characteristics. Formula \eqref{eq:label} only defines $u$ where the backward trajectory hits the data line, so the data must meet each trajectory exactly once. Units: $v$ is length per time.
\end{remark}

\subsection{Examples of Trajectories}
\begin{example}
    The curve $\vec{x}(s) = (x(s), y(s)) = (3e^{s-2}, 4e^{s-2})$ satisfies $\vec{x}'(s) = \vec{x}(s)$, so it is a trajectory of the velocity field $\vec{v}(x,y) = (x,y)$. It passes through $\vec{x}(2) = (3,4)$ and $\vec{x}(5) = (3e^3, 4e^3)$.
\end{example}
\begin{example}
    The curve $\vec{x}(s) = (3s+5, 5s+6)$ has constant velocity $\vec{v}(s) = \vec{x}'(s) = (3,5)$, with $\vec{x}(0) = (5,6)$ and $\vec{x}(2) = (11,16)$.
\end{example}
\begin{example}[Decoupled linear field]
    Take $\vec{v}(x,y) = (c_1 x, c_2 y)$ for constants $c_1, c_2$, so the ODE is
    \begin{equation}
        \begin{cases}
            x'(s) = c_1 x(s) \\
            y'(s) = c_2 y(s)
        \end{cases}
        \qquad
        \begin{cases}
            x(0) = x_0 \\
            y(0) = y_0.
        \end{cases}
    \end{equation}
    The solution is $x(s) = x_0 e^{c_1 s}$, $y(s) = y_0 e^{c_2 s}$. Check: $x'(s) = c_1 x_0 e^{c_1 s} = c_1 x(s)$, and similarly for $y$.
\end{example}
\begin{example}[Rotation field]
    Take $\vec{v}(x,y) = (-y, x)$. A circle of radius $r$, $\vec{x}(s) = (r\cos s, r\sin s)$, has $\vec{x}'(s) = (-r\sin s, r\cos s) = (-y, x)$, so circles about the origin are trajectories. The general solution of
    \begin{equation}
        \begin{cases}
            x' = -y \\
            y' = x
        \end{cases}
        \qquad
        \begin{cases}
            x(0) = x_0 \\
            y(0) = y_0
        \end{cases}
    \end{equation}
    is
    \begin{equation}
        x(s) = x_0 \cos s - y_0 \sin s, \qquad y(s) = x_0 \sin s + y_0 \cos s.
    \end{equation}
    The circle above is the case $x_0 = r$, $y_0 = 0$.
\end{example}

\subsection{Constant Coefficient Transport}
Consider
\begin{equation}
    a u_x + b u_y = 0, \qquad u = u(x,y), \qquad a, b \in \mathbb{R}.
\end{equation}
The left side is $\nabla u \cdot (a,b)$, the directional derivative of $u$ in the direction $(a,b)$. So the PDE says $u$ does not change along lines parallel to $(a,b)$.
\begin{proposition}[Constancy along characteristics]
    Let $\vec{v} = (a,b)$ and let $\vec{x}(s) = (x(s), y(s))$ solve $\vec{x}'(s) = \vec{v}$, that is $x'(s) = a$ and $y'(s) = b$. Then $u(\vec{x}(s))$ is constant in $s$.
\end{proposition}
\begin{proof}
    By the chain rule,
    \begin{equation}
        \frac{d}{ds} u(x(s), y(s)) = u_x\, x'(s) + u_y\, y'(s) = a u_x + b u_y = 0. \qedhere
    \end{equation}
\end{proof}
The trajectories of $\vec{v}$ are called \emph{characteristics}. Since $u$ is constant along each one, its value is determined by wherever the characteristic meets the data.
\begin{definition}[Characteristic, informal]
    A curve along which the PDE reduces to an ODE. Knowing $u$ at one point of it, the PDE determines $u$ along the whole curve. (Special to first order: $u$ at a point depends on the data at a single point. Not true for heat or Laplace.)
\end{definition}
\begin{proposition}[General solution]
    For any differentiable $f$,
    \begin{equation}
        u(x,y) = f(-bx + ay)
    \end{equation}
    solves $a u_x + b u_y = 0$.
\end{proposition}
\begin{proof}
    $u_x = -b\, f'(-bx+ay)$ and $u_y = a\, f'(-bx+ay)$, so $a u_x + b u_y = (-ab + ab) f'(-bx+ay) = 0$.
\end{proof}
\begin{remark}
    The argument $-bx + ay$ is constant exactly along the lines of direction $(a,b)$, which are the characteristics. When $b = 0$ the equation reduces to $a u_x = 0$ and the formula gives $u(x,y) = f(ay)$, a function of $y$ alone, as expected.
\end{remark}

\subsection{Adding Data}
\begin{example}
    Solve
    \begin{equation}
        \begin{cases}
            3u_x + 2u_y = 0 & \text{(PDE)} \\
            u(x,0) = x^3 & \text{(IC)},
        \end{cases}
    \end{equation}
    where the data is given on the line $\Gamma = \{(x,y) \mid y = 0\}$.
\end{example}
\begin{method}[Via the general solution]
    With $a = 3$, $b = 2$, the general solution is $u(x,y) = f(-2x + 3y)$. On $y = 0$ this reads $u(x,0) = f(-2x) = x^3$. Setting $z = -2x$, so $x = -z/2$, gives $f(z) = -z^3/8$. Therefore
    \begin{equation}
        u(x,y) = f(-2x + 3y) = -\frac{(-2x+3y)^3}{8} = \frac{(2x-3y)^3}{8}.
    \end{equation}
    Check: $u(x,0) = (2x)^3/8 = x^3$, and $3u_x + 2u_y = \tfrac{3}{8}\cdot 3(2x-3y)^2 \cdot 2 + \tfrac{2}{8}\cdot 3(2x-3y)^2 \cdot (-3) = 0$.
\end{method}
\begin{method}[Via characteristics]
    \begin{enumerate}[label=(\arabic*)]
        \item $u$ is constant along curves with $x'(s) = 3$, $y'(s) = 2$.
        \item These are the lines $x(s) = 3s + x_0$, $y(s) = 2s + y_0$.
        \item Given a target point $(x,y)$, find the point $(x_0, 0)$ on $\Gamma$ whose characteristic reaches it. From $y = 2s + 0$ we get $s = y/2$, and then $x = 3s + x_0$ gives $x_0 = x - 3y/2$.
        \item Since $u$ is constant along the characteristic, $u(x,y) = u(x_0, 0) = x_0^3$:
        \begin{equation}
            u(x,y) = \left( x - \frac{3y}{2} \right)^3 = \frac{(2x-3y)^3}{8}.
        \end{equation}
    \end{enumerate}
    Both methods agree.
\end{method}

\subsection{Which Data Curves Work?}
A data curve works if it meets every characteristic exactly once (here: the $x$-axis, the $y$-axis, $y = x^2$\dots). It fails if it \emph{is} a characteristic. For $3u_x + 2u_y = 0$, $u$ is forced to be constant on the line $2x - 3y = 0$, so:
\begin{itemize}
    \item nonconstant data on that line: \textbf{no solution};
    \item constant data $c$ on that line: \textbf{infinitely many} ($u = f(2x-3y)$ with any $f$ having $f(0) = c$).
\end{itemize}
Either way the problem is not well-posed.

\subsection{Problem Setup}
\begin{problem}[Linear transport with variable coefficients]
    Inputs:
    \begin{enumerate}[label=(\arabic*)]
        \item A curve $\Gamma \subset \mathbb{R}^2$ and data $u_0 : \Gamma \to \R$.
        \item Coefficients $a, b : \R^2 \to \R$.
    \end{enumerate}
    Find $u : \R^2 \to \R$ such that
    \begin{equation}
        \begin{cases}
            a(x,y)\,u_x + b(x,y)\,u_y = 0 & \text{(PDE)} \\
            u(x,y) = u_0(x,y) \quad \forall (x,y) \in \Gamma & \text{(IC)}.
        \end{cases}
    \end{equation}
\end{problem}

\subsection{Method of Characteristics}
\begin{method}[Homogeneous case]
    \begin{enumerate}[label=(\arabic*)]
        \item Solve the characteristic ODE
        \begin{equation}
            \begin{cases}
                x'(s) = a(x(s), y(s)) \\
                y'(s) = b(x(s), y(s))
            \end{cases}
            \qquad
            \begin{cases}
                x(0) = x_0 \\
                y(0) = y_0
            \end{cases}
            \qquad (x_0, y_0) \in \Gamma.
        \end{equation}
        \item Set $z(s) = u(x(s), y(s))$. By the chain rule
        \begin{equation}
            z'(s) = u_x\, x' + u_y\, y' = a u_x + b u_y = 0,
        \end{equation}
        so $z$ is constant and $z(s) = u_0(x_0, y_0)$.
        \item Given a target $(x,y)$, find $s_0$ and $(x_0, y_0) \in \Gamma$ with $x(s_0) = x$, $y(s_0) = y$. Then
        \begin{equation}
            u(x,y) = u_0\big(x_0(x,y),\, y_0(x,y)\big).
        \end{equation}
    \end{enumerate}
\end{method}

\subsection{Inhomogeneous Case}
Now allow a right hand side:
\begin{equation}
    a(x,y)\,u_x + b(x,y)\,u_y = c_1(x,y)\,u + c_2(x,y).
\end{equation}
The characteristics are unchanged, but $u$ is no longer constant along them. With $z(s) = u(x(s), y(s))$,
\begin{equation}
    z'(s) = a u_x + b u_y = c_1 z + c_2,
\end{equation}
so the method becomes a three-dimensional ODE system (the characteristic equations):
\begin{equation}
    \begin{cases}
        x' = a(x,y) \\
        y' = b(x,y) \\
        z' = c_1(x,y)\, z + c_2(x,y)
    \end{cases}
    \qquad
    \begin{cases}
        x(0) = x_0 \\
        y(0) = y_0 \\
        z(0) = u_0(x_0, y_0).
    \end{cases}
\end{equation}
Because the PDE is linear, the first two equations do not involve $z$: solve them first, then the $z$ equation. If $c_1 = c_2 = 0$ we are back to the homogeneous case.
\begin{remark}[Vocabulary]
    The curves $(x(s), y(s))$ in the plane are the \emph{projected characteristics}. The curve $(x(s), y(s), z(s))$ lies on the solution surface $z = u(x,y)$, and its shadow in the plane is the projected characteristic.
\end{remark}
\begin{method}[Recipe]
    \begin{enumerate}[label=(\arabic*)]
        \item Solve for $x(s), y(s)$ starting on $\Gamma$; one free parameter (e.g.\ $x_0$) says where on $\Gamma$ you started.
        \item Solve for $z(s)$ with $z(0) =$ data at that starting point.
        \item Given $(x,y)$, solve for $s$ and $x_0$, substitute into $z$. The final formula contains only $x$ and $y$. (This is the backward flow $S_{t\to0}$ again.)
    \end{enumerate}
\end{method}

\subsection{Worked Example}
\begin{example}
    Solve
    \begin{equation}
        \begin{cases}
            -y\,u_x + x\,u_y = u & \text{(PDE)} \\
            u(x,0) = g(x) & \text{(IC)}, \quad \Gamma = \{(x,0) \mid x > 0\}.
        \end{cases}
    \end{equation}
\end{example}
\begin{solution}
    Here $a = -y$, $b = x$, $c_1 = 1$, $c_2 = 0$.\par
    \textbf{Step 1: characteristic system.}
    \begin{equation}
        x' = -y, \qquad y' = x, \qquad z' = z,
        \qquad
        x(0) = x_0,\; y(0) = 0,\; z(0) = g(x_0).
    \end{equation}
    \textbf{Step 2: solve.} From the rotation field example, the general solution of $x' = -y$, $y' = x$ is
    \begin{equation}
        x(s) = a_1 \cos s - a_2 \sin s, \qquad y(s) = a_1 \sin s + a_2 \cos s.
    \end{equation}
    The initial conditions give $a_1 = x_0$, $a_2 = 0$, so
    \begin{equation}
        x(s) = x_0 \cos s, \qquad y(s) = x_0 \sin s, \qquad z(s) = g(x_0)\, e^{s}.
    \end{equation}
    \textbf{Step 3: invert.} Require $x(s_0) = x$, $y(s_0) = y$:
    \begin{equation}
        x = x_0 \cos s_0, \quad y = x_0 \sin s_0
        \quad \Longrightarrow \quad
        x_0 = \sqrt{x^2 + y^2}, \qquad s_0 = \arctan(y/x).
    \end{equation}
    \textbf{Step 4: read off $u$.}
    \begin{equation}
        u(x,y) = z(s_0) = g\!\left(\sqrt{x^2 + y^2}\right) \exp\!\big(\arctan(y/x)\big).
    \end{equation}
    In polar coordinates this is $u = g(r)\,e^{\theta}$, and the PDE is $\partial_\theta u = u$, which checks.
\end{solution}
\begin{remark}
    The characteristics are circles centred at the origin, so every point except the origin is reached from $\Gamma$. At the origin both coefficients vanish and the characteristic direction degenerates, so $u$ is undefined there. The formula with $\arctan$ is valid for $x > 0$; elsewhere use the polar angle $\theta \in [0, 2\pi)$ measured from the positive $x$-axis. Note $u$ jumps by a factor $e^{2\pi}$ across $\Gamma$ itself when $g \neq 0$, since going once around the circle multiplies $z$ by $e^{2\pi}$.
\end{remark}

\subsection{Time as the Parameter}
If one variable is time, data is at $t = 0$, and the coefficient of $u_t$ is $1$,
\begin{equation}
    u_t + a(x,t)\, u_x = \dots,
\end{equation}
then the characteristic equation for $t$ is $t' = 1$, $t(0) = 0$, so $t = s$. Drop $s$ and use $t$ itself as the parameter: characteristics are curves $x(t)$ in the $(x,t)$ plane. This is exactly the setting of the solution operator section.
\begin{example}[Variable speed transport]
    \begin{equation}
        u_t - x\,u_x = 0, \qquad u(x,0) = \frac{1}{1+x^2}.
    \end{equation}
    Characteristics: $x'(t) = -x$, $x(0) = x_0$, so $x(t) = x_0 e^{-t}$, and $z' = 0$ so $z = \frac{1}{1+x_0^2}$. Invert: $x_0 = x e^{t}$ (this is $S_{t\to0}$). Therefore
    \begin{equation}
        u(x,t) = \frac{1}{1 + x^2 e^{2t}}.
    \end{equation}
    Every point of the signal moves toward $0$ at speed $|x|$, so the blip narrows while its height stays $1$. Variable speed distorts the shape; constant speed only translates it.
\end{example}

\section{Distributions A: The Delta ``Function'' Is Not a Function}\label{sec:distA}
\marginpar{September 16, 2026}
Source: Choksi Ch.~5. Goal: a weaker notion of solving, e.g.\ $u'' = f$, that allows a right hand side like a point source. The one idea to keep: \textbf{pointwise values are the wrong data; integrals against test functions are the right data.}

\subsection{Integrable Functions}
\begin{definition}
    $f : \R \to \R$ is \emph{integrable} if $\int_{-\infty}^{\infty} |f(x)|\,dx < \infty$. It is \emph{locally integrable} if $\int_a^b |f(x)|\,dx < \infty$ for all $a < b$.
\end{definition}
\begin{example}
    \leavevmode
    \begin{itemize}
        \item $f(x) = 1$ (or any polynomial): locally integrable, $\int_a^b 1\,dx = b - a$, but not integrable.
        \item If $f$ is continuous and $f(x) = 0$ for $|x| > R$: integrable.
        \item $\frac{1}{1+x^2}$: integrable (total integral $\pi$).
        \item $\frac{1}{\sqrt{|x|}}$: locally integrable, since $\int_{-1}^1 |x|^{-1/2}dx = 4$ is finite.
        \item $\frac{1}{x}$: not locally integrable, $\int_0^1 \frac{dx}{x} = \log x\big|_0^1 = \infty$.
    \end{itemize}
\end{example}
\begin{remark}[Paradigm]
    If $f(x) = g(x)$ for all $x \neq 3$ but $f(3) \neq g(3)$, then $\int f = \int g$ over every interval, and $\int f\phi = \int g\phi$ for every $\phi$. Integration cannot see single points, so we treat $f$ and $g$ as the same function.
\end{remark}

\subsection{What We Want from $\delta_0$}
Informally, $\delta_0$ is a ``point mass'' at $0$: $\delta_0(x) = 0$ for $x \neq 0$ but $\int_{-\infty}^{\infty} \delta_0(x)\,dx = 1$. No function $\delta_0 : \R \to \R$ can do this (by the paradigm, a function that is $0$ except at one point has integral $0$). What we really want is: for every continuous $f$,
\begin{equation}
    \text{``}\int_{-\infty}^{\infty} \delta_0(x) f(x)\,dx\text{''} = f(0).
\end{equation}

\subsection{First Appearance: the Derivative of Heaviside}
\begin{equation}
    H(x) = \begin{cases} 0, & x < 0 \\ 1, & x \geq 0 \end{cases}
\end{equation}
Pointwise $H'(x) = 0$ for $x \neq 0$ and undefined at $0$. Ignoring the point gives $H' = 0$, but then $H$ would be constant, which it is not. The whole jump happens at $0$.
\textbf{The computation that reveals $H'$.} Fix $\phi$ smooth, vanishing for large $|x|$, and set
\begin{equation}
    F(a) = \int_{-\infty}^{\infty} H(x-a)\phi(x)\,dx = \int_a^{\infty} \phi(x)\,dx.
\end{equation}
\begin{itemize}
    \item Differentiating under the integral: $F'(a) = -\int H'(x-a)\phi(x)\,dx$. If $H' = 0$ this gives $0$.
    \item Directly (fundamental theorem of calculus): $F'(a) = -\phi(a)$. Not zero.
\end{itemize}
So whatever $H'$ is, $\int H'(x-a)\phi(x)\,dx = \phi(a)$: it picks out the value of $\phi$ where the jump sits. That is exactly $\delta$ at $a$.
\textbf{Same idea with a ramp (in class).} Smooth out the jump:
\begin{equation}
    g_\epsilon(x) = \begin{cases} 0, & x < 0 \\ x/\epsilon, & 0 \leq x \leq \epsilon \\ 1, & x \geq \epsilon \end{cases}
    \qquad
    g_\epsilon'(x) = \begin{cases} 0, & x < 0 \\ 1/\epsilon, & 0 < x < \epsilon \\ 0, & x > \epsilon \end{cases}
\end{equation}
Then $\int g_\epsilon'(x) f(x)\,dx = \frac{1}{\epsilon}\int_0^\epsilon f(x)\,dx$, the average of $f$ over $[0,\epsilon]$. If $f$ is continuous near $0$, this tends to $f(0)$ as $\epsilon \to 0$. As $g_\epsilon \to H$, the derivatives act like $\delta_0$.

\subsection{Second Appearance: Spikes}
Define triangles of base $2/n$ and height $n$ (area $1$):
\begin{equation}
    f_n(x) = \begin{cases} n - n^2|x|, & |x| \leq 1/n \\ 0, & |x| > 1/n. \end{cases}
\end{equation}
Pointwise, $f_n(x) \to 0$ for $x \neq 0$ and $f_n(0) = n \to \infty$: useless, and many sequences with total mass other than $1$ have the same pointwise limit. But for continuous $\phi$, $\phi \approx \phi(0)$ on the tiny interval $[-1/n, 1/n]$, so
\begin{equation}
    \int f_n \phi\,dx \approx \phi(0)\int f_n\,dx = \phi(0) \quad \Longrightarrow \quad \lim_{n\to\infty}\int f_n\phi\,dx = \phi(0).
\end{equation}
\begin{example}[In class]
    $\phi = 1$: $\int f_n = 1$. $\phi = x$: $\int f_n x\,dx = 0$ ($f_n$ even, $x$ odd). $\phi = x^2$:
    \begin{equation}
        2\int_0^{1/n}(n x^2 - n^2 x^3)\,dx = 2\left(\frac{1}{3n^2} - \frac{1}{4n^2}\right) = \frac{1}{6n^2} \to 0.
    \end{equation}
    Each limit equals $\phi(0)$ ($1$, $0$, $0$), as predicted. This is Dirac's picture: a huge spike of area $1$, shape irrelevant.
\end{example}
\begin{remark}[Punchline]
    $\delta$ appears when differentiating $H$ and as the limit of concentrating functions. Both times pointwise values fail and integrating against a generic $\phi$ succeeds. So we will \emph{define} objects by what they do to test functions.
\end{remark}


\section{Distributions B: Distributions and Their Derivatives}\label{sec:distB}
\marginpar{September 21, 2026}
Source: Choksi 5.3, 5.4.

\subsection{Motivation: Weak Solutions of $u'' = f$}
A classical solution of $u''(x) = f(x)$ needs $u \in C^2$, so $f$ must be continuous. For example $u'' = -1$ on $(-1,1)$ with $u(-1) = u(1) = 0$ has $u(x) = \frac{1-x^2}{2}$. We want to allow a point source: replace $f$ by concentrating $f_\epsilon$, solve
\begin{equation}
    u_\epsilon'' = f_\epsilon, \qquad u_\epsilon(-1) = u_\epsilon(1) = 0,
\end{equation}
and let $\epsilon \to 0$. To make sense of the limit equation $u'' = \delta_0$ we need distributions.

\subsection{Test Functions}
\begin{definition}[Support]
    The support of a continuous $\phi$ is the closure of $\{x : \phi(x) \neq 0\}$. $\phi$ has \emph{compact support} if $\phi = 0$ outside some interval $[-R,R]$.
\end{definition}
\begin{definition}[Test functions]
    $\D$ is the set of infinitely differentiable functions with compact support.
\end{definition}
No polynomial works (a polynomial vanishing on an interval is $0$). The standard example is the \textbf{bump function}
\begin{equation}
    \phi(x) = \begin{cases} e^{-1/(1-x^2)}, & |x| < 1 \\ 0, & |x| \geq 1, \end{cases}
\end{equation}
with support $[-1,1]$. The only issue is smoothness at $x = \pm 1$: every derivative tends to $0$ there because the exponential beats every power (Exercise 5.1).\par
Building more test functions from one: translates and rescalings $\phi(ax+b)$, sums and scalar multiples, products $\psi\phi$ with $\psi$ any $C^\infty$ function, and most importantly \textbf{derivatives}: $\phi \in \D \Rightarrow \phi^{(k)} \in \D$ for all $k$. This is why we insist on $C^\infty$.

\subsection{Definition of a Distribution}
\begin{definition}[Distribution]
    A distribution $F$ is a rule assigning to each test function $\phi$ a real number,
    \begin{equation}
        F : \D \to \R, \qquad \phi \mapsto \ip{F}{\phi} \in \R,
    \end{equation}
    which is
    \begin{itemize}
        \item \textbf{linear}: $\ip{F}{a\phi + b\psi} = a\ip{F}{\phi} + b\ip{F}{\psi}$,
        \item \textbf{continuous}: $\phi_n \to \phi$ in $\D$ implies $\ip{F}{\phi_n} \to \ip{F}{\phi}$.
    \end{itemize}
    A map from functions to numbers is called a \emph{functional}. $\ip{F}{\phi}$ is just notation for $F(\phi)$, chosen to look like an integral.
\end{definition}
\begin{definition}[Convergence in $\D$]
    $\phi_n \to \phi$ in $\D$ if
    \begin{enumerate}[label=(\roman*)]
        \item all $\phi_n$ and $\phi$ vanish outside one common $[-R,R]$, and
        \item for every $k \geq 0$, $\max_x |\phi_n^{(k)}(x) - \phi^{(k)}(x)| \to 0$ (every derivative converges uniformly).
    \end{enumerate}
\end{definition}
Distributions form a vector space: $\ip{F+G}{\phi} = \ip{F}{\phi} + \ip{G}{\phi}$ and $\ip{cF}{\phi} = c\ip{F}{\phi}$.
\begin{remark}[Rule of the game]
    Never ask for the value of a distribution at a point $x$. Ask for its value on a test function $\phi$: think of $\phi$ as a probe.
\end{remark}

\subsection{Functions as Distributions}
\begin{proposition}
    If $f$ is locally integrable, then
    \begin{equation}
        \ip{F_f}{\phi} = \int_{-\infty}^{\infty} f(x)\phi(x)\,dx
    \end{equation}
    defines a distribution $F_f$, the distribution \emph{generated by} $f$.
\end{proposition}
\begin{proof}
    If $\phi = 0$ outside $[-R,R]$, then $\left|\int f\phi\right| \leq \max|\phi| \int_{-R}^R |f| < \infty$, so it is well defined (this is exactly why we need local integrability). Linearity is linearity of the integral. Continuity: with common support $[-R,R]$,
    \begin{equation}
        |\ip{F_f}{\phi_n} - \ip{F_f}{\phi}| \leq \max|\phi_n - \phi| \int_{-R}^R |f|\,dx \to 0. \qedhere
    \end{equation}
\end{proof}
\begin{example}[5.3.1]
    $f \equiv 1$: $\ip{F_1}{\phi} = \int \phi\,dx$. Likewise $f \equiv 3$: $\ip{F_3}{\phi} = 3\int\phi\,dx$.
\end{example}
\begin{remark}
    Functions that differ only on a set integration cannot see (measure zero) generate the same distribution; otherwise $F_f$ determines $f$. Notation: we usually just write $f$ for $F_f$. ``In the sense of distributions'' means the statement is about $F_f$.
\end{remark}

\subsection{The Delta Function}
\begin{definition}[5.3.2]
    $\ip{\delta_0}{\phi} = \phi(0)$, and for $a \in \R$, $\ip{\delta_a}{\phi} = \phi(a)$.
\end{definition}
It is linear, and continuous since $\phi_n \to \phi$ uniformly gives $\phi_n(0) \to \phi(0)$.
\begin{itemize}
    \item There is no integral in the definition. Never write $\delta_0(x)$ or $\int \delta_0(x)\phi(x)\,dx$.
    \item It is not a function:
\end{itemize}
\begin{proposition}
    There is no locally integrable $f$ with $F_f = \delta_0$.
\end{proposition}
\begin{proof}
    Suppose $\int f\phi = \phi(0)$ for every test function. Take $\phi = e\cdot(\text{bump})$, so $\phi(0) = 1$, $0 \leq \phi \leq 1$, support $[-1,1]$. Let $\phi_n(x) = \phi(nx)$: support $[-1/n, 1/n]$, $\phi_n(0) = 1$. Then
    \begin{equation}
        1 = \phi_n(0) = \int f\phi_n\,dx \leq \int_{-1/n}^{1/n} |f|\,dx \to 0,
    \end{equation}
    since the integral of a locally integrable function over a shrinking interval goes to $0$. Contradiction.
\end{proof}

\subsection{Derivative of a Distribution}
\textbf{Motivation.} For $f \in C^1$ and a test function $\phi$, integrate by parts (boundary terms vanish by compact support):
\begin{equation}
    \int f'\phi\,dx = -\int f\phi'\,dx.
\end{equation}
The derivative moves onto $\phi$ at the cost of a sign, and the right side makes sense even if $f$ is not differentiable.
\begin{definition}[5.4.1]
    \begin{equation}
        \ip{F'}{\phi} = -\ip{F}{\phi'}.
    \end{equation}
\end{definition}
This is a distribution ($\phi'$ is a test function; linear; continuous since $\phi_n \to \phi$ implies $\phi_n' \to \phi'$). So \textbf{every distribution, hence every locally integrable function, has a derivative}.
\begin{proposition}[Consistency, 5.20]
    If $f \in C^1$, then $(F_f)' = F_{f'}$: the distributional derivative agrees with the classical one. (Proof: the integration by parts above.)
\end{proposition}
\begin{example}
    $\ip{\delta_0'}{\phi} = -\phi'(0)$. Higher derivatives, one sign per derivative:
    \begin{equation}
        \ip{F^{(k)}}{\phi} = (-1)^k \ip{F}{\phi^{(k)}}.
    \end{equation}
\end{example}
\begin{remark}
    Like the transpose in linear algebra ($\langle Ax, y\rangle = \langle x, A^T y\rangle$): the transpose of $\frac{d}{dx}$ is $-\frac{d}{dx}$. Every operation on distributions will be defined this way.
\end{remark}

\subsection{Derivatives of Piecewise Smooth Functions}
\begin{example}[Heaviside, 5.4.1]
    \begin{equation}
        \ip{H'}{\phi} = -\int_{-\infty}^\infty H\phi'\,dx = -\int_0^\infty \phi'\,dx = -\big(0 - \phi(0)\big) = \phi(0),
    \end{equation}
    using $\phi(\infty) = 0$. So $H' = \delta_0$ in the sense of distributions. (We never wrote $\int\delta_0(x)\phi(x)dx$.)
\end{example}
\begin{remark}[Quiz hint, September 28]
    Be able to show $H' = \delta_0$ from the definition, exactly as above.
\end{remark}
\begin{example}[$|x|$, 5.4.2]
    Split at $0$ and integrate by parts on each side:
    \begin{equation}
        \ip{|x|'}{\phi} = -\int |x|\phi'\,dx = \int_{-\infty}^0 x\phi'\,dx - \int_0^\infty x\phi'\,dx = -\int_{-\infty}^0\phi\,dx + \int_0^\infty\phi\,dx,
    \end{equation}
    (boundary terms vanish: at $0$ because of the factor $x$, at $\pm\infty$ by compact support). So $|x|' = \operatorname{sgn}(x)$ ($-1$ for $x<0$, $+1$ for $x>0$); the value at $0$ is irrelevant because $|x|$ is continuous. Second derivative:
    \begin{equation}
        \ip{\operatorname{sgn}'}{\phi} = -\int\operatorname{sgn}\,\phi' = \int_{-\infty}^0\phi' - \int_0^\infty\phi' = \phi(0) + \phi(0) = 2\phi(0),
    \end{equation}
    so $|x|'' = 2\delta_0$. The jump of size $2$ in $\operatorname{sgn}$ shows up as $2\delta_0$.
\end{example}
\begin{theorem}[Derivative of a function with a jump]
    Let $f$ be $C^1$ except at one point $a$, with $f$ and $f'$ having one-sided limits at $a$. Let $\{f'\}$ denote the pointwise derivative for $x \neq a$. Then, in the sense of distributions,
    \begin{equation}
        f' = \{f'\} + \big(f(a^+) - f(a^-)\big)\,\delta_a.
    \end{equation}
    In particular, if $f$ is continuous at $a$, then $f' = \{f'\}$.
\end{theorem}
\begin{proof}[Proof idea]
    Split $-\int f\phi'$ at $a$ and integrate by parts on each side. The interior terms give $\int\{f'\}\phi$; the boundary terms at $a$ give $f(a^+)\phi(a) - f(a^-)\phi(a)$.
\end{proof}
Checks: $H$ has jump $1$ at $0$, $\{H'\} = 0$, so $H' = \delta_0$. $|x|$ has no jump, so $|x|' = \operatorname{sgn}$.
\begin{example}
    \marginpar{September 28, 2026}
    Let
    \begin{equation}
        f(x) = \begin{cases} 2x + 3, & x < 0 \\ x^2, & x \geq 0. \end{cases}
    \end{equation}
    Remove the jump: $f = g - 3H$, where
    \begin{equation}
        g(x) = \begin{cases} 2x + 3, & x < 0 \\ x^2 + 3, & x \geq 0 \end{cases}
    \end{equation}
    is continuous and piecewise differentiable, so $g' = \{g'\}$. With $(-3H)' = -3\delta_0$,
    \begin{equation}
        f' = \{f'\} - 3\delta_0, \qquad \{f'\}(x) = \begin{cases} 2, & x < 0 \\ 2x, & x > 0, \end{cases}
    \end{equation}
    matching the theorem with jump $f(0^+) - f(0^-) = 0 - 3 = -3$.
\end{example}
\begin{method}[Recipe]
    \begin{enumerate}[label=(\arabic*)]
        \item Fix a test function $\phi$ and write $\ip{f'}{\phi} = -\int f\phi'$.
        \item Split $\R$ into intervals where $f$ is smooth.
        \item Integrate by parts on each piece. Boundary terms at $\pm\infty$ vanish; boundary terms at a singularity add up to (jump of $f$) $\times\,\phi(a)$, i.e.\ a delta.
    \end{enumerate}
\end{method}

\subsection{Fundamental Solution of $\frac{d^2}{dx^2}$}
Divide the $|x|$ example by $2$:
\begin{equation}
    \left(\frac{|x|}{2}\right)'' = \delta_0 .
\end{equation}
A solution of $u'' = \delta_0$ is a \emph{fundamental solution}: the response to a unit source at one point. $u = |x|/2$ is continuous, has a kink at the source, and solves $u'' = 0$ away from it. The solution is two derivatives smoother than the right side.
\begin{example}
    Back to the motivation: $u'' = \delta_0$ on $(-1,1)$ with $u(\pm 1) = 0$ is solved by $u(x) = \frac{|x| - 1}{2}$ (add the constant $-\tfrac12$, which has zero second derivative, to fix the boundary values).
\end{example}
\begin{remark}
    This is the 1D version of the point mass from week 1: in $\R^3$, $\Delta\frac{1}{|x|} = 0$ away from $0$, and in the sense of distributions $\Delta\frac{1}{|x|} = -4\pi\delta_0$. Proved later with Laplace's equation (Choksi Ch.~8).
\end{remark}

\section{Distributions C: Convergence in the Sense of Distributions}\label{sec:distC}
\marginpar{September 23, 2026}
Source: Choksi 5.5.

\subsection{Definition}
\begin{definition}[5.5.1]
    Distributions $F_n$ converge to a distribution $F$ \emph{in the sense of distributions} if
    \begin{equation}
        \ip{F_n}{\phi} \xrightarrow{n\to\infty} \ip{F}{\phi} \qquad \text{for all } \phi \in \D.
    \end{equation}
\end{definition}
\begin{definition}[5.5.2]
    Locally integrable $f_n$ converge to $F$ in the sense of distributions if $F_{f_n} \to F$, that is,
    \begin{equation}
        \int_{-\infty}^\infty f_n(x)\phi(x)\,dx \to \ip{F}{\phi} \qquad \text{for all } \phi \in \D.
    \end{equation}
\end{definition}
\begin{remark}
    For each fixed probe $\phi$ this is convergence of real numbers. Different probes may converge at different rates; nothing uniform in $\phi$ is required. Two cases: $f_n \to f$ means $\int f_n\phi \to \int f\phi$; $f_n \to \delta_0$ means $\int f_n\phi \to \phi(0)$.
\end{remark}
\begin{proposition}[5.5.1, derivatives converge automatically]
    If $F_n \to F$ in the sense of distributions, then $F_n' \to F'$, and likewise for every higher derivative.
\end{proposition}
\begin{proof}
    $\ip{F_n'}{\phi} = -\ip{F_n}{\phi'} \to -\ip{F}{\phi'} = \ip{F'}{\phi}$, because $\phi'$ is a test function.
\end{proof}
\begin{remark}
    In ordinary calculus $f_n \to f$ uniformly says nothing about $f_n'$. Here the transpose trick pays off: limits of distributions can always be differentiated term by term.
\end{remark}

\subsection{Distributional versus Pointwise Convergence}
\begin{proposition}[5.5.2]
    Let $f_n$ be locally integrable with $f_n \to f$ pointwise, $f$ locally integrable, and suppose there is a locally integrable $g$ with $|f_n(x)| \leq g(x)$ for all $n$ and $x$. Then $f_n \to f$ in the sense of distributions.
\end{proposition}
\begin{proof}
    Dominated convergence (Choksi Theorem A.8) applied to $\int f_n\phi$ on the support of $\phi$.
\end{proof}
\begin{example}
    $f_n(x) = \frac{1}{n}e^{-x^2/4n} \to 0$ pointwise and $|f_n| \leq 1$, so $f_n \to 0$ in the sense of distributions.
\end{example}
\begin{remark}[Neither direction is free]
    \leavevmode
    \begin{itemize}
        \item The spikes of Section~\ref{sec:distA} tend to $0$ pointwise for $x \neq 0$, yet converge to $\delta_0$, not $0$. The proposition fails: any dominating $g$ must be of size $\frac{1}{4|x|}$ near $0$, which is not locally integrable.
        \item $\sin(nx) \to 0$ in the sense of distributions (Theorem below), yet $\sin(nx)$ does not converge for fixed $x$ not a multiple of $\pi$.
    \end{itemize}
\end{remark}

\subsection{Approximations to the Identity}
With $\sigma_n = 1/n$, four sequences:
\begin{align}
    &\text{Hats:} && f_n(x) = \tfrac{n}{2} \text{ for } |x| \leq \sigma_n,\ 0 \text{ otherwise}, \\
    &\text{Spikes:} && f_n(x) = n - n^2|x| \text{ for } |x| < \sigma_n,\ 0 \text{ otherwise}, \\
    &\text{Arctangent derivative:} && f_n(x) = \frac{1}{\pi}\frac{\sigma_n}{x^2 + \sigma_n^2}, \\
    &\text{Gaussians:} && f_n(x) = \frac{1}{\sqrt{2\pi\sigma_n}}\,e^{-x^2/2\sigma_n}.
\end{align}
All four converge to $\delta_0$ in the sense of distributions, i.e.\ $\int f_n\phi \to \phi(0)$. Such a sequence is an \emph{approximation to the identity}.
\begin{definition}[Properties 5.5.1]
    \begin{enumerate}[label=(\arabic*)]
        \item Nonnegativity: $f_n \geq 0$.
        \item Unit mass: $\int_{-\infty}^\infty f_n\,dx = 1$ for every $n$.
        \item Concentration at $0$: as $n \to \infty$ the mass of $f_n$ concentrates near $x = 0$.
    \end{enumerate}
\end{definition}
\begin{example}
    Unit mass for the arctangent sequence: $\int \frac{\sigma}{x^2+\sigma^2}\,dx = \big[\arctan(x/\sigma)\big]_{-\infty}^\infty = \pi$.
\end{example}
\begin{remark}[Heuristic]
    By (1) and (3), almost all the mass sits in $[-\delta,\delta]$, where $\phi \approx \phi(0)$, so $\int f_n\phi \approx \phi(0)\int_{-\delta}^{\delta} f_n \approx \phi(0)$. Making (3) precise is what the proofs do.
\end{remark}
\begin{theorem}[5.1]
    The spikes converge to $\delta_0$ in the sense of distributions.
\end{theorem}
\begin{proof}
    Fix $\phi$ and $\epsilon > 0$. Since $\int f_n = 1$, $\phi(0) = \int f_n(x)\phi(0)\,dx$, so using $f_n \geq 0$,
    \begin{equation}
        \left|\int f_n\phi\,dx - \phi(0)\right| \leq \int f_n(x)\,|\phi(x) - \phi(0)|\,dx.
    \end{equation}
    By continuity there is $\delta > 0$ with $|\phi(x) - \phi(0)| < \epsilon$ for $|x| < \delta$. Choose $N$ with $1/N < \delta$. For $n > N$, $f_n$ vanishes outside $[-1/n,1/n]$, so the right side is $< \epsilon\int f_n = \epsilon$.
\end{proof}
\begin{remark}
    The proof used only (1), (2) and the support $[-1/n,1/n]$, never the formula. The shape is irrelevant, as Dirac said.
\end{remark}
\begin{theorem}[5.2, arctangent sequence]
    The arctangent derivatives converge to $\delta_0$.
\end{theorem}
\begin{proof}[Proof sketch]
    They are positive everywhere, so the tails must be handled. Split the integral into $|x| < \delta$ (below $\epsilon/2$ by continuity) and $|x| \geq \delta$, where $|\phi(x) - \phi(0)| \leq 2C$ with $C = \max|\phi|$ and
    \begin{equation}
        \int_{|x|\geq\delta} f_n\,dx = \frac{2}{\pi}\left(\frac{\pi}{2} - \arctan\frac{\delta}{\sigma_n}\right) \to 0.
    \end{equation}
    Choose $N$ so this is below $\epsilon/(4C)$. Full proof: Choksi pp.~178 to 180.
\end{proof}
\begin{theorem}[5.3, shape does not matter]
    Let $f \geq 0$ be integrable with $\int f = 1$, and put $f_n(x) = n f(nx)$. Then $f_n \to \delta_0$ in the sense of distributions.
\end{theorem}
\begin{proof}
    Substitute $y = nx$: $\int n f(nx)\phi(x)\,dx = \int f(y)\phi(y/n)\,dy$. As $n \to \infty$, $\phi(y/n) \to \phi(0)$, and $|f(y)\phi(y/n)| \leq \max|\phi|\,|f(y)|$ is integrable, so by dominated convergence the integral tends to $\phi(0)\int f = \phi(0)$.
\end{proof}
\begin{remark}
    Nonnegativity was not used. Hats and spikes are $nf(nx)$ with $f$ the first hat or spike; the arctangent sequence uses $f = \frac{1}{\pi(1+x^2)}$. The Gaussians are a squeeze by $\sqrt{n}$ instead of $n$ (Exercise 5.13).
\end{remark}

\subsection{Oscillation: $\sin(nx) \to 0$}
\begin{remark}[A tempting argument, and why to distrust it]
    For large $n$, $\phi$ is nearly constant on each period of $\sin(nx)$, so each period contributes nearly $0$. But there are about $n$ periods per unit length, so ``many small pieces'' needs an actual estimate (the same argument would prove every Riemann integral is $0$).
\end{remark}
\begin{theorem}[5.4]
    $\sin(nx) \to 0$ in the sense of distributions. The same holds for $\cos(nx)$.
\end{theorem}
\begin{proof}
    Let $\phi = 0$ for $|x| \geq L$ and $C = \max|\phi'|$. Integrate by parts using $\phi(\pm L) = 0$:
    \begin{equation}
        \left|\int_{-L}^L \sin(nx)\phi(x)\,dx\right| = \left|\frac{1}{n}\int_{-L}^L \cos(nx)\phi'(x)\,dx\right| \leq \frac{2LC}{n} \to 0. \qedhere
    \end{equation}
\end{proof}
\begin{remark}
    Only compact support and one derivative of $\phi$ were used. Under much weaker hypotheses this is the \emph{Riemann--Lebesgue lemma}: Fourier coefficients of integrable functions tend to $0$ (Section~\ref{sec:fourierC}).
\end{remark}
\begin{example}
    By Proposition 5.5.1, $(\sin nx)' = n\cos(nx) \to 0$ in the sense of distributions, although the amplitude $n$ is unbounded. Oscillation cancels growth.
\end{example}
\begin{remark}[Averaging, Exercise 5.17]
    If $f$ is $\ell$-periodic with integral $A$ over a period, then $f(nx) \to A/\ell$, its average, in the sense of distributions.
\end{remark}

\subsection{Limits That Produce $\delta_0'$}
For $h > 0$,
\begin{equation}
    \ip{\frac{\delta_0 - \delta_h}{h}}{\phi} = \frac{\phi(0) - \phi(h)}{h} \xrightarrow{h \to 0} -\phi'(0) = \ip{\delta_0'}{\phi}.
\end{equation}
Two opposite point sources of size $1/h$, close together, converge to a \emph{dipole}. Equivalently, $(\delta_0 - \delta_h)/h$ is the derivative of the function equal to $1/h$ on $[0,h]$, which tends to $\delta_0$, so Proposition 5.5.1 applies.

    \hfill \emph{Solution: p.~\pageref{sol:a2-518}.}
\end{problem}

\subsection{Preview: Sinc and Dirichlet Kernels}
Two sequences that take negative values, so Properties 5.5.1 fail:
\begin{equation}
    S_n(x) = \frac{\sin(nx)}{\pi x}, \qquad K_n(x) = \frac{\sin\big((n+\tfrac12)x\big)}{\sin(x/2)},
\end{equation}
with $S_n(0) = n/\pi$, $K_n(0) = 2n+1$. Both have $\int_\R S_n = 1$ and $\int_{-\pi}^\pi K_n = 2\pi$, and
\begin{equation}
    S_n \to \delta_0 \text{ on } \R, \qquad K_n \to 2\pi\delta_0 \text{ on } (-\pi,\pi).
\end{equation}
The tails of $K_n$ oscillate with amplitude that does not tend to $0$, so continuity of $\phi$ is not enough. This is why pointwise convergence of Fourier series is delicate (Section~\ref{sec:fourierC}).

\subsection{Summary}
\begin{itemize}
    \item A distribution is defined by its action on test functions. Never ask for its value at a point.
    \item Every distribution has a derivative, $\ip{F'}{\phi} = -\ip{F}{\phi'}$. A jump gives a delta with the jump as coefficient. To remember: $H' = \delta_0$ and $(|x|/2)'' = \delta_0$.
    \item Convergence is tested one probe at a time. It captures concentration (limit $\delta_0$) and oscillation (limit the average), and derivatives of convergent sequences converge.
    \item Six sequences converge to $\delta_0$: hats, spikes, arctangent derivatives, Gaussians, sinc functions, Dirichlet kernels. The Gaussians solve the diffusion equation; the last two run Fourier analysis.
\end{itemize}


\section{Fourier Series A: Orthogonality Gives the Formulas}\label{sec:fourierA}
\marginpar{September 30, 2026}
Source: Choksi 11.1 to 11.3. Notation: in Chapter 5, $\phi$ was a test function. Here, following Choksi, $\phi$ is the function being expanded.

Fourier's claim (1807, memoir on heat flow): almost any function on an interval is a sum of sines and cosines. Lagrange objected that smooth sines cannot add up to a corner or jump. Yet the partial sums
\begin{equation}
    S_N(x) = \frac{4}{\pi}\sum_{\substack{n \leq N \\ n \text{ odd}}} \frac{1}{n}\sin\left(\frac{n\pi x}{l}\right)
\end{equation}
flatten toward $1$ on $(0,l)$, toward $-1$ on $(-l,0)$, and pass through $0$ at $x = 0$: smooth waves building a jump. Choksi's three questions:
\begin{enumerate}[label=(\arabic*)]
    \item Does it work? Do the partial sums converge to the function? (Section~\ref{sec:fourierC})
    \item What are the coefficients? (this section)
    \item What is special about sines and cosines? (Section~\ref{sec:fourierB})
\end{enumerate}
The one idea: \textbf{in an orthogonal basis, each coefficient is a single inner product, and no linear system has to be solved.}

\subsection{Coefficients in an Orthogonal Basis of $\R^N$}
\begin{definition}[Orthogonal basis]
    $\vec{x}, \vec{y} \in \R^N$ are orthogonal if $\vec{x}\cdot\vec{y} = 0$. A basis $\{\vec{v}_1, \dots, \vec{v}_N\}$ is orthogonal if $\vec{v}_i \cdot \vec{v}_j = 0$ for all $i \neq j$.
\end{definition}
Write $\vec{x} = \sum_i c_i\vec{v}_i$ and dot with $\vec{v}_j$; every term with $i \neq j$ vanishes:
\begin{equation}\label{eq:RN-coeff}
    \vec{x}\cdot\vec{v}_j = c_j\|\vec{v}_j\|^2, \qquad c_j = \frac{\vec{x}\cdot\vec{v}_j}{\|\vec{v}_j\|^2}.
\end{equation}
Each $c_j\vec{v}_j = \operatorname{Proj}_{\vec{v}_j}\vec{x}$, so every vector is the sum of its projections onto an orthogonal basis.
\begin{remark}
    In a non-orthogonal basis, dotting gives an $N \times N$ linear system with matrix entries $\vec{v}_i \cdot \vec{v}_j$; the unknowns are coupled. Orthogonality makes the matrix diagonal. With infinitely many unknowns, a diagonal system is the only kind we can solve by hand.
\end{remark}
\begin{example}[In class]
    $\vec{v}_1 = (1,1,1)$, $\vec{v}_2 = (1,-1,0)$, $\vec{v}_3 = (1,1,-2)$ are mutually orthogonal. For $\vec{x} = (2,-3,5)$, \eqref{eq:RN-coeff} gives
    \begin{equation}
        c_1 = \frac{4}{3}, \qquad c_2 = \frac{5}{2}, \qquad c_3 = \frac{-11}{6},
    \end{equation}
    and indeed $\tfrac43(1,1,1) + \tfrac52(1,-1,0) - \tfrac{11}{6}(1,1,-2) = (2,-3,5)$.
\end{example}

\subsection{Functions as Vectors}
Think of $f$ on $(a,b)$ as a vector with one component $f(x)$ per point. Sampling at $N$ points with spacing $\Delta x$, the dot product times $\Delta x$ is a Riemann sum:
\begin{equation}
    \sum_{i=1}^N f(x_i)g(x_i)\,\Delta x \xrightarrow{N\to\infty} \int_a^b f(x)g(x)\,dx.
\end{equation}
\begin{definition}[Inner product, norm, orthogonality]
    For real functions on $(a,b)$,
    \begin{equation}
        (f,g) = \int_a^b f(x)g(x)\,dx, \qquad \|f\| = (f,f)^{1/2},
    \end{equation}
    and $f, g$ are orthogonal on $(a,b)$ if $(f,g) = 0$. The interval is part of the definition.
\end{definition}
\begin{remark}
    Unlike $\R^N$, $(f,f) = 0$ only forces $f = 0$ except on a set integration cannot see. This is the paradigm of Chapter 5 again.
\end{remark}
\begin{definition}[Projection]
    $\operatorname{Proj}_g f = \dfrac{(f,g)}{\|g\|^2}\,g$.
\end{definition}

\subsection{Orthogonality of the Sines}
\begin{proposition}[Orthogonality relations]
    For positive integers $n \neq m$,
    \begin{equation}
        \int_0^l \sin\left(\frac{n\pi x}{l}\right)\sin\left(\frac{m\pi x}{l}\right)dx = 0, \qquad \int_0^l \sin^2\left(\frac{n\pi x}{l}\right)dx = \frac{l}{2}.
    \end{equation}
    So $X_n(x) = \sin(n\pi x/l)$ are mutually orthogonal on $(0,l)$ with $\|X_n\|^2 = l/2$.
\end{proposition}
\begin{lemma}[Exponentials integrate to zero over a period]
    For an integer $k$,
    \begin{equation}
        \int_{-\pi}^{\pi} e^{ik\theta}\,d\theta = \begin{cases} 2\pi, & k = 0 \\ 0, & k \neq 0. \end{cases}
    \end{equation}
\end{lemma}
\begin{proof}
    For $k \neq 0$ an antiderivative is $e^{ik\theta}/(ik)$, which is $2\pi$-periodic, so it takes the same value at $\pm\pi$.
\end{proof}
\begin{proof}[Proof of the proposition]
    By Euler, $\sin\theta = (e^{i\theta} - e^{-i\theta})/(2i)$, so
    \begin{equation}
        \sin(n\theta)\sin(m\theta) = -\frac14\left[e^{i(n+m)\theta} - e^{i(n-m)\theta} - e^{-i(n-m)\theta} + e^{-i(n+m)\theta}\right].
    \end{equation}
    Integrate over $(-\pi,\pi)$ with the lemma. Since $n + m \neq 0$, the outer terms give $0$. If $n \neq m$ the middle terms give $0$; if $n = m$ they give $2\pi$ each, so the total is $\pi$. The integrand is even, so over $(0,\pi)$ we get $0$ or $\pi/2$. Substituting $\theta = \pi x/l$ gives $0$ or $l/2$.
\end{proof}
\begin{remark}
    Euler's formula replaces the table of product-to-sum identities: any product of sines and cosines is a sum of exponentials, and integrating over a period keeps only frequency $0$. For complex functions $(f,g) = \int f\bar{g}$, and the lemma says $e^{in\theta}$, $n \in \mathbb{Z}$, are orthogonal on $(-\pi,\pi)$. The computation explains nothing; Section~\ref{sec:fourierB} gives the reason.
\end{remark}

\subsection{The Coefficient Formula}
\begin{proposition}[The coefficients are forced]
    Suppose $\phi(x) = \sum_{m=1}^\infty b_m\sin(m\pi x/l)$ on $(0,l)$ and the series can be integrated term by term after multiplying by $\sin(n\pi x/l)$. Then
    \begin{equation}\label{eq:sine-coeff}
        b_n = \frac{2}{l}\int_0^l \phi(x)\sin\left(\frac{n\pi x}{l}\right)dx, \qquad n = 1, 2, \dots
    \end{equation}
\end{proposition}
\begin{proof}
    Multiply by $\sin(n\pi x/l)$, integrate over $(0,l)$, exchange sum and integral, and use orthogonality: only $m = n$ survives, giving $\int_0^l\phi\sin(n\pi x/l)\,dx = \frac{l}{2}b_n$.
\end{proof}
\begin{remark}
    The exchange of sum and integral is an exchange of two limits; uniform convergence on $[0,l]$ allows it, and without some such hypothesis it can fail. The proposition runs one way: \emph{if} $\phi$ has an expansion, \emph{then} the coefficients are \eqref{eq:sine-coeff}. It does not say the series converges, or converges to $\phi$.
\end{remark}
\begin{definition}[11.1.1, Fourier sine series]
    For $\phi$ integrable on $(0,l)$ with $b_n$ from \eqref{eq:sine-coeff}, $\sum_{n=1}^\infty b_n\sin(n\pi x/l)$ is the Fourier sine series of $\phi$ on $(0,l)$.
\end{definition}
As in $\R^N$: $b_n = (\phi, X_n)/\|X_n\|^2$ and $b_nX_n = \operatorname{Proj}_{X_n}\phi$. The factor $\frac{2}{l}$ is $1/\|X_n\|^2$ and nothing more.

\subsection{Cosine and Full Fourier Series}
On $(0,l)$, for integers $n, m \geq 0$,
\begin{equation}
    \int_0^l \cos\left(\frac{n\pi x}{l}\right)\cos\left(\frac{m\pi x}{l}\right)dx = \begin{cases} 0, & n \neq m \\ l/2, & n = m \geq 1 \\ l, & n = m = 0. \end{cases}
\end{equation}
For $\phi(x) = \frac12 a_0 + \sum_{n\geq1} a_n\cos(n\pi x/l)$ this forces
\begin{equation}
    a_n = \frac{2}{l}\int_0^l \phi(x)\cos\left(\frac{n\pi x}{l}\right)dx, \qquad n = 0, 1, 2, \dots
\end{equation}
The $\frac12$ in front of $a_0$ lets one formula cover $n = 0$, since the constant $1$ has twice the squared norm of the other cosines. This is the \emph{Fourier cosine series} (Definition 11.1.2).

On $(-l,l)$, any two different functions of $\{1, \cos(n\pi x/l), \sin(n\pi x/l)\}$ are orthogonal (a sine times a cosine is odd), with $\|1\|^2 = 2l$ and $\|\cos\|^2 = \|\sin\|^2 = l$. So the \emph{full Fourier series} (Definition 11.1.3) is
\begin{gather}
    \frac12 a_0 + \sum_{n=1}^\infty\left[a_n\cos\left(\frac{n\pi x}{l}\right) + b_n\sin\left(\frac{n\pi x}{l}\right)\right], \\
    a_n = \frac1l\int_{-l}^l\phi\cos\frac{n\pi x}{l}\,dx \ (n \geq 0), \qquad b_n = \frac1l\int_{-l}^l\phi\sin\frac{n\pi x}{l}\,dx \ (n \geq 1).
\end{gather}
\begin{remark}[The interval matters]
    On $(0,l)$ sines and cosines are not orthogonal to each other: $\int_0^l\sin(\pi x/l)\cdot 1\,dx = 2l/\pi \neq 0$. That is why the full series lives on $(-l,l)$ and carries $\frac1l$ instead of $\frac2l$.
\end{remark}
Every coefficient is $(\phi, X_n)/\|X_n\|^2$, so the table below can be rebuilt from the $\|X_n\|^2$ column. The cosine and full series start with $\frac12a_0$.
\begin{center}
\small
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{lllll}
    Series & Interval & $X_n$ & $\|X_n\|^2$ & Coefficients \\
    \hline
    Sine & $(0,l)$ & $\sin\frac{n\pi x}{l}$, $n \geq 1$ & $\frac{l}{2}$ & $b_n = \frac{2}{l}\int_0^l\phi\sin\frac{n\pi x}{l}\,dx$ \\
    Cosine & $(0,l)$ & $1$; $\cos\frac{n\pi x}{l}$ & $l$; $\frac{l}{2}$ & $a_n = \frac{2}{l}\int_0^l\phi\cos\frac{n\pi x}{l}\,dx$ \\
    Full & $(-l,l)$ & $1$; $\cos\frac{n\pi x}{l}$, $\sin\frac{n\pi x}{l}$ & $2l$; $l$ & $a_n, b_n = \frac{1}{l}\int_{-l}^l\phi\,(\cos,\sin)\frac{n\pi x}{l}\,dx$
\end{tabular}
\end{center}

\subsection{Examples}
\begin{example}[$\phi \equiv 1$, Choksi 11.1.1]
    Sine series on $(0,l)$:
    \begin{equation}
        b_n = \frac{2}{l}\int_0^l\sin\frac{n\pi x}{l}\,dx = \frac{2}{n\pi}\big(1 - (-1)^n\big) = \begin{cases}\frac{4}{n\pi}, & n \text{ odd} \\ 0, & n \text{ even},\end{cases}
    \end{equation}
    so the sine series of $1$ is
    \begin{equation}\label{eq:SQ}
        \frac{4}{\pi}\sum_{m=0}^\infty \frac{1}{2m+1}\sin\left(\frac{(2m+1)\pi x}{l}\right),
    \end{equation}
    the series at the start of this section. Cosine series: $a_0 = 2$, $a_n = 0$ for $n \geq 1$, so the series is $1$. Nothing to compute, since $1$ is already the $n = 0$ cosine.
\end{example}
\begin{remark}
    If $\phi$ is a finite combination of the $X_n$, its Fourier series is that combination (a finite sum can always be integrated term by term). E.g.\ the full series of $\cos(26\pi x) - 4 - 3\sin(\pi x)$ on $(-1,1)$ is itself: $a_{26} = 1$, $a_0 = -8$, $b_1 = -3$ (Choksi Exercise 11.5(a)).
\end{remark}
\begin{example}[In class: $\phi(x) = x$ on $(0,l)$]
    Sine series (one integration by parts): $b_n = \dfrac{2l(-1)^{n+1}}{n\pi}$, size $1/n$. Cosine series (two integrations by parts): $a_0 = l$ and $a_n = \dfrac{2l\big((-1)^n - 1\big)}{n^2\pi^2}$, i.e.\ $-\frac{4l}{n^2\pi^2}$ for $n$ odd and $0$ for $n$ even, size $1/n^2$. The cosine coefficients decay faster.
\end{example}

\subsection{What the Series Converges To on $\R$}
The series is defined for every $x \in \R$, and the terms decide its shape: each $\sin(n\pi x/l)$ is odd and $2l$-periodic, each $\cos(n\pi x/l)$ even and $2l$-periodic. So a convergent sine series is odd and $2l$-periodic whatever $\phi$ was, and similarly for the others.
\begin{definition}[11.1.4, piecewise continuous]
    $\phi$ on $(a,b)$ is piecewise continuous if it is continuous except at finitely many points, where the one-sided limits $\phi(x\pm) = \lim_{h\to0^+}\phi(x\pm h)$ exist and are finite, and $\phi(a+)$, $\phi(b-)$ exist and are finite.
\end{definition}
\begin{definition}[Extensions]
    For $\phi$ on $(0,l)$: the \emph{odd extension} to $(-l,l)$ is $-\phi(-x)$ for $x < 0$; the \emph{even extension} is $\phi(-x)$ for $x < 0$. The \emph{periodic extension} of a function on $(-l,l)$ repeats it with period $2l$. The odd (even) periodic extension is the periodic extension of the odd (even) extension. Values at the joining points are left unspecified.
\end{definition}
\begin{theorem}[Pointwise convergence of the three series]
    Suppose $\phi$ and $\phi'$ are piecewise continuous on the sampled interval. Let $\phi_{\text{ext}}$ be the odd periodic extension (sine series on $(0,l)$), the even periodic extension (cosine series on $(0,l)$), or the periodic extension (full series on $(-l,l)$). Then for every $x \in \R$ the series converges to
    \begin{equation}
        \frac12\big[\phi_{\text{ext}}(x+) + \phi_{\text{ext}}(x-)\big],
    \end{equation}
    and to $\phi_{\text{ext}}(x)$ wherever $\phi_{\text{ext}}$ is continuous.
\end{theorem}
Taken on trust here; proved for the full series in Section~\ref{sec:fourierC}.
\begin{example}[Back to $\phi \equiv 1$]
    The odd periodic extension of $1$ is the square wave ($1$ on $(0,l)$, $-1$ on $(-l,0)$, period $2l$). It jumps at every multiple of $l$, where the average is $0$. So \eqref{eq:SQ} converges to the square wave, with value $0$ at $x = kl$. The jumps come from the extension, not from $\phi$. The cosine series uses the even extension, which is $1$ everywhere.
\end{example}
\begin{remark}
    Endpoint values are invisible: the sine series gives $0$ at $x = 0, l$ whatever value we assign $\phi$ there, since coefficients are integrals.
\end{remark}
\begin{example}[A free series]
    At $x = l/2$, $\phi_{\text{ext}}$ is continuous and $\sin\big((2m+1)\pi/2\big) = (-1)^m$, so \eqref{eq:SQ} gives
    \begin{equation}
        1 = \frac{4}{\pi}\sum_{m=0}^\infty\frac{(-1)^m}{2m+1}, \qquad \text{that is,} \qquad \frac{\pi}{4} = 1 - \frac13 + \frac15 - \frac17 + \cdots
    \end{equation}
\end{example}
\begin{example}[$\phi(x) = e^x$, Choksi Figure 11.5]
    Sine and cosine series on $(0,1)$, full series on $(-1,1)$, limits on $(-3,3)$ with no coefficient computed:
    \begin{itemize}
        \item Sine: the odd periodic extension jumps at $x = 0$ (from $-1$ to $1$) and at $x = \pm1$ (from $e$ to $-e$). The series gives $0$ there.
        \item Cosine: the even periodic extension is continuous. The series gives $1$ at even integers and $e$ at odd integers.
        \item Full: the periodic extension jumps at odd integers from $e$ down to $e^{-1}$. The series gives $\frac12(e + e^{-1}) = \cosh 1$ there.
    \end{itemize}
\end{example}
\begin{remark}[Decay and smoothness]
    For $\phi = x$, the odd periodic extension is a sawtooth (jumps) and the coefficients are $O(1/n)$; the even periodic extension is a triangle wave (corners) and the coefficients are $O(1/n^2)$. Smoothness of the extension controls the decay (Section~\ref{sec:fourierC}).
\end{remark}
\begin{example}[In class, Choksi 11.3(a)]
    $\phi = 1, 2, 3, 4$ on $(0,1], (1,2], (2,3], (3,4]$, extended with period $4$, full series with $l = 2$. At $x = 0$ the extension jumps from $4$ to $1$, so the series gives $\frac52$. At $x = 1$: $\frac32$. At $x = 4$ and $x = 40$ (multiples of $4$): $\frac52$. At $x = 7.4 \equiv 3.4$: $4$.
\end{example}

\subsection{Orthogonal Is Not Yet a Basis}
\begin{proposition}
    Nonzero, mutually orthogonal $\vec{v}_1, \dots, \vec{v}_N \in \R^N$ form a basis.
\end{proposition}
\begin{proof}
    If $\sum c_i\vec{v}_i = \vec{0}$, dot with $\vec{v}_j$: $c_j\|\vec{v}_j\|^2 = 0$, so $c_j = 0$. Independent, and $N$ independent vectors span $\R^N$.
\end{proof}
The last step counts dimensions; with functions there is nothing to count.
\begin{example}
    Remove the first sine: $\{\sin(n\pi x/l) : n \geq 2\}$ is still orthogonal, but expanding $\phi = \sin(\pi x/l)$ gives every coefficient $0$. Nothing in the computation warned us.
\end{example}
\begin{remark}
    So the formulas come from orthogonality, a calculus fact. \emph{Completeness} (every reasonable $\phi$ is the sum of its projections) is a theorem, ``the miracle'', proved in Section~\ref{sec:fourierC}. The orthogonality itself comes from $X_n = \sin(n\pi x/l)$ satisfying $-X_n'' = \frac{n^2\pi^2}{l^2}X_n$, $X_n(0) = 0 = X_n(l)$: the sines are eigenfunctions of a symmetric operator (Section~\ref{sec:fourierB}).
\end{remark}


\section{Fourier Series B: Eigenfunctions of $-\frac{d^2}{dx^2}$}\label{sec:fourierB}
\marginpar{October 7, 2026}
Source: Choksi 11.2, 11.3; Strauss 5.3. The one idea: \textbf{orthogonality comes from symmetry, and the boundary condition is part of what makes the operator symmetric.}

\subsection{Template: Symmetric Matrices}
\begin{definition}[Symmetric matrix, eigenvector]
    An $N \times N$ matrix $A$ is symmetric if $A\vec{x}\cdot\vec{y} = \vec{x}\cdot A\vec{y}$ for all $\vec{x}, \vec{y}$ (equivalently $A = A^T$). A nonzero $\vec{v}$ is an eigenvector with eigenvalue $\lambda$ if $A\vec{v} = \lambda\vec{v}$.
\end{definition}
\begin{theorem}[11.1, Spectral Theorem]
    A symmetric matrix has real eigenvalues and an orthogonal basis of eigenvectors.
\end{theorem}
\begin{proposition}
    If $A$ is symmetric, $A\vec{v}_1 = \lambda_1\vec{v}_1$, $A\vec{v}_2 = \lambda_2\vec{v}_2$, $\lambda_1 \neq \lambda_2$, then $\vec{v}_1\cdot\vec{v}_2 = 0$.
\end{proposition}
\begin{proof}
    $\lambda_1\,\vec{v}_1\cdot\vec{v}_2 = A\vec{v}_1\cdot\vec{v}_2 = \vec{v}_1\cdot A\vec{v}_2 = \lambda_2\,\vec{v}_1\cdot\vec{v}_2$, so $(\lambda_1 - \lambda_2)\,\vec{v}_1\cdot\vec{v}_2 = 0$.
\end{proof}
\begin{remark}
    Orthogonality is easy and uses only symmetry. Having \emph{enough} eigenvectors for a basis is the hard part. For functions: orthogonality here, completeness in Section~\ref{sec:fourierC}.
\end{remark}

\subsection{The Operator $\mathcal{A} = -\frac{d^2}{dx^2}$}
Fix an interval $[a,b]$ and a boundary condition, and set $\mathcal{A}X = -X''$ on functions satisfying it. $\mathcal{A}$ is linear and plays the role of the matrix.
\begin{definition}[Eigenfunction]
    $\lambda$ is an eigenvalue and $X \not\equiv 0$ an eigenfunction if $X$ satisfies the boundary condition and $\mathcal{A}X = \lambda X$, that is,
    \begin{equation}
        X'' + \lambda X = 0.
    \end{equation}
\end{definition}
Without a boundary condition every $\lambda$ works ($\cos(\sqrt\lambda x)$, $1$, $e^{\sqrt{-\lambda}x}$). \textbf{The boundary condition selects the eigenvalues.} Standard conditions on $[a,b]$:
\begin{itemize}
    \item Dirichlet: $X(a) = 0 = X(b)$.
    \item Neumann: $X'(a) = 0 = X'(b)$.
    \item Periodic: $X(a) = X(b)$ and $X'(a) = X'(b)$.
\end{itemize}
The minus sign in $\mathcal{A}$ makes the eigenvalues nonnegative (below); mathematicians prefer $-\Delta$ to $\Delta$ for the same reason.

\subsection{Green's Second Identity and Symmetric Boundary Conditions}
By the product rule, $-X_1''X_2 + X_1X_2'' = \frac{d}{dx}\left(-X_1'X_2 + X_1X_2'\right)$. Integrating,
\begin{equation}\label{eq:green2}
    \int_a^b\left(-X_1''X_2 + X_1X_2''\right)dx = \Big[X_1X_2' - X_1'X_2\Big]_{x=a}^{x=b}.
\end{equation}
This is \emph{Green's second identity} in 1D, valid for all $C^2$ functions. The left side is $(\mathcal{A}X_1, X_2) - (X_1, \mathcal{A}X_2)$.
\begin{definition}[11.2.1, symmetric boundary condition]
    A boundary condition is symmetric if, whenever $X_1, X_2$ satisfy it,
    \begin{equation}
        X_1(b)X_2'(b) - X_1'(b)X_2(b) = X_1(a)X_2'(a) - X_1'(a)X_2(a).
    \end{equation}
    Then $(\mathcal{A}X_1, X_2) = (X_1, \mathcal{A}X_2)$, and $\mathcal{A}$ is a symmetric operator.
\end{definition}
\begin{example}
    Dirichlet is symmetric: every boundary term contains $X_1$ or $X_2$ at $a$ or $b$, which is $0$.
\end{example}
\begin{example}[In class, Choksi 11.2]
    Neumann: every boundary term contains $X_1'$ or $X_2'$ at an end, which is $0$. Periodic: each of $X_1, X_1', X_2, X_2'$ takes the same value at $a$ and $b$, so the two sides are equal. Both are symmetric.
\end{example}
\begin{example}[Not symmetric, Strauss 5.3]
    $X(a) = X(b)$, $X'(a) = 2X'(b)$. With $p_j = X_j(a) = X_j(b)$, the left side is $p_1X_2'(b) - X_1'(b)p_2$ and the right side is twice that, so the condition fails for most pairs.
\end{example}
\begin{remark}
    Symmetry is not a property of $\frac{d^2}{dx^2}$ alone. The boundary condition is part of the operator, as the entries are part of a matrix.
\end{remark}

\subsection{Consequences of Symmetry}
\begin{theorem}[Orthogonality of eigenfunctions]
    For a symmetric boundary condition, eigenfunctions $X_1, X_2$ with eigenvalues $\lambda_1 \neq \lambda_2$ satisfy $(X_1, X_2) = 0$.
\end{theorem}
\begin{proof}
    $\lambda_1(X_1,X_2) = (\mathcal{A}X_1, X_2) = (X_1, \mathcal{A}X_2) = \lambda_2(X_1,X_2)$.
\end{proof}
This is the reason behind the orthogonality relations of Section~\ref{sec:fourierA}, with no trigonometry and no eigenfunction found yet.
\begin{proposition}[11.3.1, eigenvalues are real]
    For a symmetric boundary condition with real coefficients (such as the three above), every eigenvalue is real.
\end{proposition}
\begin{proof}
    Let $X$ be a complex eigenfunction with eigenvalue $\lambda$. Conjugating, $\overline{X}'' + \bar\lambda\overline{X} = 0$ and $\overline{X}$ satisfies the same condition. Apply \eqref{eq:green2} with $X_1 = X$, $X_2 = \overline{X}$: the right side is $0$ by symmetry, and the left side is
    \begin{equation}
        \int_a^b\left(\lambda X\overline{X} - \bar\lambda X\overline{X}\right)dx = (\lambda - \bar\lambda)\int_a^b|X|^2\,dx.
    \end{equation}
    Since $X \not\equiv 0$, $\lambda = \bar\lambda$.
\end{proof}
With $\lambda$ real, the real and imaginary parts of $X$ are real eigenfunctions, so from now on eigenfunctions are real.
\begin{proposition}[No negative eigenvalues, Green's first identity]
    For Dirichlet, Neumann, or periodic conditions, every eigenvalue satisfies $\lambda \geq 0$, with $\lambda = 0$ only if $X$ is constant.
\end{proposition}
\begin{proof}
    Multiply $-X'' = \lambda X$ by $X$ and integrate by parts:
    \begin{equation}\label{eq:green1}
        \lambda\int_a^b X^2\,dx = -\int_a^b X''X\,dx = \int_a^b (X')^2\,dx - \Big[X'X\Big]_{x=a}^{x=b}.
    \end{equation}
    The boundary term vanishes (a factor is $0$ at each end for Dirichlet and Neumann; equal values cancel for periodic), so $\lambda = \int(X')^2/\int X^2 \geq 0$.
\end{proof}

\subsection{Dirichlet Eigenfunctions}
Take $X'' + \lambda X = 0$, $X(0) = 0 = X(l)$. We know $\lambda \geq 0$.
\begin{itemize}
    \item $\lambda = 0$: $X$ is constant, and $X(0) = 0$ forces $X \equiv 0$. Not an eigenvalue.
    \item $\lambda > 0$: $X = A\cos(\sqrt\lambda x) + B\sin(\sqrt\lambda x)$. $X(0) = 0$ gives $A = 0$; $X(l) = 0$ with $B \neq 0$ gives $\sin(\sqrt\lambda l) = 0$, so $\sqrt\lambda l = n\pi$.
\end{itemize}
\begin{equation}
    \lambda_n = \frac{n^2\pi^2}{l^2}, \qquad X_n(x) = \sin\left(\frac{n\pi x}{l}\right), \qquad n = 1, 2, \dots
\end{equation}
The eigenvalues are distinct, so orthogonality is automatic; only $\|X_n\|^2 = l/2$ still has to be computed. (Choksi rules out $\lambda < 0$ directly on p.~461.)

\subsection{The Four Standard Boundary Conditions}
\begin{center}
\small
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{llll}
    Condition & Eigenvalues & Eigenfunctions & $\|X_n\|^2$ \\
    \hline
    Dirichlet on $[0,l]$: $X(0) = 0 = X(l)$ & $\frac{n^2\pi^2}{l^2}$, $n\geq1$ & $\sin\frac{n\pi x}{l}$ & $\frac{l}{2}$ \\
    Neumann on $[0,l]$: $X'(0) = 0 = X'(l)$ & $\frac{n^2\pi^2}{l^2}$, $n\geq0$ & $\cos\frac{n\pi x}{l}$ & $l$; $\frac{l}{2}$ \\
    Periodic on $[-l,l]$ & $\frac{n^2\pi^2}{l^2}$, $n\geq0$ & $1$; $\cos\frac{n\pi x}{l}$, $\sin\frac{n\pi x}{l}$ & $2l$; $l$ \\
    Mixed on $[0,l]$: $X(0) = 0 = X'(l)$ & $\frac{(n+\frac12)^2\pi^2}{l^2}$, $n\geq0$ & $\sin\frac{(n+\frac12)\pi x}{l}$ & $\frac{l}{2}$
\end{tabular}
\end{center}
The first three give the sine, cosine, and full Fourier series.
\begin{remark}
    \leavevmode
    \begin{itemize}
        \item Neumann: $\lambda = 0$ is an eigenvalue (constant eigenfunction), the $\frac12a_0$ term.
        \item Periodic: each $\lambda_n$, $n \geq 1$, is \emph{degenerate}, with two eigenfunctions $\cos$ and $\sin$. The theorem says nothing about two eigenfunctions with the same eigenvalue. These two happen to be orthogonal (their product is odd), but $\sin$ and $\cos + \sin$ span the same eigenspace and are not. Within a degenerate eigenspace choose an orthogonal pair (Gram--Schmidt).
        \item Mixed: a new series, reused in the pointwise convergence proof.
    \end{itemize}
\end{remark}
\begin{example}[In class, Choksi 11.10(b)]
    $X'' + \lambda X = 0$, $X(0) = 0$, $X'(l) = 0$. Green's first identity \eqref{eq:green1} still applies (at $l$ the factor $X'$ vanishes, at $0$ the factor $X$), so $\lambda \geq 0$. For $\lambda = 0$, $X = A + Bx$ with $A = 0$ and $B = 0$: not an eigenvalue. For $\lambda > 0$, $X = B\sin(\sqrt\lambda x)$ and $X'(l) = B\sqrt\lambda\cos(\sqrt\lambda l) = 0$, so $\sqrt\lambda l = (n + \frac12)\pi$, the mixed row of the table.
\end{example}

\subsection{General Fourier Series}
Fix a symmetric boundary condition with mutually orthogonal eigenfunctions $\{X_n\}$ and eigenvalues $\{\lambda_n\}$ (a degenerate eigenvalue listed once per eigenfunction of an orthogonal pair). If $\phi = \sum A_nX_n$ can be integrated term by term against each $X_n$, then
\begin{equation}\label{eq:general-coeff}
    A_n = \frac{(\phi, X_n)}{\|X_n\|^2} = \frac{\int_a^b\phi X_n\,dx}{\int_a^b X_n^2\,dx}.
\end{equation}
\begin{definition}[11.3.1, general Fourier series]
    With $A_n$ from \eqref{eq:general-coeff}, $\sum_n A_nX_n$ is the general Fourier series of $\phi$. For Dirichlet, Neumann, and periodic conditions it is a \emph{classical} Fourier series.
\end{definition}
\begin{remark}[Why this basis]
    $\mathcal{A}$ is diagonal: formally, $\mathcal{A}\big(\sum A_nX_n\big) = \sum\lambda_nA_nX_n$. Two previews (formal; term-by-term differentiation needs Choksi 11.6):
    \begin{itemize}
        \item $-u'' = f$ on $(0,l)$, $u(0) = u(l) = 0$: with $f = \sum f_nX_n$ and $u = \sum u_nX_n$, each mode gives $\lambda_nu_n = f_n$, so $u_n = f_n/\lambda_n$. Inverting the operator is division, possible since no Dirichlet eigenvalue is $0$.
        \item $u_t = \alpha u_{xx}$ with Dirichlet ends: $u = \sum u_n(t)X_n(x)$ gives $u_n' = -\alpha\lambda_nu_n$, so $u_n(t) = u_n(0)e^{-\alpha\lambda_nt}$. High frequencies decay fastest (Section~\ref{sec:diffA}).
    \end{itemize}
\end{remark}


\section{Fourier Series C: Convergence}\label{sec:fourierC}
\marginpar{October 19, 2026}
Source: Choksi 11.4, 11.5, 11.7; Strauss 5.4. Question 1: does the Fourier series of $\phi$ converge to $\phi$? The one idea: \textbf{the Fourier partial sum is an orthogonal projection, so it is the closest approximation.}

\subsection{Modes of Convergence}
\begin{definition}[Modes of convergence]
    For $f_N, f$ on $(a,b)$:
    \begin{itemize}
        \item pointwise if $f_N(x) \to f(x)$ for every $x \in (a,b)$ (11.5.1);
        \item uniformly on $[a,b]$ if $\sup_{x\in[a,b]}|f_N(x) - f(x)| \to 0$;
        \item in $L^2$ (mean-square) if $\int_a^b|f_N - f|^2\,dx \to 0$, i.e.\ $\|f_N - f\| \to 0$ (11.4.1).
    \end{itemize}
\end{definition}
\begin{definition}[$L^2$]
    $L^2((a,b))$ is the set of $f$ with $\int_a^b|f|^2\,dx < \infty$.
\end{definition}
\begin{example}
    Every piecewise continuous function on $(a,b)$ is in $L^2$. So is the unbounded $|x|^{-1/4}$ on $(-1,1)$, but not $|x|^{-1/2}$.
\end{example}
\begin{remark}
    On a bounded interval, uniform implies both others, since $\int_a^b|f_N - f|^2 \leq (b-a)\sup|f_N - f|^2$. No other implication holds: on $(0,1)$, $x^N \to 0$ pointwise and in $L^2$ ($\int_0^1 x^{2N} = \frac{1}{2N+1}$), but $\sup x^N = 1$.
\end{remark}
\begin{proposition}[Cauchy--Schwarz]
    For $f, g \in L^2((a,b))$, $|(f,g)| \leq \|f\|\,\|g\|$. (Same proof as for vectors, Assignment 3.) So Fourier coefficients of any $\phi \in L^2$ are defined.
\end{proposition}

\subsection{Best Approximation and Bessel's Inequality}
Let $\{X_n\}$ be \emph{any} orthogonal family of nonzero functions, $\phi \in L^2$, $A_n = (\phi,X_n)/\|X_n\|^2$, and $S_N = \sum_{n=1}^N A_nX_n$.
\begin{theorem}[Least squares]
    Among all $\sum_{n=1}^N c_nX_n$, the closest to $\phi$ in $L^2$ is $S_N$ ($c_n = A_n$), and
    \begin{equation}\label{eq:LS}
        \|\phi - S_N\|^2 = \|\phi\|^2 - \sum_{n=1}^N A_n^2\|X_n\|^2.
    \end{equation}
\end{theorem}
\begin{proof}
    Expand; orthogonality removes the cross terms, then complete the square in each $c_n$:
    \begin{align}
        \Big\|\phi - \sum c_nX_n\Big\|^2 &= \|\phi\|^2 - 2\sum c_n(\phi,X_n) + \sum c_n^2\|X_n\|^2 \\
        &= \sum\|X_n\|^2(c_n - A_n)^2 + \|\phi\|^2 - \sum A_n^2\|X_n\|^2.
    \end{align}
    The $c_n$ appear only in the first sum, which is $\geq 0$ and $= 0$ exactly when $c_n = A_n$.
\end{proof}
\begin{remark}[The picture]
    $(\phi - S_N, X_n) = (\phi,X_n) - A_n\|X_n\|^2 = 0$ for $n \leq N$, so $S_N$ is the orthogonal projection onto $\Span\{X_1,\dots,X_N\}$ and \eqref{eq:LS} is Pythagoras. Adding $X_{N+1}$ leaves the earlier coefficients alone.
\end{remark}
\begin{theorem}[11.3, Bessel's inequality]
    For $\phi \in L^2((a,b))$ and any orthogonal family,
    \begin{equation}
        \sum_{n=1}^\infty A_n^2\int_a^b X_n^2\,dx \leq \int_a^b\phi^2\,dx.
    \end{equation}
\end{theorem}
\begin{proof}
    The left side of \eqref{eq:LS} is $\geq 0$; let $N \to \infty$.
\end{proof}
It holds for every orthogonal family, complete or not (with a gap for the sines missing $\sin(\pi x/l)$).
\begin{theorem}[11.5, Riemann--Lebesgue lemma]
    If $\phi \in L^2((a,b))$, its classical Fourier coefficients tend to $0$.
\end{theorem}
\begin{proof}
    For the classical series $\|X_n\|^2$ is the same for all $n \geq 1$, so Bessel gives $\sum A_n^2 < \infty$, and terms of a convergent series tend to $0$.
\end{proof}
\begin{remark}
    Compare $\sin(nx) \to 0$ in Section~\ref{sec:distC}, whose proof needed a derivative of the probe. Bessel needs none: any square-integrable $\phi$ is a valid probe.
\end{remark}

\subsection{Parseval's Equality and Completeness}
Now let $\{X_n\}$ be the eigenfunctions of a symmetric boundary condition. By \eqref{eq:LS}, $\|\phi - S_N\| \to 0$ exactly when the gap in Bessel closes.
\begin{proposition}
    The general Fourier series of $\phi \in L^2$ converges to $\phi$ in $L^2$ if and only if
    \begin{equation}\label{eq:parseval}
        \sum_{n=1}^\infty A_n^2\int_a^bX_n^2\,dx = \int_a^b\phi^2\,dx \qquad \text{(Parseval's equality)}.
    \end{equation}
\end{proposition}
\begin{definition}[Complete]
    An orthogonal family is complete in $L^2((a,b))$ if Parseval's equality holds for every $\phi \in L^2((a,b))$. This is ``the miracle'' made precise.
\end{definition}
\begin{theorem}[11.2 and 11.4, $L^2$ convergence]
    For any symmetric boundary condition and any $\phi \in L^2((a,b))$, Parseval's equality holds and the general Fourier series converges to $\phi$ in $L^2$. The eigenfunctions are complete.
\end{theorem}
Not proved: it needs the spectral theorem for symmetric operators, beyond the course.
\begin{example}[Parseval computes sums]
    $\phi \equiv 1$ on $(0,\pi)$, sine series: $\|X_n\|^2 = \pi/2$, $A_n = \frac{4}{n\pi}$ for $n$ odd. Then
    \begin{equation}
        \sum_{n \text{ odd}}\left(\frac{4}{n\pi}\right)^2\frac{\pi}{2} = \int_0^\pi 1\,dx = \pi, \qquad \text{so} \qquad 1 + \frac19 + \frac{1}{25} + \frac{1}{49} + \cdots = \frac{\pi^2}{8}.
    \end{equation}
\end{example}
\begin{example}[In class, Choksi 11.11(b)]
    $\phi(x) = x$ on $(-\pi,\pi)$ is odd, so $a_n = 0$ and $b_n = \frac{1}{\pi}\int_{-\pi}^\pi x\sin(nx)\,dx = \frac{2(-1)^{n+1}}{n}$. With $\|\sin(nx)\|^2 = \pi$, Parseval gives
    \begin{equation}
        \sum_{n=1}^\infty\frac{4}{n^2}\,\pi = \int_{-\pi}^\pi x^2\,dx = \frac{2\pi^3}{3}, \qquad \text{so} \qquad \sum_{n=1}^\infty\frac{1}{n^2} = \frac{\pi^2}{6}.
    \end{equation}
\end{example}

\subsection{The Dirichlet Kernel}
$L^2$ convergence says nothing about a given point. Take the full series on $(-\pi,\pi)$:
\begin{gather}
    a_n = \frac1\pi\int_{-\pi}^\pi\phi(y)\cos(ny)\,dy, \qquad b_n = \frac1\pi\int_{-\pi}^\pi\phi(y)\sin(ny)\,dy, \\
    S_N(x) = \frac12a_0 + \sum_{n=1}^N\big[a_n\cos(nx) + b_n\sin(nx)\big].
\end{gather}
Substitute the coefficients (finite sum, so it goes inside the integral) and use $\cos(ny)\cos(nx) + \sin(ny)\sin(nx) = \cos\big(n(x-y)\big)$:
\begin{equation}
    S_N(x) = \frac{1}{2\pi}\int_{-\pi}^\pi\left[1 + 2\sum_{n=1}^N\cos\big(n(x-y)\big)\right]\phi(y)\,dy.
\end{equation}
\begin{definition}[Dirichlet kernel]
    For $N \geq 1$, the $2\pi$-periodic function $K_N(x) = 1 + 2\sum_{n=1}^N\cos(nx)$.
\end{definition}
So the partial sum is a convolution:
\begin{equation}\label{eq:SN-conv}
    S_N(x) = \frac{1}{2\pi}\int_{-\pi}^\pi K_N(x - y)\phi(y)\,dy.
\end{equation}
\begin{proposition}[Properties of $K_N$]
    \leavevmode
    \begin{enumerate}[label=(\arabic*)]
        \item $K_N(0) = 2N + 1$, and $K_N$ is even.
        \item $\frac{1}{2\pi}\int_{-\pi}^\pi K_N(x)\,dx = 1$ (every cosine integrates to $0$).
        \item For $x$ not a multiple of $2\pi$,
        \begin{equation}\label{eq:KN}
            K_N(x) = \frac{\sin\big((N+\frac12)x\big)}{\sin(x/2)}.
        \end{equation}
    \end{enumerate}
\end{proposition}
\begin{proof}[Proof of (3)]
    $K_N(x) = \sum_{n=-N}^N e^{inx}$, a geometric sum with ratio $e^{ix} \neq 1$. Sum it and multiply top and bottom by $e^{-ix/2}$:
    \begin{equation}
        K_N(x) = e^{-iNx}\frac{1 - e^{i(2N+1)x}}{1 - e^{ix}} = \frac{e^{-i(N+\frac12)x} - e^{i(N+\frac12)x}}{e^{-ix/2} - e^{ix/2}} = \frac{\sin\big((N+\frac12)x\big)}{\sin(x/2)}. \qedhere
    \end{equation}
\end{proof}
\begin{remark}
    For $S_N(x) \to \phi(x)$ we need $\frac{1}{2\pi}K_N(x - y)$ to act like $\delta_x$. Unit mass and a peak hold, but $K_N$ is not nonnegative and its oscillations away from $0$ do not shrink. So the proof of Theorem 5.1 does not apply, and continuity of $\phi$ will not be enough.
\end{remark}

\subsection{Pointwise Convergence}
\begin{theorem}[11.6, pointwise convergence of the full series]
    Suppose $\phi$ and $\phi'$ are piecewise continuous on $(-\pi,\pi)$ and $\phi((-\pi)+)$, $\phi'((-\pi)+)$, $\phi(\pi-)$, $\phi'(\pi-)$ exist. Extend $\phi$ periodically to $\R$. Then for every $x$,
    \begin{equation}
        S_N(x) \to \frac{\phi(x+) + \phi(x-)}{2},
    \end{equation}
    and $S_N(x) \to \phi(x)$ wherever the periodic extension is continuous.
\end{theorem}
\begin{proof}[Proof when the periodic extension is $C^1$]
    Fix $x$. In \eqref{eq:SN-conv} substitute $y = x + \theta$; since $K_N$ is even and everything is $2\pi$-periodic, $S_N(x) = \frac{1}{2\pi}\int_{-\pi}^\pi K_N(\theta)\phi(x+\theta)\,d\theta$. By property (2), $\phi(x) = \frac{1}{2\pi}\int K_N(\theta)\phi(x)\,d\theta$. Subtract and use \eqref{eq:KN}:
    \begin{equation}
        S_N(x) - \phi(x) = \frac{1}{2\pi}\int_{-\pi}^\pi\psi(\theta)\sin\big((N+\tfrac12)\theta\big)\,d\theta, \qquad \psi(\theta) = \frac{\phi(x+\theta) - \phi(x)}{\sin(\theta/2)}.
    \end{equation}
    \textbf{An orthogonal family.} $X_N(\theta) = \sin\big((N+\frac12)\theta\big)$, $N = 0, 1, 2, \dots$, are the mixed eigenfunctions ($X(0) = 0 = X'(\pi)$), hence orthogonal on $(0,\pi)$. Each is odd, so they are also orthogonal on $(-\pi,\pi)$, with $\|X_N\|^2 = \pi$. So
    \begin{equation}
        S_N(x) - \phi(x) = \frac{1}{2\pi}(\psi, X_N) = \frac12B_N, \qquad B_N = \frac{(\psi,X_N)}{\|X_N\|^2}.
    \end{equation}
    \textbf{Bessel finishes it, if $\psi \in L^2$.} Bessel gives $\sum_NB_N^2\pi \leq \|\psi\|^2$, so $B_N \to 0$ and $S_N(x) \to \phi(x)$.\par
    \textbf{$\psi$ is bounded.} $\sin(\theta/2)$ vanishes on $[-\pi,\pi]$ only at $\theta = 0$. By the mean value theorem $|\phi(x+\theta) - \phi(x)| \leq M|\theta|$ with $M = \max|\phi'|$, and $|\sin(\theta/2)| \geq |\theta|/\pi$ for $|\theta| \leq \pi$ (concavity of $\sin$ on $[0,\pi/2]$). So $|\psi| \leq \pi M$ and $\psi \in L^2$.
\end{proof}
\begin{remark}
    The case with jumps splits the integral at $\theta = 0$ and uses $\phi(x+)$ on $(0,\pi)$ and $\phi(x-)$ on $(-\pi,0)$ (Choksi p.~476). The derivative was used only to bound $\psi$ near $0$. It cannot be dropped entirely: some continuous functions have Fourier series diverging at a point.
\end{remark}
\begin{corollary}
    \leavevmode
    \begin{itemize}
        \item For $\phi \in C_c^\infty((-\pi,\pi))$ and $x = 0$: $\frac{1}{2\pi}\int K_N\phi = S_N(0) \to \phi(0)$, so $K_N \to 2\pi\delta_0$ in the sense of distributions on $(-\pi,\pi)$ (the preview of Section~\ref{sec:distC}).
        \item The convergence theorem of Section~\ref{sec:fourierA} follows for the sine and cosine series by applying Theorem 11.6 to the odd or even extension (Assignment 3).
    \end{itemize}
\end{corollary}

\subsection{Uniform Convergence and the Gibbs Phenomenon}
\begin{theorem}[11.9, uniform convergence]
    If $\phi$ is continuous and piecewise $C^1$ on $[-\pi,\pi]$ with $\phi(-\pi) = \phi(\pi)$, its full Fourier series converges absolutely and uniformly to the periodic extension of $\phi$ on $\R$.
\end{theorem}
\begin{remark}
    $\phi(-\pi) = \phi(\pi)$ says the periodic extension has no jumps. It is needed: each $S_N$ is continuous, and a uniform limit of continuous functions is continuous. The square wave is a counterexample.
\end{remark}
\begin{remark}[Gibbs phenomenon]
    Near a jump the partial sums overshoot. For the square wave $\pm1$, the first maximum of $S_N$ right of $0$ tends to $\frac{2}{\pi}\int_0^\pi\frac{\sin x}{x}\,dx \approx 1.18$, about $9\%$ of the jump. It does not shrink with $N$, it only moves toward the jump.
\end{remark}
\begin{remark}[Smoothness and decay]
    If the periodic extension of $\phi$ is continuous, the Fourier coefficients of $\phi'$ are $n$ times those of $\phi$ (Choksi Theorem 11.7, one integration by parts). Each continuous derivative of the periodic extension buys one more power of $1/n$.
\end{remark}

\subsection{Summary: The Three Questions}
\begin{enumerate}[label=(\arabic*)]
    \item \textbf{Does it work?} In $L^2$ for any $\phi \in L^2$ and any symmetric condition (stated). Pointwise, to the average at jumps, for $\phi, \phi'$ piecewise continuous (proved via the Dirichlet kernel and Bessel). Uniformly for a continuous periodic extension with piecewise continuous derivative; at jumps it cannot, and Gibbs appears.
    \item \textbf{What are the coefficients?} $A_n = (\phi,X_n)/\|X_n\|^2$, the orthogonal projection, giving the best $L^2$ approximation.
    \item \textbf{Why sines and cosines?} They are eigenfunctions of the symmetric operator $-\frac{d^2}{dx^2}$ with Dirichlet, Neumann, or periodic conditions. Symmetry gives orthogonality and real eigenvalues.
\end{enumerate}


\section{The Diffusion Equation A: Separation of Variables}\label{sec:diffA}
\marginpar{October 21, 2026}
Source: Choksi 12.1, 7.1, 7.6; Strauss 4.1, 4.2. The diffusion (heat) equation is
\begin{equation}\label{eq:heat}
    u_t = \alpha u_{xx}, \qquad \alpha > 0,
\end{equation}
where $u(x,t)$ is a temperature or a concentration and $\alpha$ is the diffusivity (Strauss writes $k$). The initial data is $g$. In $N$ dimensions, $u_t = \alpha\Delta u$ with $\Delta u = \operatorname{div}\nabla u = u_{x_1x_1} + \cdots + u_{x_Nx_N}$. The one idea: \textbf{in the eigenfunction basis the diffusion equation is a decoupled list of ODEs, and mode $n$ decays like $e^{-\alpha\lambda_nt}$. High frequencies die first.}

\subsection{The Dirichlet Problem on an Interval}
A thin bar on $0 \leq x \leq l$, insulated along its sides, ends held at temperature $0$, initial temperature $g$:
\begin{equation}\label{eq:heat-dir}
    \begin{cases}
        u_t = \alpha u_{xx}, & 0 < x < l,\ t > 0, \\
        u(0,t) = u(l,t) = 0, & t > 0, \\
        u(x,0) = g(x), & 0 < x < l.
    \end{cases}
\end{equation}
\begin{remark}
    The boundary conditions are imposed for $t > 0$ only, so $g$ need not vanish at the ends (a bar at temperature $1$ plunged into ice water at $t = 0$). They are \emph{homogeneous}: if $u, v$ satisfy them, so does $au + bv$. The PDE is linear too, so superposition applies to both.
\end{remark}

\subsection{Separated Solutions}
Look for $u(x,t) = X(x)T(t)$. Then $XT' = \alpha X''T$, and dividing by $\alpha XT$,
\begin{equation}
    \frac{T'(t)}{\alpha T(t)} = \frac{X''(x)}{X(x)}.
\end{equation}
The left side depends only on $t$ and the right only on $x$, so both equal a constant $-\lambda$:
\begin{equation}
    X'' + \lambda X = 0, \qquad T' = -\alpha\lambda T.
\end{equation}
The boundary conditions give $X(0)T(t) = 0 = X(l)T(t)$, so for a nonzero solution $X(0) = 0 = X(l)$. This is the Dirichlet eigenvalue problem (Section~\ref{sec:fourierB}):
\begin{equation}
    \lambda_n = \left(\frac{n\pi}{l}\right)^2, \qquad X_n(x) = \sin\left(\frac{n\pi x}{l}\right), \qquad n = 1, 2, \dots
\end{equation}
Then $T_n(t) = e^{-\alpha\lambda_nt}$, and
\begin{equation}
    u_n(x,t) = \sin\left(\frac{n\pi x}{l}\right)\exp\left(-\frac{\alpha n^2\pi^2t}{l^2}\right)
\end{equation}
solves the PDE and boundary conditions: a fixed shape decaying in place. Mode $n$ has $n - 1$ interior zeros and decays $n^2$ times faster than mode $1$.
\begin{remark}
    $\lambda$ is not a choice: the boundary conditions force it to be an eigenvalue. All eigenvalues are real and positive, so there are no growing modes.
\end{remark}

\subsection{Superposition and the Initial Condition}
Take an infinite sum of the $u_n$:
\begin{equation}\label{eq:heat-series}
    u(x,t) = \sum_{n=1}^\infty b_n\sin\left(\frac{n\pi x}{l}\right)e^{-\alpha n^2\pi^2t/l^2}.
\end{equation}
At $t = 0$ this is $\sum b_n\sin(n\pi x/l)$, so to match $g$ take the Fourier sine coefficients:
\begin{equation}\label{eq:heat-coeff}
    b_n = \frac{2}{l}\int_0^l g(x)\sin\left(\frac{n\pi x}{l}\right)dx.
\end{equation}
\begin{method}[Separation of variables, Choksi p.~498]
    \begin{enumerate}[label=(\roman*)]
        \item Find separated solutions.
        \item Impose the boundary conditions to get an eigenvalue problem.
        \item Solve for $T_n$.
        \item Superpose.
        \item Choose the coefficients by projection.
    \end{enumerate}
    Steps (i) to (iii) find the eigenfunctions; (iv) and (v) find the coefficients.
\end{method}
\begin{example}[Hot bar plunged into ice water, Strauss 4.1 Exercise 2]
    $g \equiv 1$. From Section~\ref{sec:fourierA}, $b_n = \frac{4}{n\pi}$ for $n$ odd and $0$ for $n$ even, so
    \begin{equation}
        u(x,t) = \frac{4}{\pi}\sum_{n \text{ odd}}\frac1n\sin\left(\frac{n\pi x}{l}\right)e^{-\alpha n^2\pi^2t/l^2}.
    \end{equation}
    At $t = 0$ the series converges slowly, like $\sum 1/n$, to a function that jumps at the ends. For any $t > 0$ the factors $e^{-\alpha n^2\pi^2t/l^2}$ make it converge faster than any geometric series. Once $\alpha t$ is comparable to $l^2$, only the first term matters:
    \begin{equation}
        u(x,t) \approx \frac{4}{\pi}\sin\left(\frac{\pi x}{l}\right)e^{-\alpha\pi^2t/l^2}.
    \end{equation}
    The ratio of the third term to the first is $\frac13e^{-8\alpha\pi^2t/l^2}$, about $10^{-4}$ at $t = l^2/(\alpha\pi^2)$.
\end{example}
\begin{remark}[Time scale]
    The bar cools on the time scale $l^2/(\alpha\pi^2)$. Doubling the length multiplies the cooling time by four: time scales like length squared.
\end{remark}
\begin{example}[In class]
    $l = \pi$, $\alpha = 1$, $g(x) = \sin x + 3\sin 5x$. No integrals are needed: $g$ is already a finite sine series, so $b_1 = 1$, $b_5 = 3$, and
    \begin{equation}
        u(x,t) = e^{-t}\sin x + 3e^{-25t}\sin 5x.
    \end{equation}
    At $t = 0.1$: $u \approx 0.90\sin x + 0.25\sin 5x$, the wiggles are already much smaller. At $t = 1$: $3e^{-25} \approx 4\times10^{-11}$, so $u \approx e^{-1}\sin x$, a single hump.
\end{example}
\begin{example}[Easy exam question]
    $\alpha = 1$, $l = \pi$, $g(x) = 3\sin x + 10\sin 7x$ on $(0,\pi)$. Again $g$ is already a sine series, so $b_1 = 3$, $b_7 = 10$, and
    \begin{equation}
        u(x,t) = 3e^{-t}\sin x + 10e^{-7^2t}\sin 7x.
    \end{equation}
    A harder exam question would give a general $g$.\footnote{For a hard version, compute the Fourier sine coefficients of $g$ with \eqref{eq:heat-coeff}, then put them into \eqref{eq:heat-series}, as in the hot bar example.}
\end{example}

\subsection{Neumann Boundary Conditions}
Insulate the ends as well:
\begin{equation}\label{eq:heat-neu}
    \begin{cases}
        u_t = \alpha u_{xx}, & 0 < x < l,\ t > 0, \\
        u_x(0,t) = u_x(l,t) = 0, & t > 0, \\
        u(x,0) = g(x), & 0 < x < l.
    \end{cases}
\end{equation}
Separation now gives $X'' + \lambda X = 0$, $X'(0) = 0 = X'(l)$: $\lambda_n = (n\pi/l)^2$, $X_n = \cos(n\pi x/l)$, $n \geq 0$. The new feature is $\lambda_0 = 0$ with constant $X_0$ and constant $T_0$:
\begin{equation}
    u(x,t) = \frac12a_0 + \sum_{n=1}^\infty a_n\cos\left(\frac{n\pi x}{l}\right)e^{-\alpha n^2\pi^2t/l^2}, \qquad a_n = \frac2l\int_0^l g(x)\cos\left(\frac{n\pi x}{l}\right)dx.
\end{equation}
As $t \to \infty$ every mode with $n \geq 1$ dies, and
\begin{equation}
    u(x,t) \to \frac12a_0 = \frac1l\int_0^l g(x)\,dx.
\end{equation}
The bar equalizes to its average initial temperature.
\begin{proposition}[Conservation of heat]
    If $u$ is a smooth solution of \eqref{eq:heat-neu}, then $\int_0^l u(x,t)\,dx$ is constant in $t$.
\end{proposition}
\begin{proof}
    \begin{equation}
        \frac{d}{dt}\int_0^l u\,dx = \int_0^l\alpha u_{xx}\,dx = \alpha\big(u_x(l,t) - u_x(0,t)\big) = 0. \qedhere
    \end{equation}
\end{proof}
So the only constant the solution can approach is the average. Under Dirichlet conditions the boundary term does not vanish and heat leaves through the ends.

\subsection{Where the Equation Comes From}
Measure heat so that the heat in $[a,b]$ is $\int_a^b u\,dx$, and let $F(x,t)$ be the \emph{flux}: the rate at which heat crosses $x$ from left to right.\par
\textbf{Conservation.} Heat in $[a,b]$ changes only by flow through the endpoints:
\begin{equation}
    \frac{d}{dt}\int_a^b u\,dx = F(a,t) - F(b,t) = -\int_a^b F_x\,dx.
\end{equation}
This holds for every $[a,b]$, so $u_t = -F_x$.\par
\textbf{Fourier's law (Fick's law).} Heat flows from hot to cold at a rate proportional to the slope:
\begin{equation}\label{eq:fourier-law}
    F = -\alpha u_x.
\end{equation}
Together, $u_t = \alpha u_{xx}$. In $N$ dimensions the divergence theorem gives $u_t = -\operatorname{div}\vec{F}$ and $\vec{F} = -\alpha\nabla u$, so $u_t = \alpha\Delta u$. (Choksi Exercise 7.1: how $\alpha$ depends on density, specific heat, and conductivity.) The physics is in \eqref{eq:fourier-law}: read $u_x$ as minus the flux.

\subsection{Boundary Conditions and Heat Flow}
Read each condition at the left end $x = 0$:
\begin{itemize}
    \item \textbf{Dirichlet}, $u(0,t) = T_0$: the end touches a reservoir so large that its temperature never changes. Homogeneous case $T_0 = 0$.
    \item \textbf{Neumann}, $u_x(0,t) = 0$: the flux $F(0,t) = -\alpha u_x(0,t)$ is $0$, the end is insulated.
    \item \textbf{Robin}: the end touches a medium at $T_0$, and heat passes at a rate proportional to the temperature difference (Newton's law of cooling). The heat entering at $x = 0$ is $F(0,t)$, so for some $h > 0$
    \begin{equation}
        -\alpha u_x(0,t) = h\big(T_0 - u(0,t)\big), \qquad \text{i.e.} \qquad u_x(0,t) = \frac{h}{\alpha}\big(u(0,t) - T_0\big).
    \end{equation}
    At the right end the heat leaving is $F(l,t)$, giving $-\alpha u_x(l,t) = h\big(u(l,t) - T_0\big)$. As $h \to 0$ this becomes Neumann; as $h \to \infty$, Dirichlet.
\end{itemize}
\begin{remark}[A sign to check]
    If $T_0 > u(0,t)$, heat enters, $F(0,t) > 0$, so $u_x(0,t) < 0$: the temperature decreases going into the bar. Choksi (7.6.2, p.~300) writes $u_x(0,t) = c_1(T_0 - u(0,t))$ and states $u_x(0,t) > 0$, the opposite sign. The form above agrees with Strauss 4.3 and Choksi Exercises 12.5 and 12.7.
\end{remark}
\textbf{Steady states} (Choksi (7.30) to (7.32), 12.3.1). A steady state $v(x) = \lim_{t\to\infty}u(x,t)$ satisfies $v'' = 0$, so $v$ is linear, and the boundary conditions pick the line:
\begin{center}
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{ll}
    Boundary conditions & Steady state $v(x)$ \\
    \hline
    $u(0,t) = T_0$, $u(l,t) = T_1$ & $T_0 + (T_1 - T_0)\,x/l$ \\
    $u(0,t) = T_0$, $u_x(l,t) = 0$ & $T_0$ \\
    $u_x(0,t) = 0 = u_x(l,t)$ & $\frac1l\int_0^l g\,dx$
\end{tabular}
\end{center}
For Neumann conditions every constant satisfies $v'' = 0$ and the boundary conditions; conservation of heat picks the constant.
\begin{remark}[Inhomogeneous Dirichlet conditions]
    Separation needs homogeneous conditions (two solutions with $u(0,t) = T_0$ sum to $2T_0$). Subtract the steady state: if $u$ solves the first row and $v$ is its steady state, $w = u - v$ solves \eqref{eq:heat-dir} with data $g - v$. So
    \begin{equation}
        u(x,t) = v(x) + \sum_{n=1}^\infty b_n\sin\left(\frac{n\pi x}{l}\right)e^{-\alpha n^2\pi^2t/l^2},
    \end{equation}
    with $b_n$ the sine coefficients of $g - v$. The solution approaches the line at rate $\alpha\pi^2/l^2$. The second row uses the mixed eigenfunctions $\sin\big((n+\frac12)\pi x/l\big)$ (Choksi 12.3.2, Exercise 12.3).
\end{remark}

\subsection{The Solution Operator and the Kernel (Optional)}
For a symmetric matrix $A$ with orthonormal eigenvectors $\vec{v}_n$ and eigenvalues $\lambda_n$, the system $\vec{x}\,' = -\alpha A\vec{x}$, $\vec{x}(0) = \vec{x}_0$, has solution
\begin{equation}
    \vec{x}(t) = \sum_n e^{-\alpha\lambda_nt}(\vec{x}_0\cdot\vec{v}_n)\,\vec{v}_n.
\end{equation}
The coordinates decouple and each decays on its own. Formula \eqref{eq:heat-series} is this with $A = -\frac{d^2}{dx^2}$ and Dirichlet conditions: with orthonormal $e_n = \sqrt{2/l}\,\sin(n\pi x/l)$ and $c_n(t) = (u(\cdot,t), e_n)$,
\begin{equation}
    c_n'(t) = -\alpha\lambda_nc_n(t), \qquad c_n(0) = (g, e_n).
\end{equation}
\begin{definition}[Solution operator]
    $S(t)g = u(\cdot,t)$. It multiplies the $n$th coordinate by $e^{-\alpha\lambda_nt}$.
\end{definition}
\begin{proposition}
    \leavevmode
    \begin{enumerate}[label=(\arabic*)]
        \item $S(t)S(s) = S(t+s)$: running for time $s$ then $t$ is running for $t + s$.
        \item $\|S(t)g\| \leq e^{-\alpha\lambda_1t}\|g\|$, by Parseval. The $L^2$ norm decreases.
    \end{enumerate}
\end{proposition}
\textbf{The kernel.} Put \eqref{eq:heat-coeff} into \eqref{eq:heat-series} and exchange sum and integral (allowed for $t > 0$ since the terms decay so fast):
\begin{equation}
    u(x,t) = \int_0^l K(x,y,t)\,g(y)\,dy, \qquad K(x,y,t) = \frac2l\sum_{n=1}^\infty\sin\left(\frac{n\pi x}{l}\right)\sin\left(\frac{n\pi y}{l}\right)e^{-\alpha n^2\pi^2t/l^2}.
\end{equation}
The solution at $x$ is a weighted sum of the data with weight $K(x,\cdot,t)$ (Choksi calls it $\Phi$ in (12.7); here $\Phi$ is reserved for the heat kernel on the line).
\begin{remark}[Shape of $K$]
    \leavevmode
    \begin{itemize}
        \item For small $t$ it is a narrow bump at $y = x$, matching the Gaussian heat kernel on the line away from the ends: over a short time only data near $x$ matters.
        \item It is never negative and vanishes at $y = 0, l$: data at the ends, held at $0$, contribute nothing.
        \item Its total weight $\int_0^lK\,dy$ decreases in $t$, faster near the ends. The missing weight is heat lost through the ends, so $u$ is a weighted average with total weight at most $1$ and shrinks toward $0$.
    \end{itemize}
\end{remark}

\subsection{Summary}
\begin{itemize}
    \item Separation of variables reduces the diffusion equation on an interval to the eigenvalue problem for $-\frac{d^2}{dx^2}$ with the given boundary condition; the coefficients are Fourier coefficients of the data.
    \item Mode $n$ decays like $e^{-\alpha\lambda_nt}$. So the solution is $C^\infty$ for every $t > 0$, settles onto the first mode at rate $\alpha\lambda_1$, and the problem cannot be run backward.
    \item In the eigenbasis the solution operator is diagonal, as for $\vec{x}\,' = -\alpha A\vec{x}$.
    \item Boundary conditions are statements about the flux $F = -\alpha u_x$: fixed temperature, zero flux, or flux proportional to a temperature difference.
    \item Still open: is \eqref{eq:heat-series} the only solution? Answered in Lecture B without the formula.
\end{itemize}

\subsubsection*{Problem \ref{prob:a1-vc} (A1 Q1, vector calculus), p.~\pageref{prob:a1-vc}}
\textbf{(1)} Let $\vec{x} = (x,y,z)$, so $f = x^2 + y^2 + z^2$. Then
\begin{equation*}
    \nabla f = (2x, 2y, 2z), \qquad \nabla f(1,0,0) = (2,0,0),
\end{equation*}
\begin{equation*}
    D_{\vec{v}}f(1,0,0) = \nabla f(1,0,0)\cdot\vec{v} = (2,0,0)\cdot\tfrac{1}{\sqrt2}(1,1,0) = \frac{2}{\sqrt2} = \sqrt2.
\end{equation*}
The level sets $f = c$ are spheres of radius $\sqrt c$. In the plane ($z = 0$), $f = x^2 + y^2$ has circular contours (e.g.\ $c = 1, 4, 9$), and $\nabla f = 2\vec{x}$ points radially outward, perpendicular to each circle, longer farther out.

\textbf{(2)} Linear approximation: $f(\vec{x} + h\vec{v}) \approx f(\vec{x}) + \nabla f(\vec{x})\cdot h\vec{v}$, so
\begin{equation*}
    D_{\vec{v}}f(\vec{x}) = \lim_{h\to0}\frac{f(\vec{x}+h\vec{v}) - f(\vec{x})}{h} = \nabla f(\vec{x})\cdot\vec{v}.
\end{equation*}
By Cauchy--Schwarz, $D_{\vec{v}}f(\vec{x}) \leq \|\nabla f(\vec{x})\|\,\|\vec{v}\| = \|\nabla f(\vec{x})\|$. Since $\nabla f(\vec{x}) \neq \vec{0}$, set $\vec{v}^* = \nabla f(\vec{x})/\|\nabla f(\vec{x})\|$; then $\nabla f\cdot\vec{v}^* = \|\nabla f\|^2/\|\nabla f\| = \|\nabla f\|$, so the bound is attained.
Uniqueness: if $\nabla f\cdot\vec{v} = \|\nabla f\|$, dividing by $\|\nabla f\|$ gives $\vec{v}\cdot\vec{v}^* = 1$, and
\begin{equation*}
    \|\vec{v} - \vec{v}^*\|^2 = \|\vec{v}\|^2 - 2\vec{v}\cdot\vec{v}^* + \|\vec{v}^*\|^2 = 1 - 2 + 1 = 0,
\end{equation*}
so $\vec{v} = \vec{v}^*$.

\textbf{(3)}
\begin{itemize}
    \item $f = x^2 + 2y^2 + 3z^2$: $f_x = 2x$, $f_y = 4y$, $f_z = 6z$ vanish only at the origin, and $H = \operatorname{diag}(2,4,6)$ with leading minors $\Delta_1 = 2$, $\Delta_2 = 8$, $\Delta_3 = 48$, all positive. Local minimum.
    \item $f = x^2 + y^2 - z^2$: critical point $(0,0,0)$, $H = \operatorname{diag}(2,2,-2)$, $\Delta_1 = 2$, $\Delta_2 = 4$, $\Delta_3 = -8$. Saddle point.
    \item $f = xy$: $f_x = y$, $f_y = x$, critical point $(0,0)$, $H = \begin{pmatrix}0&1\\1&0\end{pmatrix}$, $\Delta_1 = 0$, $\Delta_2 = -1$. Eigenvalues: $\det(H - \lambda I) = \lambda^2 - 1 = 0$, so $\lambda = \pm1$. Mixed signs, so a saddle point.\footnote{Viewed as a function on $\R^3$, the Hessian also has the eigenvalue $0$ from $x_3$; the mixed signs $\pm1$ still give a saddle.}
\end{itemize}

(III) $S > 0$, $0 \leq t \leq T$, terminal condition $u(S,T) = \max(S - K, 0)$ for a call with strike $K$, boundary condition $u(0,t) = 0$.

    \item[(o)] Fisher--KPP, $u_t = au_{xx} + bu(1-u)$. (I) Second order, semilinear. (II) Models the spread of an advantageous gene in a population: $u$ is the gene frequency (a normalized density), $x$ position, $t$ time, $a$ the diffusion coefficient, $b$ the growth rate. (III) $x \in \R$, $t > 0$, $u(x,0) = g(x)$ with $0 \leq g \leq 1$.
\end{itemize}

\phantomsection\label{sol:a1-16}
\subsubsection*{Problem \ref{prob:a1-16} (A1 Q2, Choksi 1.6), p.~\pageref{prob:a1-16}}
\textbf{(a)} Existence for every $\vec{b}$ means $A$ is surjective, i.e.\ $\operatorname{rank}A = N$. Uniqueness means $\ker A = \{\vec{0}\}$. For a square matrix these are equivalent, and both hold iff $A$ is invertible ($\det A \neq 0$).

\textbf{(b)} Stability means small changes in $\vec{b}$ produce small changes in $\vec{x}$: there is a constant $C$ with
\begin{equation*}
    A\vec{x}_1 = \vec{b}_1,\ A\vec{x}_2 = \vec{b}_2 \implies \|\vec{x}_1 - \vec{x}_2\| \leq C\|\vec{b}_1 - \vec{b}_2\|.
\end{equation*}
Since $\vec{x}_1 - \vec{x}_2 = A^{-1}(\vec{b}_1 - \vec{b}_2)$, $C = \|A^{-1}\|$. For $A$ symmetric positive definite, $\|A^{-1}\| = 1/\lambda_{\min}$, so $\|\vec{x}_1 - \vec{x}_2\| \leq \frac{1}{\lambda_{\min}}\|\vec{b}_1 - \vec{b}_2\|$. The smaller $\lambda_{\min}$, the more a perturbation in $\vec{b}$ is amplified, so the less stable the problem.

\subsubsection*{Problem \ref{prob:a2-52} (A2 Q1, Choksi 5.2), p.~\pageref{prob:a2-52}}
\textbf{Step 1: $f'$.} Since $f = 0$ on $(-\infty,0)$, $\ip{f'}{\phi} = -\int_{-\infty}^\infty f\phi'\,dx = -\int_0^\infty x\phi'(x)\,dx$. Integrate by parts with $u = x$, $dv = \phi'dx$:
\begin{equation*}
    -\int_0^\infty x\phi'\,dx = -\left(\big[x\phi(x)\big]_0^\infty - \int_0^\infty\phi\,dx\right).
\end{equation*}
At $x = 0$ the factor $x$ is $0$, and for large $x$, $\phi = 0$. So $\ip{f'}{\phi} = \int_0^\infty\phi\,dx = \int_{-\infty}^\infty H(x)\phi(x)\,dx$, i.e.\ $f' = H$.\par
\textbf{Step 2: $f''$.} $\ip{f''}{\phi} = -\ip{H}{\phi'} = -\int_0^\infty\phi'\,dx = -\left(\lim_{R\to\infty}\phi(R) - \phi(0)\right) = \phi(0) = \ip{\delta_0}{\phi}$. So $f'' = \delta_0$.

\phantomsection\label{sol:a2-53}
\subsubsection*{Problem \ref{prob:a2-53} (A2 Q2, Choksi 5.3), p.~\pageref{prob:a2-53}}
\textbf{Step 1.} For $x \in (2m-1, 2m+1]$, $f(x) = |x - 2m|$, so $f(k) = 0$ for even $k$ and $f(k) = 1$ for odd $k$. On $(2m, 2m+1)$, $f = x - 2m$ and $f' = 1$; on $(2m-1, 2m)$, $f = 2m - x$ and $f' = -1$. Let $g$ be this square wave, with value $g_k$ on $(k, k+1)$: $g_k = +1$ for even $k$, $-1$ for odd $k$.\par
\textbf{Step 2: $f'$.} Choose an integer $N$ with $\operatorname{supp}\phi \subset [-N,N]$. Then $\ip{f'}{\phi} = -\int_{-N}^Nf\phi' = -\sum_{k=-N}^{N-1}\int_k^{k+1}f\phi'$. On each $[k,k+1]$, $f$ is $C^1$ and continuous at the endpoints, so
\begin{equation*}
    \int_k^{k+1}f\phi' = f(k+1)\phi(k+1) - f(k)\phi(k) - \int_k^{k+1}g\phi.
\end{equation*}
The boundary terms telescope: $\sum_{k=-N}^{N-1}\big(f(k+1)\phi(k+1) - f(k)\phi(k)\big) = f(N)\phi(N) - f(-N)\phi(-N) = 0$ since $\phi(\pm N) = 0$. So $\ip{f'}{\phi} = \int g\phi$, i.e.\ $f' = g$.\par
\textbf{Step 3: $f''$.} $\ip{f''}{\phi} = -\int g\phi' = -\sum_kg_k\int_k^{k+1}\phi' = -\sum_kg_k\big(\phi(k+1) - \phi(k)\big)$. Reindexing with $j = k+1$,
\begin{equation*}
    \ip{f''}{\phi} = \sum_jg_j\phi(j) - \sum_jg_{j-1}\phi(j) = \sum_j(g_j - g_{j-1})\phi(j).
\end{equation*}
For even $j$ the jump is $1 - (-1) = 2$, for odd $j$ it is $-1 - 1 = -2$. Hence
\begin{equation*}
    f'' = \sum_{m\in\mathbb{Z}}\big(2\delta_{2m} - 2\delta_{2m+1}\big).
\end{equation*}
Only finitely many terms are nonzero for each $\phi$, so this is a well-defined distribution.\par
\textbf{Sketches.} $f$: a triangle wave between $0$ (at even integers) and $1$ (at odd integers). $f'$: a square wave, $+1$ on $(2m,2m+1)$, $-1$ on $(2m-1,2m)$. $f''$: upward spikes of weight $2$ at even integers, downward spikes of weight $-2$ at odd integers.

\subsubsection*{Problem \ref{prob:a2-jumps} (A2 Q4, several jumps; not submitted, solution added)}

\subsubsection*{Problem \ref{prob:a2-511} (A2 Q5, Choksi 5.11), p.~\pageref{prob:a2-511}}
Let $\ip{T}{\phi} = \sum_{n=1}^\infty\phi(n)$.\par
\textbf{Step 1: finite.} If $\operatorname{supp}\phi \subset [-R,R]$, then $\phi(n) = 0$ for $n > R$, so the sum has at most $\lfloor R\rfloor$ nonzero terms.\par
\textbf{Step 2: linear.} The sums are finite, so $\sum(a\phi + b\psi)(n) = a\sum\phi(n) + b\sum\psi(n)$.\par

\begin{equation*}
    |\ip{T}{\phi_k} - \ip{T}{\phi}| \leq \sum_{n\leq R}|\phi_k(n) - \phi(n)| \leq R\sup|\phi_k - \phi| \to 0.
\end{equation*}

\begin{equation*}
    \ip{f'}{\phi} = -\int f\phi' = -\sum_{n=1}^Nn\int_n^{n+1}\phi' = -\sum_{n=1}^Nn\big(\phi(n+1) - \phi(n)\big).
\end{equation*}

\textbf{Step 1: uniqueness and symmetry.} If $u_1, u_2$ both solve the problem, $w = u_1 - u_2$ satisfies $w'' = 0$ on $[-1,1]$, so $w = ax + b$, and $w(-1) = -a + b = 0$, $w(1) = a + b = 0$ give $a = b = 0$. Differentiating $u(-x) = u(x)$ gives $-u'(-x) = u'(x)$, and $x = 0$ gives $u'(0) = 0$.\par
\textbf{Step 2: solve for $x \geq 0$.} $u'$ is continuous at $\pm\frac\epsilon2$ because $u'(x) = u'(0) + \int_0^xf_\epsilon$, and $u$ is continuous because it is differentiable, so $u$ and $u'$ can be matched at $x = \frac\epsilon2$.
\begin{itemize}
    \item On $[0,\frac\epsilon2]$: $u'' = \frac1\epsilon$ and $u'(0) = 0$, so $u' = \frac x\epsilon$ and $u = \frac{x^2}{2\epsilon} + b$.
    \item On $[\frac\epsilon2, 1]$: $u'' = 0$ and $u' = u'(\frac\epsilon2) = \frac12$. With $u(1) = 0$, $u = \frac12(x - 1)$.
    \item Continuity of $u$ at $\frac\epsilon2$: $\frac\epsilon8 + b = \frac\epsilon4 - \frac12$, so $b = \frac\epsilon8 - \frac12$.
\end{itemize}
\textbf{Step 3: extend evenly.}
\begin{equation*}
    u_\epsilon(x) = \begin{cases} \dfrac{x^2}{2\epsilon} + \dfrac\epsilon8 - \dfrac12, & |x| \leq \frac\epsilon2, \\[2mm] \dfrac{|x| - 1}{2}, & \frac\epsilon2 \leq |x| \leq 1. \end{cases}
\end{equation*}
\textbf{Step 4: limit.} Let $u_0 = \frac{|x| - 1}{2}$. For $|x| \geq \frac\epsilon2$, $u_\epsilon = u_0$. For $|x| \leq \frac\epsilon2$,
\begin{equation*}
    u_\epsilon - u_0 = \frac{x^2 - \epsilon|x| + \frac{\epsilon^2}{4}}{2\epsilon} = \frac{(|x| - \frac\epsilon2)^2}{2\epsilon} \in \left[0, \frac\epsilon8\right].
\end{equation*}
So $\sup|u_\epsilon - u_0| \leq \frac\epsilon8$, and for $\phi \in C_c^\infty((-1,1))$, $|\ip{u_\epsilon}{\phi} - \ip{u_0}{\phi}| \leq \frac\epsilon8\int|\phi| \to 0$.\par
Note: the limit is $u_0 = \Phi - \frac12$, not $\Phi$ exactly. The boundary conditions $u(\pm1) = 0$ force the constant $-\frac12$. It differs from $\Phi$ by a constant and has the same second derivative $\delta_0$.

\end{equation*}
\textbf{Step 2.} Let $A(x) = \int_{-\infty}^xf$, $B(x) = \int_{-\infty}^xyf(y)\,dy$, $c_0 = \int f$, $c_1 = \int yf$, all finite since $f$ has compact support. Then $\int_x^\infty f = c_0 - A$ and $\int_x^\infty yf = c_1 - B$, so
\begin{equation*}
    2u = xA - B + (c_1 - B) - x(c_0 - A) = 2xA - 2B + c_1 - c_0x.
\end{equation*}
\textbf{Step 3.} Since $f$ and $yf$ are continuous, the Fundamental Theorem of Calculus gives $A' = f$ and $B' = xf(x)$. So
\begin{equation*}
    2u' = 2A + 2xf - 2xf - c_0 = 2A - c_0, \qquad 2u'' = 2f.
\end{equation*}
Hence $u'' = f$ classically.

\subsubsection*{Problem \ref{prob:a2-d5} (A2 Q16), p.~\pageref{prob:a2-d5}}
\textbf{Step 1.} If $\psi = \chi'$ with $\chi \in \D$, then for large $R$, $\int\psi = \chi(R) - \chi(-R) = 0$.\par
\textbf{Step 2.} Conversely, suppose $\int\psi = 0$ and $\operatorname{supp}\psi \subset [-R,R]$. Define $\chi(x) = \int_{-\infty}^x\psi$. Then $\chi' = \psi$, so $\chi$ is smooth. For $x < -R$, $\chi(x) = 0$; for $x > R$, $\chi(x) = \int\psi = 0$. So $\chi \in \D$.\par
\textbf{Step 3.} Fix $\phi_0 \in \D$ with $\int\phi_0 = 1$. For any $\phi$, the function $\psi = \phi - (\int\phi)\phi_0$ satisfies $\int\psi = 0$, so by Step 2, $\psi = \chi'$ for some $\chi \in \D$.\par
\textbf{Step 4.} $\ip{G}{\psi} = \ip{G}{\chi'} = -\ip{G'}{\chi} = 0$.\par
\textbf{Step 5.} By linearity, $\ip{G}{\phi} = \ip{G}{\psi} + (\int\phi)\ip{G}{\phi_0} = c\int\phi$ with $c := \ip{G}{\phi_0}$. So $G$ is the constant function $c$.\par
\textbf{Step 6.} Let $T'' = \delta_0$ and $W = T - \frac12|x|$. Then $W'' = \delta_0 - \delta_0 = 0$. Since $(W')' = 0$, Step 5 gives $W' = a$ for a constant $a$. Also the derivative of the function $ax$ is $a$, because $-\int ax\phi' = \int a\phi$. So $(W - ax)' = 0$, and Step 5 gives $W - ax = b$. Therefore $T = \frac12|x| + ax + b$.
