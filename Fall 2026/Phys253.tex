\section{Thermodynamics}
\marginpar{August 31 - September 2, 2026}
\begin{definition}[Thermodynamics]
    Thermodynamics, from the ancient Greek \textit{therme} (heat) and \textit{dynamis} (power), is the branch of physics that deals with the relationships between heat, work, temperature and energy. It studies the equilibrium properties of large-scale systems in which temperature is an important variable.
\end{definition}
\begin{remark}
    Thermodynamics does not describe the microscopic details of a system\footnote{Statistical mechanics, and more recent extensions such as stochastic thermodynamics, do.}. It describes the macroscopic properties and behaviour of a system using a small number of state variables.
\end{remark}

\subsection{Terminologies}
\begin{definition}[System, surroundings, boundary]
    The system is everything involved in the process under study; the surroundings are the rest of the universe it interacts with; the boundary is the interface between them.
\end{definition}
\begin{definition}[Open, closed, isolated]
    An open system exchanges matter and energy with its surroundings, a closed system only energy, and an isolated system neither. The boundary determines which.
\end{definition}

\subsection{Equilibrium States}
\begin{definition}[Equilibrium state]
    A system is in an equilibrium state if all the measurable bulk physical properties of the system are uniform throughout the system and constant in time.
\end{definition}
\begin{remark}
    For a fixed volume of gas in an equilibrium state, the pressure is the same everywhere in the gas and the volume does not change in time.
\end{remark}
\begin{remark}
    The pressure, volume and mass of a gas fully describe its equilibrium state: once those are fixed, every other measurable bulk property is fixed as well.
\end{remark}

\begin{definition}[Local thermodynamic equilibrium]
    Divide the system into small volume elements. A volume element $V_i$ is in local thermodynamic equilibrium if the properties inside $V_i$ are the same as those of an equilibrium system at the local $(P, V, T)$ conditions. %can i say instead of at the local... by in the surroundings?
\end{definition}
\begin{remark}
    %what?
    This lets a system with, say, a temperature gradient still be treated cell by cell, with $U$, $H$, $S$ defined in each cell. The approximation is better when relaxation is slow and the gradients in $(P,V,T)$ are gentle.
\end{remark}
\begin{remark}
    Thermodynamics studies equilibrium states themselves, not how long it takes to reach them. \footnote{Rates of processes belong to a separate subject (transport theory or kinetics).}
\end{remark}

\subsection{Equations of State and State Functions}
\begin{definition}[Equation of state]
    An equation of state is a functional relationship between the easily measured state variables\footnote{Such as pressure, volume and temperature.} of a system in an equilibrium state.
\end{definition}
\begin{law}[Ideal gas law]
    For a dilute gas at high temperature in an equilibrium state,
    \begin{equation}
        PV = nRT = N k_B T,
    \end{equation}
    where $n$ is the number of moles, $N = n N_A$ is the number of molecules, $R$ is the ideal gas constant and $k_B = R/N_A = \SI{1.38e-23}{\joule\per\kelvin}$ is Boltzmann's constant.
\end{law}
\begin{example}
    The ideal gas law is an equation of state relating $(P, V, T)$.
\end{example}
\begin{definition}[State function]
    A state function is a property of a system that depends only on the current equilibrium state, and not on the path taken to reach that state. Internal energy, enthalpy, entropy and Gibbs free energy are all state functions.
\end{definition}
\begin{remark}
    State functions matter because their values can be compared between two equilibrium states without knowing anything about the process that connected them.
\end{remark}
\begin{example}
    Temperature is a state function. It is the property that determines whether two systems are in thermal equilibrium\footnote{Thermal equilibrium is not full thermodynamic equilibrium, which also requires mechanical and chemical (diffusive) equilibrium.} with each other.
\end{example}

\section{Laws of Thermodynamics}
\subsection{Zeroth Law}
\begin{law}[Zeroth Law of Thermodynamics]
    If two systems A and B are each separately in thermal equilibrium with a third system C, meaning no heat flows between A and C and no heat flows between B and C, then A and B are in thermal equilibrium with each other.
\end{law}

\begin{remark}
    The zeroth law is what makes a thermometer work. System C is the thermometer, and the law guarantees that two systems reading the same value on it are in thermal equilibrium with each other.
\end{remark}
\begin{remark}
    %What?
    Extrapolating $V(T)$ at fixed $P$ for any sample of ideal gas gives $V \to 0$ at the same temperature, $-273.15\,^\circ$C $= 0$ K, which defines absolute zero. A gas thermometer reads $T = PV/nR$.
\end{remark}

\subsection{First Law}
\begin{law}[First Law of Thermodynamics]
    The relationship between the heat $Q$ added to a system, the work $W$ done on it and the change in its internal energy $U$ is dictated by the conservation of energy:
    \begin{equation}
        \label{eq:first-law}
        \Delta U = Q + W.
    \end{equation}
\end{law}

\begin{remark}[Sign Convention]
    $Q > 0$ when heat flows into the system. $W > 0$ when work is done on the system. Both raise the internal energy.
\end{remark}
\begin{remark}
    For an isolated system $Q = W = 0$, so $U$ is constant.
\end{remark}
\begin{remark}
    The first law says energy is conserved. The second law (Section~\ref{sec:second-law}) says in which direction spontaneous processes go.
\end{remark}

\section{Energy, Work and Heat}
\marginpar{September 4, 2026}

\subsection{Energy}
\begin{definition}[Energy]
    Energy, from the Greek \textit{energeia} meaning "in work," is the capacity to do work.
\end{definition}

\begin{definition}[Kinetic energy]
    From the Greek \textit{kinesis} (motion), the energy associated with the motion of a body or of the particles inside it.
\end{definition}

\begin{definition}[Potential energy]
    Energy due to condition, position or composition. It is associated with interactions, meaning forces of attraction or repulsion between objects, or chemical bonding.
\end{definition}

\begin{definition}[Internal energy of an ideal gas]
    For an ideal gas the internal energy depends only on temperature. It is the number of molecules times the average kinetic energy per molecule:
    \begin{equation}
        \label{eq:internal-energy}
        U(T) = N \bar{e}_k = \frac{f}{2} N k_B T,
    \end{equation}
    where $f$ is the number of degrees of freedom per molecule. For a monatomic gas $f = 3$ and $f = 5$ for diatomic, so $U = \tfrac{3}{2} N k_B T$. More complex molecules have extra rotational or vibrational degrees of freedom, so $f > 3$.
\end{definition}

\subsection{Work}
\begin{definition}[Work]
    Work is done when a force acts through a distance, and it is associated with a change in the energy of a system:
    \[W = \int \vec{F} \cdot d\vec{r}.\]
    In thermodynamics, work is any transfer of energy into or out of a system that is not heat.
\end{definition}
\begin{definition}[Quasistatic Process]
    Idealized thermodynamic process that happens slowly enough for a system to stay in continuous internal equilibrium.
\end{definition}

\paragraph{Compression of a gas by a piston.}
Take a gas in a cylinder closed by a piston of area $A$. Pushing the piston in by $\delta x$ does work $W = F\,\delta x = PA\,\delta x$. For a quasistatic compression\footnote{Slow enough that the gas stays in internal equilibrium at every instant, so the pressure is uniform throughout the volume and the equation of state still applies.}, $\delta V = -A \, \delta x$, so the work done \textit{on} the gas is
\[W = -P \, \delta V \qquad \text{(quasistatic)}.\]
Compressing the gas means $\delta V < 0$, so $W > 0$ and the energy of the gas goes up. For a finite change,
\begin{equation}
    \label{eq:work}
    W = -\int_{V_i}^{V_f} P \, dV,
\end{equation}
which is the area under the curve on a $P$-$V$ diagram.

\begin{remark}
    Energy and work are measured in joules, $\SI{1}{\joule} = \SI{1}{\newton\metre} = \SI{1}{\kilo\gram\metre\squared\per\second\squared}$.
\end{remark}

\subsection{Heat}
\begin{definition}[Heat]
    Heat $Q$ is energy that flows between a system and its surroundings because of a temperature difference. It flows spontaneously from the warmer body to the colder one, by conduction, convection or radiation, until $T_A = T_B$. No temperature difference, no heat flow.
\end{definition}

\begin{remark}
    Heat and work are not properties of a system. They are the two ways energy can be exchanged between a system and its surroundings. A system does not "contain" heat or work, it contains energy.
\end{remark}

\section{Reversible Processes and Differentials}

\subsection{Reversible and Irreversible Processes}
\begin{definition}[Thermodynamic process]
    A process is the mechanism that takes a system from one equilibrium state to another. The two equilibrium states are its end points.
\end{definition}

\begin{definition}[Reversible process]
    A process is reversible (quasistatic) if the system passes through a succession of equilibrium states, so the equation of state holds at every point, and at every step the system and surroundings could be returned to their original states. This requires
    \begin{itemize}
        \item reversible work: mechanical equilibrium, $P_{\text{ext}} = P_{\text{sys}}$;
        \item reversible heat: thermal equilibrium, $T_{\text{surr}} = T_{\text{sys}}$.
    \end{itemize}
\end{definition}
\begin{remark}
    Reversible processes are an idealization: nothing would actually happen unless there were at least a tiny difference $dP$ or $dT$ between system and surroundings. They are still useful: a state function's change depends only on the end points, so it can always be computed along a reversible path.
\end{remark}

\begin{example}[Isothermal expansion of an ideal gas, $V_i \to V_f$]
    Reversible, with $P = Nk_BT/V$ throughout:
    \[W_{\text{rev}} = -\int_{V_i}^{V_f} \frac{Nk_BT}{V}\,dV = -Nk_BT\ln\frac{V_f}{V_i}.\]
    Irreversible, in one step against $P_{\text{ext}} = P_f \neq P_{gas}$:
    \[W_{\text{irr}} = -P_f(V_f - V_i).\]
    Same end temperature, so $\Delta U = 0$ and $Q = -W$ in both, but $|W_{\text{irr}}| < |W_{\text{rev}}|$: the rectangle under $P_f$ is smaller than the area under the isotherm.
\end{example}
\begin{remark}
    A reversible path gives the most work out of an expansion and needs the least work in for a compression. $W$ and $Q$ depend on the path; $\Delta U$ does not.
\end{remark}

\subsection{Exact and Inexact Differentials}
For a function $Z(x,y)$, a differential change has the form
\begin{equation}
    \label{eq:dZ}
    dZ = M(x,y)\,dx + N(x,y)\,dy.
\end{equation}

\begin{theorem}[Test for exactness]
    $dZ$ is exact (path independent, so $Z$ is a state function) if and only if
    \begin{equation}
        \left(\frac{\partial M}{\partial y}\right)_x = \left(\frac{\partial N}{\partial x}\right)_y.
    \end{equation}
    Otherwise $dZ$ is inexact (path dependent).
\end{theorem}
\begin{example}[Internal energy of an ideal gas]
    For $U(V,T)$, since $U$ depends only on $T$,
    \[dU = \left(\frac{\partial U}{\partial V}\right)_T dV + \left(\frac{\partial U}{\partial T}\right)_V dT = 0\,dV + C_V\,dT, \qquad C_V = \tfrac{f}{2}Nk_B.\]
    Testing: $M = 0$ and $N = C_V$ is constant, so
    \[\left(\frac{\partial M}{\partial T}\right)_V = 0 = \left(\frac{\partial N}{\partial V}\right)_T.\]
    The differential is exact, so $U$ is a state function.
\end{example}
\begin{example}[Work]
    For a reversible process,
    \[\dbar W = -P\,dV + 0\,dT.\]
    Testing: $M = -P = -nRT/V$ and $N = 0$, so
    \[\left(\frac{\partial M}{\partial T}\right)_V = -\frac{nR}{V} \neq 0 = \left(\frac{\partial N}{\partial V}\right)_T.\]
    The differential is inexact, so $W$ is not a state function.
\end{example}

\begin{remark}
    Inexact differentials are written $\dbar Q$ and $\dbar W$ to signal that they depend on the path.
\end{remark}
Totals are integrals along the path:
\[W = \int \dbar W = -\int_{V_1}^{V_2} P \, dV, \qquad Q = \int \dbar Q,\]
so different paths between the same two states\footnote{Same start and end states, different route in between. A different end state is not a different path.} give different $W$ and $Q$, but the same $\Delta U = Q + W$.

\subsection{Material Properties of a Compressible Fluid}
A fluid has an equation of state $g(P,V,T) = 0$\footnote{for an ideal gas, $g = PV - Nk_BT$}. Picking two of $(P,V,T)$ fixes the third, so we can write $V = V(P,T)$. By the chain rule, a small change in $V$ is
\[dV = \left(\frac{\partial V}{\partial T}\right)_P dT + \left(\frac{\partial V}{\partial P}\right)_T dP,\]
where the subscript is the variable held fixed. These two derivatives are measurable properties of the material:

\begin{definition}[Thermal expansivity, bulk modulus, compressibility]
    \begin{equation}
        \beta = \frac{1}{V}\left(\frac{\partial V}{\partial T}\right)_P, \qquad B = -V\left(\frac{\partial P}{\partial V}\right)_T, \qquad \kappa = \frac{1}{B} = -\frac{1}{V}\left(\frac{\partial V}{\partial P}\right)_T.
    \end{equation}
\end{definition}

In terms of these,
\begin{equation}
    dV = \beta V\,dT - \frac{V}{B}\,dP.
\end{equation}

Likewise, write $dU$ two ways. By the chain rule \eqref{eq:dZ}, and by the first law \eqref{eq:first-law} on a reversible step where $\dbar W = -P\,dV$ from \eqref{eq:work},
\[dU = \left(\frac{\partial U}{\partial V}\right)_T dV + \left(\frac{\partial U}{\partial T}\right)_V dT = \dbar Q_{\text{rev}} - P\,dV.\]
These are equal as a whole, not term by term. Solving for the heat,
\begin{equation}
    \label{eq:dQrev}
    \dbar Q_{\text{rev}} = \left(\frac{\partial U}{\partial T}\right)_V dT + \left[\left(\frac{\partial U}{\partial V}\right)_T + P\right] dV.
\end{equation}
At constant volume ($dV = 0$), $\dbar Q_{\text{rev}} = (\partial U/\partial T)_V\,dT = C_V\,dT$.

\begin{remark}
    $(\partial U/\partial V)_T$ is not $-P$: that would be the change of $U$ with $V$ when no heat flows, not at constant $T$. For an ideal gas $(\partial U/\partial V)_T = 0$, since particles do not interact. For a real fluid it is nonzero and hard to measure directly; it will be recast into measurable form later using Maxwell relations.
\end{remark}

\section{Heat Capacity}
\marginpar{September 11, 2026}

\subsection{Definition}
\begin{definition}[Heat capacity]
    The heat capacity $C$ of a system is the heat needed to raise its temperature by \SI{1}{\kelvin}:
    \begin{equation}
        \label{eq:heat-capacity}
        C = \frac{Q}{\Delta T}.
    \end{equation}
\end{definition}
\begin{definition}[Specific heat capacity]
    The specific heat capacity $c \equiv C/m$ is intensive and a property of the material. With \eqref{eq:heat-capacity}, the heat needed to change the temperature of a mass $m$ by $\Delta T$ is
    \begin{equation}
        \label{eq:mcdT}
        Q = mc\,\Delta T.
    \end{equation}
\end{definition}

\subsection{Heat Capacity Depends on the Process}
Using the first law, $Q = \Delta U - W$, so
\[C = \frac{\Delta U - W}{\Delta T}.\]
$C$ depends on how much work is done during the heating (it would almost be better named the "energy capacity"), so the process must be specified.

\begin{result}[Heat capacity at constant volume]
    If $V$ is fixed, $W = 0$ and
    \[C_V = \left(\frac{\partial U}{\partial T}\right)_V.\]
\end{result}
\begin{result}[Heat capacity at constant pressure]
    Most materials expand when heated\footnote{Some materials contract over certain ranges of $T$ (negative thermal expansion), but they are the exception.}, doing work $W = -P\,\Delta V < 0$ on the surroundings, so extra heat is needed. Dividing \eqref{eq:dQrev} by $dT$ at constant $P$,
    \begin{equation}
        C_P = C_V + \left[\left(\frac{\partial U}{\partial V}\right)_T + P\right]\left(\frac{\partial V}{\partial T}\right)_P.
    \end{equation}
\end{result}

\begin{remark}
    For solids and liquids $(\partial V/\partial T)_P \approx 0$, so $C_P \approx C_V$; for gases the correction is significant.
\end{remark}

\begin{proposition}[$C_P$ of an ideal gas]
    $U$ depends only on $T$, so $(\partial U/\partial T)_P = C_V$, and $(\partial V/\partial T)_P = Nk_B/P$. Hence
    \[C_P = C_V + Nk_B = C_V + nR, \qquad c_P = c_V + R \text{ (molar)}.\]
\end{proposition}

\subsection{Enthalpy}
\begin{definition}[Enthalpy]
    Enthalpy is the state function
    \begin{equation}
        \label{enthalpy}
        H = U + PV.
    \end{equation}
    It is the convenient energy to use for processes at constant pressure.
\end{definition}

\begin{proposition}
    At constant $P$, $dH = \dbar Q$, and therefore
    \begin{equation}
        C_P = \left(\frac{\partial H}{\partial T}\right)_P.
    \end{equation}
\end{proposition}
\begin{proof}
    Differentiate \ref{enthalpy} with the product rule, $dH = dU + P\,dV + V\,dP$. On a reversible step, the first law \eqref{eq:first-law} with $\dbar W = -P\,dV$ from \eqref{eq:work} gives $dU = \dbar Q_{\text{rev}} - P\,dV$. Substituting,
    \[dH = \left(\dbar Q_{\text{rev}} - P\,dV\right) + P\,dV + V\,dP = \dbar Q_{\text{rev}} + V\,dP.\]
    At constant pressure $dP = 0$, so $dH = \dbar Q_{\text{rev}}$. The heat capacity \eqref{eq:heat-capacity} is the heat absorbed per unit temperature change, so at constant pressure
    \[C_P = \left(\frac{\dbar Q_{\text{rev}}}{dT}\right)_P = \left(\frac{\partial H}{\partial T}\right)_P.\]
\end{proof}

\subsection{Calorimetry}
\begin{definition}[Calorimeter]
    A calorimeter is an insulated container in which two bodies at different temperatures reach a common final temperature. Since the whole is isolated,
    \[Q_{\text{system}} = -Q_{\text{surroundings}}.\]
\end{definition}
\begin{example}
    \SI{150.0}{\gram} of lead at \SI{100}{\celsius} is dropped into \SI{50.0}{\gram} of water at \SI{22.0}{\celsius}; the final temperature is \SI{28.8}{\celsius}. With $c_{\text{water}} = \SI{4.18}{\joule\per\gram\per\kelvin}$, \eqref{eq:mcdT} gives
    \[q_{\text{water}} = (\SI{50}{\gram})(\SI{4.18}{\joule\per\gram\per\celsius})(\SI{6.8}{\celsius}) = \SI{1.4e3}{\joule} = -q_{\text{lead}},\]
    \[c_{\text{lead}} = \frac{q_{\text{lead}}}{m_{\text{lead}}\,\Delta T_{\text{lead}}} = \frac{-\SI{1.4e3}{\joule}}{(\SI{150}{\gram})(28.8 - 100)\,\si{\celsius}} = \SI{0.13}{\joule\per\gram\per\celsius}.\]
\end{example}

\subsection{The Mechanical Equivalent of Heat}
In Joule's experiment (1850), a falling weight spins a paddle wheel in insulated water; the lost potential energy $mg\,\Delta h$ raises the water's temperature. He found about \SI{4.2}{\joule} per gram per \si{\celsius}, close to the modern \SI{4.18}{\joule\per\gram\per\kelvin}. This showed that $W$ and $Q$ are interchangeable forms of energy, and that a given amount of work always produces the same amount of heat (energy is conserved).

\section{Quasistatic Processes of an Ideal Gas}
\marginpar{September 9 and 14, 2026}

Since $PV = Nk_BT$, how $P$ changes with $V$ depends on what happens to $T$, so $W = -\int P\,dV$ depends on the process. All processes below are quasistatic. Figure~\ref{fig:pv-processes} shows them on a $P$-$V$ diagram.

    \label{fig:pv-processes}
\end{figure}

\subsection{Isobaric and Isochoric}
\begin{definition}[Isobaric process]
    A process at constant pressure.
\end{definition}
On a $P$-$V$ diagram it is a horizontal line. With $P$ constant, \eqref{eq:work} and the definition of $C_P$ give
\[W = -P(V_2 - V_1), \qquad Q = \int_{T_1}^{T_2} C_P\,dT = C_P\,\Delta T \quad \text{(constant $C_P$)}.\]

\begin{definition}[Isochoric process]
    A process at constant volume.
\end{definition}
On a $P$-$V$ diagram it is a vertical line. With $dV = 0$, \eqref{eq:work} gives no work, so by the first law \eqref{eq:first-law} all the heat goes into $U$:
\[W = 0, \qquad Q = \Delta U = C_V\,\Delta T.\]

\subsection{Isothermal}
\begin{definition}[Isothermal process]
    A process at constant temperature.
\end{definition}
On a $P$-$V$ diagram it follows the hyperbola $P = Nk_BT/V$, called an isotherm (Figure~\ref{fig:isothermal}). The gas is in contact with a reservoir that supplies or absorbs heat. With $P = Nk_BT/V$,
\[W = -\int_{V_i}^{V_f} \frac{N k_B T}{V} \, dV = N k_B T \ln\frac{V_i}{V_f}.\]
Since $U$ depends only on $T$ \eqref{eq:internal-energy}, $\Delta U = \frac{f}{2}Nk_B(T_f-T_i) = 0$ and
\[Q = -W = N k_B T \ln\frac{V_f}{V_i}.\]
\begin{remark}
    In an expansion the gas does work and absorbs an equal heat from the reservoir; in a compression the work done on the gas is dumped into the reservoir as heat.
\end{remark}

    \label{fig:isothermal}
\end{figure}

\subsection{Adiabatic}
\begin{definition}[Adiabatic process]
    A process with no heat flow, $Q = 0$.
\end{definition}
On a $P$-$V$ diagram it follows an adiabat, $PV^\gamma = \text{constant}$ (Figure~\ref{fig:adiabat}). The gas is insulated, or the process is too fast for heat to flow. The first law \eqref{eq:first-law} gives $W = \Delta U = C_V\,\Delta T$: an adiabatic compression heats the gas and an expansion cools it.

\paragraph{Equation of the adiabat.}
With $Q = 0$, $dU = \dbar W$. Using \eqref{eq:internal-energy} for $dU$, \eqref{eq:work} for $\dbar W$, and $P = Nk_BT/V$,
\[\frac{f}{2} N k_B \, dT = -\frac{Nk_BT}{V} \, dV \quad\Longrightarrow\quad \frac{f}{2} \frac{dT}{T} = -\frac{dV}{V} \quad\Longrightarrow\quad \frac{f}{2} \ln\frac{T_f}{T_i} = -\ln\frac{V_f}{V_i} \quad\Longrightarrow\quad V T^{f/2} = \text{constant}.\]

\begin{proposition}[Adiabat in terms of $P$ and $V$]
    \[P V^{\gamma} = \text{constant}, \qquad TV^{\gamma - 1} = \text{constant}, \qquad \gamma = \frac{f+2}{f} = \frac{C_P}{C_V}.\]
\end{proposition}
\begin{proof}
    Raise $VT^{f/2} = \text{constant}$ to the power $2/f$: $TV^{2/f} = \text{constant}$, and $2/f = \gamma - 1$. Substituting $T = PV/Nk_B$ gives $PV^{1 + 2/f} = PV^\gamma = \text{constant}$. Finally, $C_V = \tfrac{f}{2}Nk_B$ and $C_P = C_V + Nk_B = \tfrac{f+2}{2}Nk_B$, so $C_P/C_V = (f+2)/f = \gamma$.
\end{proof}

\begin{remark}
    Since $\gamma > 1$, an adiabat is steeper than an isotherm and crosses from one to a colder (expansion) or hotter (compression) one. Monatomic: $\gamma = 5/3$. Diatomic (air): $\gamma = 7/5$.
\end{remark}

\section{Kinetic Theory of Gases}
\marginpar{September 14, 2026}

\begin{definition}[Kinetic theory]
    Kinetic theory is a microscopic model for $U(T)$ and $C$ of an ideal gas: relate $P$ to the kinetic energy of the particles, then use $PV = Nk_BT$ to relate that to $T$.
\end{definition}

\subsection{Pressure from a Single Particle}
One particle of mass $m$ bounces between the walls of a box of length $L$ with $x$-velocity $u_x$. Each collision reverses its momentum, giving the wall
\[\Delta p_{\text{wall}} = -(-m u_x - m u_x) = 2 m u_x,\]
once every round trip $\Delta t = 2L/u_x$. The average force is
\[\bar{F} = \frac{\Delta p_{\text{wall}}}{\Delta t} = \frac{m u_x^2}{L} \qquad \text{(one particle)}.\]

\subsection{Pressure from All $N$ Particles}
Summing over $N$ particles and dividing by the wall area $A$ ($AL = V$),
\[P = \frac{N m \overline{u_x^2}}{V} \quad \Longrightarrow \quad PV = N m \overline{u_x^2}.\]

\subsection{Temperature and the Average Kinetic Energy}
\marginpar{September 16, 2026}
No direction is preferred, so $\overline{u^2} = 3\,\overline{u_x^2}$ and $PV = \tfrac{1}{3} N m \overline{u^2}$. Comparing with $PV = Nk_BT$,
\[\bar{e}_k = \frac{1}{2} m \overline{u^2} = \frac{3}{2} k_B T.\]

\begin{remark}
    The average translational kinetic energy per molecule depends only on temperature, not on the mass or type of molecule. This is the microscopic meaning of temperature.
\end{remark}
\begin{remark}
    The speeds follow the Maxwell--Boltzmann distribution, with most probable speed $<$ average speed $<$ rms speed $\sqrt{\overline{u^2}}$.
\end{remark}

\section{Equipartition and Heat Capacities}
\marginpar{September 16, 2026}

\subsection{The Equipartition Theorem}
\begin{definition}[Degree of freedom]
    A degree of freedom (DOF) is a quadratic term in the total energy of a molecule. The number of such terms is $f$.
\end{definition}
\begin{example}[Monatomic gas]
    Translation: $\tfrac{1}{2}mv_x^2 + \tfrac{1}{2}mv_y^2 + \tfrac{1}{2}mv_z^2$, three terms $(v_x,v_y,v_z)$, so $f = 3$.
\end{example}
\begin{principle}[Equipartition theorem]
    At temperature $T$, each degree of freedom has an average energy of $\tfrac{1}{2} k_B T$. So $N$ molecules with $f$ DOF each have
    \[U_{\text{thermal}} = N f \cdot \frac{1}{2} k_B T.\]
\end{principle}
\begin{remark}
    $U_{\text{thermal}}$ is not the total energy, which also includes fixed energy like chemical bonds. Equipartition still gives how $U$ changes with $T$, unless bonds break or the phase changes.
\end{remark}
\subsection{Counting Degrees of Freedom}
Each of the $N$ atoms in a molecule can move in $x$, $y$, $z$, giving $3N$ motions in total, split into:
\begin{itemize}
    \item \textbf{Translational:} $3$, the whole molecule moving.
    \item \textbf{Rotational:} $3$ (nonlinear) or $2$ (linear, since spinning about its own axis does nothing).
    \item \textbf{Vibrational:} the rest, $3N - 6$ (nonlinear) or $3N - 5$ (linear), atoms moving relative to each other.
\end{itemize}
Translations and rotations are kinetic only, so one DOF each. Each vibration is a spring, kinetic plus potential, so two DOF each.

\begin{example}[O$_2$, linear]
    $3N = 6$: 3 translational, 2 rotational, $6 - 5 = 1$ vibration, so $f = 3 + 2 + 2(1) = 7$.
\end{example}
\begin{example}[H$_2$O, nonlinear]
    $3N = 9$: 3 translational, 3 rotational, $9 - 6 = 3$ vibrations, so $f = 3 + 3 + 2(3) = 12$.
\end{example}

\paragraph{Solids.}
\marginpar{September 18, 2026}
Atoms are pinned near lattice sites, so only vibration remains. With $n$ atoms per unit cell there are $3n$ vibrational modes, so $f = 6n$ ($f = 6$ for a simple solid).

\subsection{Freezing Out}
Each DOF has a minimum quantum excitation energy (for a vibration, $\hbar \omega_{\text{vib}}$). If $\tfrac{1}{2} k_B T$ is much smaller than it, the DOF stays in its ground state and is frozen out. So $f \to f(T)$ and $C_V \to C_V(T)$.

% TODO: figure of the vibrational potential energy well, showing the ground state and the spacing hbar-omega to the first excited state.

\subsection{Predicting Heat Capacities}
From $U = \tfrac{f}{2} N k_B T$,
\[C_V = \frac{f}{2} N k_B.\]

\begin{example}[Monatomic gas]
    $f = 3$, which cannot freeze out: $c_V = \tfrac{3}{2} R \approx \SI{12.5}{\joule\per\mole\per\kelvin}$, independent of the atom and of $T$ ($R = \SI{8.31}{\joule\per\mole\per\kelvin}$).
\end{example}
\begin{example}[Diatomic H$_2$]
    $c_V = \tfrac{3}{2}R$ at low $T$, $\tfrac{5}{2}R$ once rotation turns on (above $\sim 100$ K), and $\tfrac{7}{2}R$ once vibration turns on ($k_BT \sim \hbar\omega_{\text{vib}}$, thousands of K).
\end{example}

\begin{proposition}[Rule of Dulong and Petit]
    At high $T$, with all vibrational modes active, a monatomic solid ($f = 6$) has
    \[c_V \approx 3R \approx \SI{25}{\joule\per\mole\per\kelvin}.\]
\end{proposition}

\section{Heat Engines and the Second Law}
\label{sec:second-law}

\begin{remark}
    Convention for this section: $T_1$ is the hot reservoir and $T_2$ the cold one, and $Q_1$, $Q_2$, $W$ are magnitudes (all positive).
\end{remark}

\subsection{Cycles and Heat Engines}
\begin{definition}[Cycle]
    A process that returns the system (working substance) to its initial state. Then $\Delta U = 0$, and the net work is the area enclosed by the loop on the $P$-$V$ diagram.
\end{definition}

\begin{definition}[Heat engine]
    A device that takes a working substance around a cycle, absorbing $Q_1$ from a hot reservoir at $T_1$, rejecting $Q_2$ to a cold one at $T_2$, and producing work $W$. By the first law, $W = Q_1 - Q_2$, and its efficiency is
    \[\eta = \frac{W}{Q_1} = 1 - \frac{Q_2}{Q_1}.\]
\end{definition}

\begin{remark}
    Getting $\eta = 1$ would need $Q_2 = 0$. The second law forbids this. Any heat taken from the hot reservoir and not turned into work is a loss.
\end{remark}

\begin{itemize}
    \item Stirling cycle: two isotherms joined by two isochores.
    \item Carnot cycle: two isotherms joined by two adiabats.
\end{itemize}

\begin{example}[Nuclear plant]
    $\eta = 0.31$, electrical output \SI{625}{\mega\watt}. Per day, $W = \SI{625e6}{\watt} \times \SI{86400}{\second} = \SI{5.4e13}{\joule}$, so $Q_1 = W/\eta = \SI{1.74e14}{\joule}$ and the waste heat is $Q_2 = Q_1 - W = \SI{1.20e14}{\joule}$, which is why plants need cooling towers.
\end{example}

\subsection{Statements of the Second Law}
\begin{law}[Kelvin--Planck statement]
    It is impossible to construct a device that, operating in a cycle, produces no effect other than the extraction of heat from a single body at uniform temperature and the production of an equivalent amount of work.
\end{law}

\begin{law}[Clausius statement]
    It is impossible to construct a device that, operating in a cycle, produces no effect other than the transfer of heat from a colder to a hotter body.
\end{law}

\begin{remark}
    The key words matter:
    \begin{itemize}
        \item \textit{Cycle}: an isothermal expansion of an ideal gas converts heat entirely into work, but it is not a cycle.
        \item \textit{No other effect}: the reservoirs only deliver heat, no work.
        \item \textit{Single body}: there is no reservoir to accept waste heat.
    \end{itemize}
    The converse is allowed: converting work entirely into heat is possible. A refrigerator or heat pump does not violate Clausius, since work is done on it.
\end{remark}

\begin{proposition}
    The Kelvin and Clausius statements are equivalent.
\end{proposition}
\begin{proof}
    If Kelvin is false, an engine $E$ turns $Q_1$ from the hot body entirely into $W = Q_1$. Use $W$ to drive a refrigerator $R$ that takes $Q_2$ from the cold body and delivers $Q_1 + Q_2$ to the hot body. The composite moves $Q_2$ from cold to hot with no other effect, violating Clausius.

    If Clausius is false, a device moves $Q_2$ from cold to hot with no work. Run an ordinary engine $E$ that takes $Q_1$ from the hot body, rejects $Q_2$ to the cold body and produces $W = Q_1 - Q_2$. The cold body is unchanged, so the composite turns $Q_1 - Q_2$ from the hot body entirely into work, violating Kelvin.
\end{proof}

\subsection{The Carnot Cycle and Carnot's Theorem}
\begin{definition}[Carnot cycle]
    A fully reversible cycle between only two reservoirs:
    \begin{enumerate}
        \item isothermal expansion at $T_1$, absorbing $Q_1$;
        \item adiabatic expansion, $T_1 \to T_2$;
        \item isothermal compression at $T_2$, rejecting $Q_2$;
        \item adiabatic compression, $T_2 \to T_1$.
    \end{enumerate}
\end{definition}

\begin{theorem}[Carnot's theorem]
    No engine operating between two reservoirs is more efficient than a Carnot engine operating between the same two reservoirs.
\end{theorem}
\begin{proof}
    Suppose an engine $E'$ has $\eta' > \eta_C$. Scale the cycles so that $W' = W$; then $W/Q_1' > W/Q_1$ gives $Q_1 > Q_1'$. Since the Carnot engine $C$ is reversible, run it backwards as a refrigerator driven by the work of $E'$. The composite does no net work, but delivers $Q_1 - Q_1' > 0$ to the hot reservoir and takes the same amount from the cold one, violating Clausius.
\end{proof}

\begin{corollary}
    All Carnot (reversible) engines between the same two reservoirs have the same efficiency, independent of the working substance. It depends only on the reservoir temperatures.
\end{corollary}

\subsection{Thermodynamic Temperature Scale}
\begin{definition}[Thermodynamic temperature]
    Since $\eta_C$ depends only on the reservoirs, define the thermodynamic temperature by
    \[\eta_C = 1 - \frac{T_2}{T_1}, \qquad \text{so} \qquad \frac{T_1}{T_2} = \frac{Q_1}{Q_2} \quad \text{(Carnot only)}.\]
\end{definition}

\begin{remark}
    The scale is self-consistent: stacking Carnot engines $C_{12}$ and $C_{23}$ on a ladder $T_1 > T_2 > T_3$ gives $T_1/T_3 = Q_1/Q_3$, without $T_2$. It is independent of the working substance (heat can always be measured), and it coincides with the ideal gas scale up to a constant factor.
\end{remark}

\subsection{Refrigerators and Heat Pumps}
A Carnot engine run backwards takes $Q_2$ from the cold reservoir, uses work $W = Q_1 - Q_2$, and delivers $Q_1$ to the hot one. Performance is measured by the coefficient of performance (COP), which can exceed 1.

\begin{proposition}[Carnot COP]
    \[\text{Refrigerator (output $Q_2$):} \quad \text{COP} = \frac{Q_2}{W} = \frac{T_2}{T_1 - T_2},\]
    \[\text{Heat pump (output $Q_1$):} \quad \text{COP} = \frac{Q_1}{W} = \frac{T_1}{T_1 - T_2} = \frac{1}{1 - T_2/T_1}.\]
\end{proposition}

\begin{example}
    Fridge compartment at \SI{276}{\kelvin} in a \SI{293}{\kelvin} room: maximum COP $\approx 16$. Freezer at \SI{255}{\kelvin}: only $\approx 6.7$. Real fridges: about 4 or 5. Heat pump from \SI{273}{\kelvin} outside to \SI{293}{\kelvin} inside: COP $\approx 15$.
\end{example}

\begin{remark}
    The COP of both drops as $T_1 - T_2$ grows (as $T_2/T_1$ decreases).
\end{remark}

\subsection{Real Engines}
\begin{example}[Steam engine]
    Between \SI{773}{\kelvin} and \SI{293}{\kelvin}, $\eta_C = 1 - 293/773 = 0.62$, but real steam engines reach about $0.4$.
\end{example}

\begin{remark}
    Real engines do not run Carnot cycles: setting up isothermal and adiabatic steps is impractical, isothermal heat flow is far too slow to deliver useful power, and hot parts always leak heat to the environment. $\eta_C = 1$ would need $T_2 = 0$, forbidden by the third law.
\end{remark}

\paragraph{Otto cycle.}
A simplified model of a gasoline engine (a good model of the 2-stroke cycle, or of the compression/expansion part of the 4-stroke cycle). The gas is taken to be air, an ideal diatomic gas with $\gamma = 7/5$ (the mixture is only $\sim 6\%$ fuel):
\begin{itemize}
    \item $a \to b$: adiabatic compression, $V_1 \to V_2$ (fast, so little heat flows);
    \item $b \to c$: ignition at constant $V_2$, $Q_1 = C_V(T_c - T_b)$;
    \item $c \to d$: adiabatic expansion, $V_2 \to V_1$;
    \item $d \to a$: exhaust at constant $V_1$, $Q_2 = C_V(T_d - T_a)$.
\end{itemize}

\begin{proposition}[Otto efficiency]
    With compression ratio $r_c = V_1/V_2$,
    \[\eta = 1 - \left(\frac{V_2}{V_1}\right)^{\gamma - 1} = 1 - \frac{1}{r_c^{\gamma - 1}}.\]
\end{proposition}
\begin{proof}
    Along the adiabats $TV^{\gamma-1}$ is constant: $T_a V_1^{\gamma-1} = T_b V_2^{\gamma-1}$ and $T_d V_1^{\gamma-1} = T_c V_2^{\gamma-1}$. Subtracting, $T_d - T_a = (T_c - T_b)(V_2/V_1)^{\gamma-1}$, so
    \[\eta = 1 - \frac{Q_2}{Q_1} = 1 - \frac{T_d - T_a}{T_c - T_b} = 1 - \left(\frac{V_2}{V_1}\right)^{\gamma - 1}.\]
\end{proof}

\begin{remark}
    Efficiency improves with $r_c$. Cars have $r_c \approx 7$ or $8$, giving $\eta \approx 1 - 7^{-2/5} = 0.54$, while real engines reach about $30\%$. High-performance cars have $r_c \geq 10$ (F1 capped $r_c$ at 16 for 2026). Schroeder 4.3 also covers the Diesel, Stirling and Rankine (steam) cycles.
\end{remark}

\section{Entropy}
The second law is about the direction of spontaneous processes: heat flows from hot to cold, and the reverse requires work. Entropy quantifies this. The plan: derive the Clausius inequality for any cycle, use it on reversible paths to define $S$ as a state function, then compare reversible and irreversible processes.

\subsection{The Clausius Inequality}
\begin{theorem}[Clausius inequality]
    For any cycle,
    \[\oint \frac{\dbar Q}{T_{\text{surr}}} \leq 0,\]
    where $T_{\text{surr}}$ is the temperature of the reservoir supplying heat at each point of the cycle (not $T_{\text{sys}}$).
\end{theorem}
\begin{proof}
    Supply each small heat $\dbar Q_i$ to the working substance from an auxiliary reservoir at $T_i$, kept unchanged by a Carnot engine $C_i$ running between it and a reservoir at $T_0$. That reservoir is in turn refilled from a single principal reservoir at $T_0$, which therefore gives $(T_0/T_i)\,\dbar Q_i$ while $C_i$ does work $\dbar W_i$. Take the composite system (working substance, Carnot engines, auxiliary reservoirs). After one full cycle it is unchanged, so $\Delta U = 0$ and the external work done by it is
    \[W_{\text{ext}} = Q = T_0 \sum_i \frac{\dbar Q_i}{T_i}.\]
    If $W_{\text{ext}} > 0$, heat from a single reservoir became work in a cycle, violating Kelvin. So $W_{\text{ext}} = Q \leq 0$, i.e.\ $\sum_i \dbar Q_i / T_i \leq 0$, which becomes the integral in the limit.
\end{proof}

\begin{corollary}
    For a fully reversible cycle, running it backwards flips the sign of every $\dbar Q$, so both $\leq$ and $\geq$ hold:
    \[\oint_R \frac{\dbar Q_R}{T} = 0.\]
    The subscript "surr" is dropped since $T_{\text{sys}} = T_{\text{surr}}$ on a reversible path. If $T_{\text{sys}} \neq T_{\text{surr}}$ anywhere, the process is irreversible.
\end{corollary}

\subsection{Definition of Entropy}
Take a reversible cycle: $i \to f$ along $R_1$, back along $R_2$. The corollary gives $\int_{R_1} \dbar Q_R/T = -\int_{R_2, f\to i} \dbar Q_R/T = \int_{R_2, i \to f} \dbar Q_R/T$, so the integral is the same along every reversible path.

\begin{definition}[Entropy]
    Entropy $S$ is the state function with
    \[\Delta S = S_f - S_i = \int_{i,\,R}^{f} \frac{\dbar Q_R}{T}, \qquad dS = \frac{\dbar Q_R}{T} \quad \text{(reversible only)}.\]
\end{definition}

\begin{remark}
    Only differences in $S$ are defined here. The integral must be along a reversible path ($T_{\text{sys}} = T_{\text{surr}}$), with $T$ in kelvin; $S$ is in \si{\joule\per\kelvin}. The same $\dbar Q$ gives a smaller $dS$ at higher $T$.
\end{remark}

\begin{remark}
    For an irreversible process between two equilibrium states, $\Delta S$ is still defined: pick any convenient reversible path between the same endpoints.
\end{remark}

\begin{example}[Heating water]
    \SI{1}{\kilo\gram} of water ($C_P = \SI{4190}{\joule\per\kelvin}$) is heated from \SI{20}{\celsius} to \SI{100}{\celsius} on a stove at \SI{100}{\celsius}. This is irreversible (large $T$ difference), so use a reversible path through reservoirs at $T + dT$: $\dbar Q = C_P\,dT$,
    \[\Delta S = C_P \int_{T_i}^{T_f} \frac{dT}{T} = C_P \ln\frac{T_f}{T_i} = (\SI{4190}{\joule\per\kelvin}) \ln\frac{373}{293} = \SI{1.01e3}{\joule\per\kelvin}.\]
\end{example}

\begin{example}[Free expansion]
    An ideal gas expands freely into vacuum, doubling its volume. $Q = W = 0$, so $\Delta U = 0$ and $T$ is unchanged. Use a reversible isothermal expansion between the same states: $Q_R = Nk_BT\ln 2$, so
    \[\Delta S = \frac{Q_R}{T} = Nk_B \ln 2 > 0,\]
    even though no heat flowed in the actual (adiabatic but irreversible) process.
\end{example}

\subsection{Principle of Increasing Entropy}
\begin{theorem}
    For any infinitesimal process between equilibrium states,
    \[dS \geq \frac{\dbar Q}{T_{\text{surr}}},\]
    with equality if and only if the process is reversible.
\end{theorem}
\begin{proof}
    Build a cycle: $i \to f$ along the actual (possibly irreversible) path, then back $f \to i$ along a reversible path $R$. Clausius gives
    \[\int_i^f \frac{\dbar Q}{T_0} + \int_{f,\,R}^{i} \frac{\dbar Q_R}{T} \leq 0 \quad\Longrightarrow\quad \int_i^f \frac{\dbar Q}{T_0} \leq \int_{i,\,R}^{f} \frac{\dbar Q_R}{T} = S_f - S_i.\]
\end{proof}

\begin{corollary}[Principle of increasing entropy]
    For a thermally isolated system, $\dbar Q = 0$, so
    \[\Delta S \geq 0.\]
    The entropy of a thermally isolated system increases in any irreversible process and is unchanged in any reversible one. Isolated systems evolve toward the equilibrium state of maximum entropy.
\end{corollary}

\begin{remark}
    If the system is also mechanically isolated (no work), $U$ is constant too.
\end{remark}
\begin{remark}
    The principle applies to the whole isolated system, not to its parts. If $T_A > T_B$ inside an adiabatic wall, heat flows from $A$ to $B$: $S_A$ decreases, $S_B$ increases, and the total increases.
\end{remark}
