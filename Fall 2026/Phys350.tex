\subsection{Terminologies}
\begin{definition}[Scalar]
    A scalar is a quantity with just a magnitude.
\end{definition}
\begin{definition}[Vector]
    A vector is a quantity with a magnitude and a direction, i.e. $\vec{v} = v_x\hat{x} + v_y\hat{y} + v_z\hat{z}$.
\end{definition}
\begin{definition}[Cross Product]
    The cross product is a vector perpendicular to both, set by the right hand rule, and it vanishes for parallel vectors:
    \[\vec{v}\times\vec{w} = |\vec{v}||\vec{w}|\sin\theta\,\hat{n}
    = \begin{vmatrix} \hat{x} & \hat{y} & \hat{z} \\ v_x & v_y & v_z \\ w_x & w_y & w_z \end{vmatrix}
    = \hat{x}(v_yw_z - w_yv_z) - \hat{y}(v_xw_z - w_xv_z) + \hat{z}(v_xw_y - w_xv_y).\]
\end{definition}
\begin{definition}[Dot Product]
    The dot product measures how much of $\vec{v}$ points along $\vec{w}$, so it is zero for perpendicular vectors:
    \[\vec{v}\cdot\vec{w} = v_xw_x + v_yw_y + v_zw_z = |\vec{v}||\vec{w}|\cos\theta.\]
\end{definition}
\begin{remark}
    Note $\vec{v}\times\vec{w} = -\vec{w}\times\vec{v}$, and $|\vec{v}| = \sqrt{\vec{v}\cdot\vec{v}}$.
\end{remark}
\begin{definition}[Field]
    A field is something with a value at every point in space. A scalar field assigns a number to each point, like temperature $T(x,y,z)$ or the height of a hill. A vector field assigns a vector to each point, like fluid velocity or $\vec{E}$.
\end{definition}

\subsection{Differential Calculus}
\begin{definition}[Del operator]
    The del operator
    \[\vec{\nabla} = \left(\frac{\partial}{\partial x}, \frac{\partial}{\partial y}, \frac{\partial}{\partial z}\right).\]
\end{definition}
\begin{definition}[Gradient]
    The gradient turns a scalar into a vector, pointing in the direction of steepest increase of $f$:
    \[\vec{\nabla}f = \frac{\partial f}{\partial x}\hat{x} + \frac{\partial f}{\partial y}\hat{y} + \frac{\partial f}{\partial z}\hat{z}.\]
\end{definition}
\begin{remark}
    The gradient points in the steepest uphill direction, and its length grows where the hill is steeper. It is always perpendicular to the contours of constant $f$.
\end{remark}
\begin{definition}[Divergence]
    Divergence turns a vector into a scalar, measuring sources and sinks. For a small box, non-zero divergence means the inflow and outflow through its faces do not balance: arrows spreading from a point (positive) mean fluid is created there, arrows converging (negative) mean it is destroyed.
    \[\vec{\nabla}\cdot\vec{F} = \frac{\partial F_x}{\partial x} + \frac{\partial F_y}{\partial y} + \frac{\partial F_z}{\partial z}.\]
\end{definition}
\begin{definition}[Curl]
    Curl turns a vector into a vector, measuring local rotation. Picture a small paddle wheel: if the flow spins it, the curl is non-zero there. Uniform flow does not spin it, but shear flow (parallel arrows, longer on one side) does, so straight field lines do not imply zero curl.
    \[\vec{\nabla}\times\vec{F} = \begin{vmatrix} \hat{x} & \hat{y} & \hat{z} \\[2pt] \frac{\partial}{\partial x} & \frac{\partial}{\partial y} & \frac{\partial}{\partial z} \\[2pt] F_x & F_y & F_z \end{vmatrix} = \left(\frac{\partial}{\partial y}F_z - \frac{\partial}{\partial z}F_y, -(\frac{\partial}{\partial x}F_z - \frac{\partial}{\partial z}F_x), \frac{\partial}{\partial x}F_y - \frac{\partial}{\partial y}F_x\right).\]
\end{definition}
\begin{remark}
    The direction of the curl follows the right-hand rule: curl your fingers along the paddle wheel's rotation and your thumb points along $\vec{\nabla}\times\vec{F}$.
\end{remark}
\begin{remark}
    Divergence and curl are themselves fields: they can vanish in some places and not others, and a general field has both.
\end{remark}

\subsection{Integral Calculus}
\begin{definition}[Integral]
    An integral is an instruction to move along a region, evaluate something at each step, and add it to a running total.
\end{definition}
\begin{definition}[Line Integral]
    A line integral sums $\vec{F}$ along a curve, picking up only the component tangent to the path:
    \[\int_C \vec{F}\cdot d\vec{l}.\]
\end{definition}
\begin{example}
    Take $\vec{F} = y\hat{x}$ along a counterclockwise quarter circle of radius $R$.

    It is negative because $\vec{F}$ points along $+\hat{x}$ while we travel in the $-\hat{x}$ direction over most of the arc.
\end{example}
\begin{definition}[Flux]
    Flux measures how much of the field pokes through a surface:
    \[\int_S \vec{F}\cdot d\vec{a} = \int_S \vec{F}\cdot\hat{n}\,da.\]
\end{definition}
\begin{remark}
    Flux is the total amount of a vector field flowing through a surface. Divergence is the flux per unit volume at a specific point.
\end{remark}
\begin{definition}[Volume Integral]
    A volume integral sums a scalar over a region\footnote{A volume integral gives the mass if $f$ is a density}:
    \[\int_V f\,d^3\vec{r} = \int_V f\,dx\,dy\,dz.\]
\end{definition}
\begin{remark}
    We write $\oint$ when the curve is a closed loop or the surface is closed.
\end{remark}

\subsubsection{The Three Fundamental Theorems}
Integrating a derivative over a region reduces to evaluating the original object on the boundary of that region.
\begin{theorem}[Gradient theorem]
    The boundary of a curve is two points:
    \[\int_C \vec{\nabla}h\cdot d\vec{l} = h(b) - h(a).\]
    To find my altitude change climbing a hill, I can compare before and after, or add up all the little changes along the way.
\end{theorem}
\begin{theorem}[Stokes' theorem]
    The boundary of a surface is a curve:
    \begin{equation}
        \iint_S (\vec{\nabla}\times\vec{F})\cdot d\vec{S} = \oint_C \vec{F}\cdot d\vec{l}.
    \end{equation}
    To get the total swirl, I can work out the rotation at the boundary, or add up all the little whirlpools inside. Neighbouring whirlpools cancel along their shared edges, leaving only the boundary.
\end{theorem}
\begin{theorem}[Gauss' divergence theorem]
    \label{Gauss-div}
    The boundary of a volume is a surface:
    \begin{equation}
        \oiint_S (\vec{F}\cdot\hat{n})dS = \iiint_V(\vec{\nabla}\cdot\vec{F})dV.
    \end{equation}
    To find the total flux leaking out through the surface, I can add up all the sources inside. Flux between neighbouring cells cancels, leaving only the outer surface.
\end{theorem}
\begin{definition}[Current density]
    %this has nothing to do here, in griffiths it should be in magnetostatics section
    The current $I$ through a wire is not a local quantity, since a wire has finite thickness. The local version is the current density
    \[\vec{J} = \frac{d\vec{I}}{da_\perp},\]
    where $da_\perp$ is the differential cross section perpendicular to the flow.
\end{definition}
\begin{remark}
    %this has nothing to do here, in griffiths it should be in magnetostatics section
    The divergence theorem \ref{Gauss-div} applied to $\vec{J}$ is how the conservation of charge gets written down.
    \[\frac{d}{dt}\iiint_V\rho dV = - \oiint_S\vec{J}\cdot d\vec{A}.\]
\end{remark}

\subsection{Index Notation}
\begin{remark}
    A matrix carries two indices, $A_{ij}$, where $i$ says which row and $j$ says which column.
\end{remark}
\begin{definition}[Transpose]
    The transpose swaps the indices, $(A^T)_{ij} = A_{ji}$.
\end{definition}
\begin{definition}[Symmetry]
    A symmetric matrix has $A_{ij} = A_{ji}$, and an antisymmetric one has $B_{ij} = -B_{ji}$.
\end{definition}
\begin{definition}[Kronecker delta]
    The Kronecker delta equals $+1$ when its condition is fulfilled and $0$ otherwise, so it could be the identity matrix in index notation:
    \[\delta_{ij} = \begin{cases} +1 & i = j \\ 0 & i \neq j \end{cases}.\]
\end{definition}
\begin{example}
    \[\sum_j \delta_{ij}v_j = v_i.\]
\end{example}
\begin{proposition}[Matrix Multiplication]
    \[\vec{w} = A\vec{v} \implies w_i = \sum_j A_{ij}v_j.\]
\end{proposition}
\begin{proposition}[Dot Product]
    \begin{equation}
        \vec{v}\cdot\vec{w} = \sum_i v_iw_i; \vec{\nabla}\cdot\vec{F} = \sum_i \partial_i F_i.
    \end{equation}
\end{proposition}
\begin{proposition}[Cross Product]
    \begin{equation}
        (\vec{v}\times\vec{w})_i = \sum_{j,k}\varepsilon_{ijk}v_jw_k; (\vec{\nabla}\times\vec{F})_i = \sum_{j,k}\varepsilon_{ijk}\partial_j F_k.
    \end{equation}
\end{proposition}
\begin{definition}[Levi-Civita symbol]
    $\varepsilon_{ijk}$ is defined by
    \begin{itemize}
        \item $\varepsilon_{123} = \varepsilon_{231} = \varepsilon_{312} = +1$.
        \item $\varepsilon_{321} = \varepsilon_{213} = \varepsilon_{132} = -1$.
        \item If any two indices are equal, $\varepsilon_{ijk} = 0$.
    \end{itemize}
\end{definition}
\begin{example}
    For $i = 1$, only the terms with $(j,k) = (2,3)$ and $(3,2)$ survive:
    \[(\vec{v}\times\vec{w})_1 = \varepsilon_{123}v_2w_3 + \varepsilon_{132}v_3w_2 = v_2w_3 - v_3w_2.\]
\end{example}
\begin{proposition}[Epsilon-delta identity]
    \label{Epsilon-delta-id}
    Contracting two Levi-Civita symbols on a shared index gives
    \begin{equation}
        \sum_i \varepsilon_{ijk}\varepsilon_{imn} = \delta_{jm}\delta_{kn} - \delta_{jn}\delta_{km}.
    \end{equation}
\end{proposition}

\subsection{Second Derivatives Identities}
\begin{definition}[Laplacian]
    Taking the divergence of a gradient gives
    \[\vec{\nabla}\cdot(\vec{\nabla}f) = \frac{\partial^2 f}{\partial x^2} + \frac{\partial^2 f}{\partial y^2} + \frac{\partial^2 f}{\partial z^2} \equiv \nabla^2 f,\]
    where $\nabla^2$ is the Laplacian\footnote{Mathematicians often write $\Delta$, but in physics we use $\nabla^2$.}.
\end{definition}
\begin{remark}
    $\nabla^2$ also acts on vector fields, component by component: $(\nabla^2\vec{F})_i = \nabla^2 F_i$.
\end{remark}
The full list of second derivative identities:
\begin{table}[H]
    \centering
    \renewcommand{\arraystretch}{1.4}
    \begin{tabular}{l|l}
        \toprule
        \textbf{Name} & \textbf{Identity} \\
        \midrule
        Divergence of a gradient & $\vec{\nabla}\cdot(\vec{\nabla}f) = \nabla^2 f$ \\
        Curl of a gradient & $\vec{\nabla}\times(\vec{\nabla}f) = 0$ \\
        Divergence of a curl & $\vec{\nabla}\cdot(\vec{\nabla}\times\vec{F}) = 0$ \\
        Curl of a curl & $\vec{\nabla}\times(\vec{\nabla}\times\vec{F}) = \vec{\nabla}(\vec{\nabla}\cdot\vec{F}) - \nabla^2\vec{F}$ \\

\end{table}

\subsubsection{Proving the Identities}
\begin{proof}[Curl of a Curl]
    Set $\vec{G} \equiv \vec{\nabla}\times(\vec{\nabla}\times\vec{F})$. Then,
    \[G_i = \sum_{j,k}\varepsilon_{ijk}\partial_j\sum_{m,n} \varepsilon_{kmn}\partial_m F_n=\sum_{j,k,m,n}\varepsilon_{ijk}\varepsilon_{kmn}\partial_j\partial_mF_n.\]
    Using the cyclic permutation $\varepsilon_{ijk} = \varepsilon_{kij}$, so that the summed index $k$ sits first in both epsilons, the epsilon-delta identity~\ref{Epsilon-delta-id} gives
    \[\sum_k \varepsilon_{kij}\varepsilon_{kmn} = \delta_{im}\delta_{jn} - \delta_{in}\delta_{jm}.\]
    Reading off the proposition with the summed index $k$ removed: 
    \[G_i = \sum_{j,m,n}\left(\delta_{im}\delta_{jn} - \delta_{in}\delta_{jm}\right)\partial_j\partial_m F_n.\]
    Each Kronecker delta collapses one sum by setting its indices equal. In the first term $\delta_{im}$ forces $m = i$ and $\delta_{jn}$ forces $n = j$; in the second, $\delta_{in}$ forces $n = i$ and $\delta_{jm}$ forces $m = j$:
    \[G_i = \sum_j \partial_j\partial_i F_j - \sum_j \partial_j\partial_j F_i \overset{\footnotemark}{=} \partial_i\sum_j\partial_j F_j - \sum_j \partial_j\partial_j F_i = \left(\vec{\nabla}(\vec{\nabla}\cdot\vec{F})\right)_i - (\nabla^2\vec{F})_i.\]
    \footnotetext{This equality uses $\partial_j\partial_i = \partial_i\partial_j$, valid for $\vec{F}$ twice continuously differentiable.}
    \[\therefore \vec{\nabla}\times(\vec{\nabla}\times\vec{F}) = \vec{\nabla}(\vec{\nabla}\cdot\vec{F}) - \nabla^2\vec{F}.\]
\end{proof}
\begin{proof}[Curl of a Gradient]
    Let $f$ be twice continuously differentiable. With $(\vec{\nabla}f)_k = \partial_k f$,
    \[\left(\vec{\nabla}\times(\vec{\nabla}f)\right)_i = \sum_{j,k}\varepsilon_{ijk}\partial_j\partial_k f.\]
    The indices $j$ and $k$ are summed over, so they can be renamed $j \leftrightarrow k$:
    \[\left(\vec{\nabla}\times(\vec{\nabla}f)\right)_i = \sum_{j,k}\varepsilon_{ikj}\partial_k\partial_j f.\]
    Now use antisymmetry, $\varepsilon_{ikj} = -\varepsilon_{ijk}$, and the fact that partial derivatives commute, $\partial_k\partial_j f = \partial_j\partial_k f$:
    \[\left(\vec{\nabla}\times(\vec{\nabla}f)\right)_i = -\sum_{j,k}\varepsilon_{ijk}\partial_j\partial_k f = -\left(\vec{\nabla}\times(\vec{\nabla}f)\right)_i.\]
    A number equal to its own negative is zero, so every component vanishes and $\vec{\nabla}\times(\vec{\nabla}f) = \vec{0}$.
\end{proof}

\subsection{The Theory of Vector Fields}
\begin{theorem}[Helmholtz]
    If $\vec{\nabla}\cdot\vec{F}$ and $\vec{\nabla}\times\vec{F}$ are specified everywhere and $\vec{F}\to 0$ at infinity, then $\vec{F}$ is uniquely determined. It can be written as $\vec{F} = \vec{\nabla}\psi + \vec{\nabla}\times\vec{A}$, an irrotational part plus a solenoidal part.
\end{theorem}
\begin{theorem}[Irrotational fields]
    Let $\vec{F}$ be a $C^1$ vector field on a simply connected domain (e.g.\ all of $\mathbb{R}^3$). The following are equivalent:
    \begin{enumerate}[label=(\roman*)]
        \item $\vec{\nabla}\times\vec{F} = 0$ everywhere.
        \item $\int_a^b \vec{F}\cdot d\vec{l}$ is independent of path, for any given end points.
        \item $\oint_C \vec{F}\cdot d\vec{l} = 0$ for every closed loop $C$.
        \item $\vec{F} = -\vec{\nabla}\psi$ for some scalar field $\psi$.
    \end{enumerate}
\end{theorem}
\begin{theorem}[Solenoidal fields]
    Let $\vec{F}$ be a $C^1$ vector field on a domain with no enclosed voids (e.g.\ all of $\mathbb{R}^3$). The following are equivalent:
    \begin{enumerate}[label=(\roman*)]
        \item $\vec{\nabla}\cdot\vec{F} = 0$ everywhere.
        \item $\int_S \vec{F}\cdot d\vec{a}$  is independent of surface, for any given boundary line.
        \item $\oint_S \vec{F}\cdot d\vec{a} = 0$ for any closed surface $S$.
        \item $\vec{F} = \vec{\nabla}\times\vec{A}$ for some vector field $\vec{A}$.
    \end{enumerate}
\end{theorem}
\begin{remark}
    In electrostatics we will find $\vec{\nabla}\times\vec{E} = 0$, which licenses $\vec{E} = -\vec{\nabla}V$ and the electric potential. In magnetostatics we will find $\vec{\nabla}\cdot\vec{B} = 0$, which licenses $\vec{B} = \vec{\nabla}\times\vec{A}$, the vector potential.
\end{remark}

\subsection{Electric Field}
\begin{law}[Coulomb's law]
    The force on a point charge $q_2$ due to a point charge $q_1$ is
    \begin{equation}
        \vec{F} = \frac{1}{4\pi\epsilon_0}\frac{q_1q_2}{\mathfrak{r}^2}\,\hat{\mathfrak{r}},
    \end{equation}
    where $\vec{\mathfrak{r}}$ points from $q_1$ to $q_2$.\footnote{Coulomb's law has the same shape as Newton's gravity. The reason is geometric rather than a coincidence, and Gauss' law makes that explicit: in $d$ spatial dimensions the flux spreads over a $(d-1)$-dimensional sphere, so $F\propto 1/\mathfrak{r}^{d-1}$, and the $4\pi$ is the area of the unit sphere in $d=3$.}
\end{law}
\begin{definition}[Permittivity of free space]
    The permittivity of free space measures how easily an electric field can form in a classical vacuum:
    \[\epsilon_0 = 8.85 \times 10^{-12}\ \mathrm{C^2/(N\cdot m^2)}.\]
\end{definition}
\begin{definition}[Electric field]
    The electric field is the force per unit test charge. For a set of point charges $q_i$,
    \begin{equation}
        \vec{E}(\vec{r}) = \frac{1}{4\pi\epsilon_0}\sum_{i=1}^{n}\frac{q_i}{\mathfrak{r}_i^2}\,\hat{\mathfrak{r}}_i,
        \qquad \vec{\mathfrak{r}}_i \equiv \vec{r} - \vec{r}_i.
    \end{equation}
\end{definition}
\begin{principle}[Superposition]
    \label{Superposition}
    The field of several sources is the sum of their individual fields, $\vec{E} = \vec{E}_1 + \vec{E}_2 + \dots$
\end{principle}
\begin{remark}[Field theory]
    Source charges create $\vec{E}$; a test charge $Q$ then feels $\vec{F} = Q\vec{E}$.
\end{remark}

\subsubsection{Continuous Charge Distributions}
\begin{definition}[Continuous distributions]
    Replacing $q_i$ by $dq = \lambda\,dl' = \sigma\,da' = \rho\,d\tau'$ turns the sum into an integral:
    \begin{equation}
        \vec{E}(\vec{r}) = \frac{1}{4\pi\epsilon_0}\int\frac{dq}{\mathfrak{r}^2}\,\hat{\mathfrak{r}},
        \qquad \vec{\mathfrak{r}} = \vec{r} - \vec{r}',
    \end{equation}
    where $\vec{r}$ is the field point and $\vec{r}'$ the source point.
\end{definition}
\begin{remark}
    $\hat{\mathfrak{r}}$ is not constant: it points differently for every source point $\vec{r}'$, so it cannot come out of the integral. Split it into components first.
\end{remark}

\end{example}

\subsection{Gauss' Law}
\begin{law}[Gauss' law]
    The flux of $\vec{E}$ through any closed surface is set by the charge inside it:
    \begin{equation}
        \oiint_S \vec{E}\cdot d\vec{a} \overset{Gauss' Law}{=} \iiint_V(\vec{\nabla}\cdot\vec{E})d\tau = \frac{Q_{enc}}{\epsilon_0} = \frac{1}{\epsilon_0}\iiint_V \rho\,d\tau.
    \end{equation}
    %explain how it equals to Q_enc/eps0 cuz it was not explained previously or maybe I missed it.
    Since this holds for every volume $V$, the integrands match, giving the differential form
    \begin{equation}
        \vec{\nabla}\cdot\vec{E} = \frac{\rho}{\epsilon_0}.
    \end{equation}
\end{law}
\begin{definition}[Gaussian Surface]
    The Gaussian surface just means an imaginary surface (nothing is physically there)
\end{definition}
\begin{remark}
    %instead to make it shorter can we just say: if no charge inside gaussian surface the net flux is 0, cause field enter from one side and exit from another.
    Flux is a charge detector. Since $d\vec{a}$ always points outward, a field just passing through a closed surface gives negative flux where it enters and positive where it leaves, and these tend to cancel. Net flux needs field lines diverging out of the volume, which needs charge inside.
\end{remark}

\subsubsection{Spherical symmetry}
\begin{example}[Point charge]
    Surround a point charge $q$ with a sphere $S$ of radius $r$ centred on it. Two facts reduce the integral to a product:
    \begin{itemize}
        \item By symmetry, $\vec{E} = E(r)\,\hat{r}$: the field is radial and has the same magnitude everywhere on $S$.
        \item By geometry, $d\vec{a} = \hat{n}\,da = \hat{r}\,da$ on a sphere.
    \end{itemize}
    So $\vec{E}\cdot d\vec{a} = E\,da$, and
    \[\oiint_S \vec{E}\cdot d\vec{a} = E\underbrace{\oiint_S da}_{4\pi r^2} = \frac{q}{\epsilon_0}\quad\Longrightarrow\quad E = \frac{q}{4\pi\epsilon_0 r^2}.\]
\end{example}

\subsubsection{Cylindrical symmetry}
\begin{example}[Infinite line of charge]
    %Can you add like a reference to the "line of charge" example to say the derivation comes from there that would greatly reduce the size of this current example; and also when saying end caps carry no flux, also say that on a cap (a flat disk), the arrows of E lie in the disk, pointing outward from its center like spokes of a wheel. They are perpendicular to the line of charge, and parallel to the cap's surface.
    For a line with charge $\lambda$ per unit length, take a coaxial\footnote{Sharing a common central axis.} cylinder of radius $z$ and length $d$. $\vec{E}$ is radial, so the end caps carry no flux:
    \[E\,(2\pi z d) = \frac{\lambda d}{\epsilon_0}      \quad\Longrightarrow\quad E = \frac{\lambda}{2\pi\epsilon_0 z}.\]
    The field falls off as $1/z$, not $1/z^2$.
\end{example}
\begin{example}[Finite versus infinite line]
    %Too long
    Near the midpoint, a finite line gives a weaker field than an infinite one with the same $\lambda$. The same cylinder encloses the same charge, hence the same total flux, but for the finite line some of it leaks through the end caps, leaving less through the curved surface.
\end{example}

\subsubsection{Planar symmetry}
\begin{example}[Infinite plane of charge]
    %What? curved? I thought we were talking about planes. Gaussian pillbox straddling???
    For a plane with surface charge density $\sigma$, take a Gaussian pillbox straddling the plane with caps of area $A$. By symmetry $\vec{E}$ points straight away from the plane on both sides, so the curved side carries no flux and each cap carries $EA$:
    \[2EA = \frac{\sigma A}{\epsilon_0} \quad\Longrightarrow\quad E = \frac{\sigma}{2\epsilon_0}.\]
    The field does not depend on the distance from the plane.
\end{example}
\begin{example}[Infinite plane via Coulomb's law]
    %So plane is a disk here? and too long
    Use polar coordinates $(r',\phi')$ on the plane, centred below the field point at height $z$. Each piece $\sigma r'\,dr'\,d\phi'$ pushes along $\hat{\mathfrak{r}}$. Sideways components cancel in pairs, so keep only the $\hat{z}$ part via $\cos\psi = z/\sqrt{r'^2+z^2}$:
    \[\vec{E} = \hat{z}\int_0^{2\pi}\!d\phi'\int_0^\infty \frac{\sigma}{4\pi\epsilon_0}\frac{r'z}{(r'^2+z^2)^{3/2}}\,dr' = \frac{\sigma z}{2\epsilon_0}\left[-\frac{1}{\sqrt{r'^2+z^2}}\right]_0^\infty\hat{z} = \frac{\sigma}{2\epsilon_0}\hat{z}.\]
    Why distance drops out: up close each piece pushes hard but mostly at a bad angle; far away each piece is weak but more of them push at a good angle. The two effects cancel exactly. For a line, stepping back also brings in more charge at good angles, but only $\propto r$, not enough to beat $1/r^2$, hence $E \propto 1/r$.
\end{example}

\subsubsection{Why the fall-off depends on the geometry}
%what fall-off? i dont see any equation about fall-off referred
The exponent in the fall-off counts how much charge a cone of field lines can ``see'' as it grows. Picture a narrow cone opening from the observation point toward the source: its cross section at distance $r$ grows like $r^2$, which alone gives the $1/r^2$ dilution of a point charge. For extended sources, the charge inside the cone grows too:
\begin{itemize}
    \item Line of charge: the cone intercepts a length $\propto r$, so $E \propto r/r^2 = 1/r$.
    \item Sheet of charge: the cone intercepts an area $\propto r^2$, so $E \propto r^2/r^2$, a constant. The infinite sheet has a uniform field.
\end{itemize}
\begin{remark}
    For two equal positive charges, no field lines cross the midplane: the field there is parallel to the plane and vanishes at the midpoint. Gauss' law still holds, but without symmetry it cannot solve for $\vec{E}$.
\end{remark}

\subsection{The Curl of $\vec{E}$}
\begin{example}[Line integral of a point charge]
    %What are we  calculating? a point charge moving?
    For a point charge at the origin, $d\vec{l} = dr\,\hat{r} + r\,d\theta\,\hat{\theta} + r\sin\theta\,d\phi\,\hat{\phi}$ and only the radial part survives:
    \[\int_{\vec{a}}^{\vec{b}}\vec{E}\cdot d\vec{l} = \frac{q}{4\pi\epsilon_0}\int_{r_a}^{r_b}\frac{dr}{r^2} = \frac{q}{4\pi\epsilon_0}\left(\frac{1}{r_a} - \frac{1}{r_b}\right).\]
    This depends only on the endpoints, so around a closed loop ($r_a = r_b$) it vanishes.
\end{example}
\begin{theorem}[Electrostatic fields are curl-free]
    For any electrostatic field, $\oint\vec{E}\cdot d\vec{l} = 0$ and $\vec{\nabla}\times\vec{E} = 0$.
\end{theorem}
\begin{proof}
    %for the first sentence can i just say if you sum up all the E you get 0?
    By superposition, any static $\vec{E}$ is a sum of point-charge fields, each with zero loop integral. Stokes' theorem then gives $\iint_S(\vec{\nabla}\times\vec{E})\cdot d\vec{a} = 0$ for every surface, so $\vec{\nabla}\times\vec{E} = 0$.
\end{proof}
\begin{remark}
    Physically, $Q\oint\vec{E}\cdot d\vec{l}$ is the work done on a test charge around a closed loop. If it were nonzero, a charge could gain energy just by circling, violating energy conservation.
\end{remark}

\subsection{Electric Potential}
\begin{definition}[Electric potential]
    Since $\vec{\nabla}\times\vec{E} = 0$, the irrotational fields theorem guarantees a scalar $V$ with
    \begin{equation}
        \vec{E} = -\vec{\nabla}V.
    \end{equation}
    The minus sign is convention: positive charges are pushed toward lower potential energy.
\end{definition}

\begin{remark}
    %why? explain a bit more but dont make it too long
    $\oint\vec{E}\cdot d\vec{l} = 0$ holds only in electrostatics. If fields vary in time, a charge can gain net energy around a loop.
\end{remark}
\begin{proposition}[Potential from the field]
    \begin{equation}
        V(\vec{r}) = -\int_{\text{ref}}^{\vec{r}}\vec{E}\cdot d\vec{l}.
    \end{equation}
    The path does not matter since the integral is path-independent\footnote{Changing the reference point adds a constant to $V$, which leaves $\vec{E} = -\vec{\nabla}V$ unchanged (gauge freedom). Statement (ii) of the irrotational fields theorem.}. By convention the reference is at infinity, unless the charge distribution extends to infinity.
\end{proposition}
\begin{example}[Point charge]
    Along a radial path,
    \[V(r) = -\frac{q}{4\pi\epsilon_0}\int_\infty^r\frac{dr'}{r'^2} = \frac{q}{4\pi\epsilon_0 r}.\]
\end{example}
\begin{proposition}[Potential of a distribution]
    By superposition,
    \begin{equation}
        V(\vec{r}) = \frac{1}{4\pi\epsilon_0}\int\frac{\rho(\vec{r}')}{\mathfrak{r}}\,d\tau',
    \end{equation}
    where $\rho(\vec{r}')d\tau'=dq$. Unlike the integral for $\vec{E}$, there is no $\hat{\mathfrak{r}}$, so computing $V$ first and then $\vec{E} = -\vec{\nabla}V$ is often easier.
\end{proposition}
\begin{law}[Poisson's equation]
    Combining $\vec{\nabla}\cdot\vec{E} = \rho/\epsilon_0$ with $\vec{E} = -\vec{\nabla}V$,
    \begin{equation}
        \nabla^2 V = -\frac{\rho}{\epsilon_0}.
    \end{equation}
\end{law}

\subsection{Work and Energy in Electrostatics}
\begin{proposition}[Potential of a test charge]
    Bringing $Q$ in from infinity against the field takes
    \[W = -Q\int_\infty^{\vec{r}}\vec{E}\cdot d\vec{l} = QV(\vec{r}).\]
    So the electric potential $V$ is the potential energy per unit charge, in $\mathrm{J/C} = \mathrm{V}$.
\end{proposition}
\begin{proposition}[Energy of $N$ point charges]
    The first charge costs no work, since no field exists yet, and each later one feels the potential $V(\vec{r}_i)$ of those already placed, giving one term per pair:
    \[U = \frac{1}{2}\sum_{i\neq j}\frac{q_iq_j}{4\pi\epsilon_0|\vec{r}_i - \vec{r}_j|} = \frac{1}{2}\sum_iq_iV(\vec{r}_i),\]
    where the $\tfrac12$ corrects for counting each pair twice.
\end{proposition}
\begin{remark}
    %What?
    $U$ does not obey superposition: fields add, but energy goes like $E^2$, so combining two configurations gives $U = U_1 + U_2 + (\text{interaction cross terms})$.
\end{remark}
\begin{theorem}[Energy stored in the field]
    \label{E-stored-in-field}
    For a continuous distribution,
    \[U = \frac{1}{2}\int\rho V\,d\tau = \frac{\epsilon_0}{2}\int_{\text{all space}} E^2\,d\tau.\]
\end{theorem}
\begin{definition}[Energy Density]
    From the previous theorem ~\ref{E-stored-in-field}, we can unfold the energy density
    \[u = \frac{\epsilon_0}{2}E^2 \text{ (energy density).}.\]
\end{definition}
\begin{proof}
    Using the energy of N point charges, $U = \frac{1}{2}\sum_i q_i V(\vec{r}_i)$. For a continuous distribution this becomes $U = \frac{1}{2}\int\rho V\,d\tau$. Substituting $\rho = -\epsilon_0\nabla^2 V$ from Poisson's equation,
    \[U = -\frac{\epsilon_0}{2}\int(\nabla^2 V)V\,d\tau.\]
    By the product rule in index notation,
    %Where does this below come from? and rearrange it so both side are same way.
    \[\sum_i\partial_i(V\partial_i V) = \sum_i(\partial_i V)^2 + V\sum_i\partial_i\partial_i V
    \quad\Longrightarrow\quad
    (\nabla^2 V)V = \vec{\nabla}\cdot(V\vec{\nabla}V) - |\vec{\nabla}V|^2.\]
    So
    %should the negative be on the first term in the equation below?
    \[U = \frac{\epsilon_0}{2}\int|\vec{\nabla}V|^2\,d\tau - \frac{\epsilon_0}{2}\int\vec{\nabla}\cdot(V\vec{\nabla}V)\,d\tau.\]
    %What? explain with easier wording
    The divergence theorem turns the second integral over all space into a flux through its boundary, a sphere at infinity: $\oint V\vec{\nabla}V\cdot d\vec{a}$. For a bounded distribution $V \sim 1/r$ and $|\vec{\nabla}V| \sim 1/r^2$ while the area grows as $r^2$, so this term falls off as $1/r$ and vanishes. With $\vec{E} = -\vec{\nabla}V$, $U = \frac{\epsilon_0}{2}\int E^2\,d\tau$.
\end{proof}
\begin{remark}[Self-energy]
    %What? explain with easier wording
    The discrete sum excludes $i = j$; the continuous integral does not. So $\frac{\epsilon_0}{2}\int E^2\,d\tau$ is always positive, while the pair sum for two opposite charges is negative. The difference is the infinite self-energy of point charges.
\end{remark}

\begin{remark}[Compactified dimensions]
    %Add this to the appendix useless info for exam
    Suppose an extra dimension exists but is curled up, the way a flat sheet with coordinates $(x,y)$ can be rolled into a cylinder whose second coordinate $w$ is periodic with a small radius. At distances much smaller than that radius, field lines spread out into all the dimensions and the field falls off faster than $1/r^2$. At distances much larger than it, there is only room to spread out in the large dimensions and the usual $1/r^2$ is recovered. So a measured deviation from the inverse square law at short range would be evidence for extra dimensions.

    This has been tested for gravity by the E\"ot-Wash torsion balance experiments (2004, 2007, 2020), which checked the $1/r^2$ law down to separations of $52\,\mu$m. Newtonian gravity fit the data, ruling out gravitational strength Yukawa interactions with ranges above $38.6\,\mu$m at 95\% confidence.
\end{remark}
