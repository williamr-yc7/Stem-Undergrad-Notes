\section{Vector Analysis}

\subsection{Scalars, Vectors, and Fields}
\begin{definition}[Scalar]
    A scalar is a quantity with just a magnitude
\end{definition}
\begin{definition}[Vector]
    Vector is a quantity with a magnitude and a direction. A vector can be written in the form of 
    \[\vec{v} = v_x\hat{x} + v_y\hat{y} + v_z\hat{z}.\]
\end{definition}
\begin{definition}[Cross Product]
    The cross product is a vector perpendicular to both, set by the right hand rule, and it vanishes for parallel vectors, and can be mathematically written as
    \[\vec{v}\times\vec{w} = |\vec{v}||\vec{w}|\sin\theta\,\hat{n}
    = \begin{vmatrix} \hat{x} & \hat{y} & \hat{z} \\ v_x & v_y & v_z \\ w_x & w_y & w_z \end{vmatrix}
    = \hat{x}(v_yw_z - w_yv_z) - \hat{y}(v_xw_z - w_xv_z) + \hat{z}(v_xw_y - w_xv_y).\]
\end{definition}
\begin{definition}[Dot Product]
    The dot product measures how much of $\vec{v}$ points along $\vec{w}$, so it is zero for perpendicular vectors, and can be mathematically written as
    \[\vec{v}\cdot\vec{w} = v_xw_x + v_yw_y + v_zw_z = |\vec{v}||\vec{w}|\cos\theta.\]
\end{definition}
\begin{remark}
    Note $\vec{v}\times\vec{w} = -\vec{w}\times\vec{v}$, and $|\vec{v}| = \sqrt{\vec{v}\cdot\vec{v}}$.
\end{remark}
\begin{definition}[Field]
    A field is something with a value at every point in space. A scalar field assigns a number to each point, like temperature $T(x,y,z)$ or the height of a hill. A vector field assigns a vector to each point, like fluid velocity or $\vec{E}$.
\end{definition}

\subsection{Divergence, Gradient, and Curl}
\begin{definition}[Del operator]
    The del operator
    \[\vec{\nabla} = \left(\frac{\partial}{\partial x}, \frac{\partial}{\partial y}, \frac{\partial}{\partial z}\right),\]
    means nothing until it is fed a field.
\end{definition}

\begin{align}
    \vec{\nabla}f &= \frac{\partial f}{\partial x}\hat{x} + \frac{\partial f}{\partial y}\hat{y} + \frac{\partial f}{\partial z}\hat{z} && \text{(gradient: scalar $\to$ vector)}, \\
    \vec{\nabla}\cdot\vec{F} &= \frac{\partial F_x}{\partial x} + \frac{\partial F_y}{\partial y} + \frac{\partial F_z}{\partial z} && \text{(divergence: vector $\to$ scalar)}, \\
    \vec{\nabla}\times\vec{F} &= \begin{vmatrix} \hat{x} & \hat{y} & \hat{z} \\[2pt] \frac{\partial}{\partial x} & \frac{\partial}{\partial y} & \frac{\partial}{\partial z} \\[2pt] F_x & F_y & F_z \end{vmatrix} && \text{(curl: vector $\to$ vector)}.
\end{align}
For the divergence we differentiate each component along its own direction. For the curl we mix directions, which is why terms like $\partial F_y/\partial x$ show up. The direction of the curl comes from the right hand rule: curl your fingers the way the paddle wheel turns and your thumb points along $\vec{\nabla}\times\vec{F}$.

The gradient points in the steepest uphill direction, and its length grows where the hill is steeper. It is always perpendicular to the contours of constant $f$.

\begin{remark}[Reading divergence and curl off a picture]
    Divergence measures sources and sinks. Arrows pointing away from a point mean fluid is being created there (positive divergence); arrows converging on a point mean fluid is being destroyed (negative divergence). Think of a small box: non-zero divergence means the inflow and outflow through its faces do not balance.

    Curl is tested with a small paddle wheel. If the flow makes it spin, the curl is non-zero at that point. A uniform flow does not spin it, but a shear flow, where the arrows are all parallel yet longer on one side, does: one blade is pushed harder than the other, giving a net rotation. Straight field lines therefore do not imply zero curl.
\end{remark}

\begin{example}
    Judging fields by eye:
    \begin{itemize}
        \item $y\hat{x} - x\hat{y}$ circulates clockwise about the origin; $-y\hat{x} + x\hat{y}$ circulates counterclockwise. Both have non-zero curl.
        \item $y\hat{x}$ and $x\hat{y}$ have straight arrows whose length changes across the flow. These are shear fields: zero divergence, non-zero curl.
        \item Radial arrows growing with distance from a point: $\vec{\nabla}\cdot\vec{F} \neq 0$, $\vec{\nabla}\times\vec{F} = 0$.
        \item Hyperbolic flow, in along one axis and out along the other: what enters a small box leaves it, and nothing rotates, so both vanish.
        \item A spiral converging on a point: both are non-zero.
    \end{itemize}
\end{example}

\subsection{Integrals}
An integral is an instruction to move along a region, evaluate something at each step, and add it to a running total. The multivariable versions are:
\begin{align}
    \int_C \vec{F}\cdot d\vec{l} && \text{line integral, the work done if $\vec{F}$ is a force}, \\
    \int_S \vec{F}\cdot d\vec{a} = \int_S \vec{F}\cdot\hat{n}\,da && \text{flux, how much of the field pokes through the surface}, \\
    \int_V f\,d^3\vec{r} = \int_V f\,dx\,dy\,dz && \text{volume integral, the mass if $f$ is a density}.
\end{align}
We write $\oint$ when the curve is a closed loop or the surface is closed.

\begin{example}
    Take $\vec{F} = y\hat{x}$ along a counterclockwise quarter circle of radius $R$. On the circle $d\vec{l} = R\,d\phi(-\sin\phi\,\hat{x} + \cos\phi\,\hat{y})$ and $y = R\sin\phi$, so
    \[
        \int_C \vec{F}\cdot d\vec{l} = -R^2\int_0^{\pi/2}\sin^2\phi\,d\phi = -\frac{R^2\pi}{4}.
    \]
    It is negative because $\vec{F}$ points along $+\hat{x}$ while we travel in the $-\hat{x}$ direction over most of the arc.
\end{example}

\subsection{The Three Fundamental Theorems}
In single variable calculus, $\int_a^b (\partial f/\partial x)\,dx = f(b) - f(a)$. The same pattern holds in higher dimensions: integrating a derivative over a region reduces to evaluating the original object on the boundary of that region. The three theorems differ only in the dimension of the region.

\begin{theorem}[Gradient theorem]
    The boundary of a curve is two points:
    \[
        \int_a^b \vec{\nabla}h\cdot d\vec{l} = h(b) - h(a).
    \]
    To find my altitude change climbing a hill, I can compare before and after, or add up all the little changes along the way.
\end{theorem}

\begin{theorem}[Stokes' theorem]
    The boundary of a surface is a curve:
    \[
        \int_S (\vec{\nabla}\times\vec{F})\cdot d\vec{a} = \oint_C \vec{F}\cdot d\vec{l}.
    \]
    To get the total swirl, I can work out the rotation at the boundary, or add up all the little whirlpools inside. Neighbouring whirlpools cancel along their shared edges, leaving only the boundary.
\end{theorem}

\begin{theorem}[Gauss' divergence theorem]
    The boundary of a volume is a surface:
    \[
        \int_V (\vec{\nabla}\cdot\vec{F})\,d^3\vec{r} = \oint_S \vec{F}\cdot d\vec{a}.
    \]
    To find the total flux leaking out through the surface, I can add up all the sources inside. Flux between neighbouring cells cancels, leaving only the outer surface.
\end{theorem}

\begin{remark}
    Maxwell's equations come in a differential form written with $\vec{\nabla}\cdot$ and $\vec{\nabla}\times$, and an equivalent integral form written with $\oint\vec{F}\cdot d\vec{a}$ and $\oint\vec{F}\cdot d\vec{l}$. These two theorems are what let us pass between them. Gauss' law and Amp\`ere's law are the first examples.
\end{remark}

\begin{definition}[Current density]
    The current $I$ through a wire is not a local quantity, since a wire has finite thickness. The local version is the current density
    \[
        \vec{J} = \frac{d\vec{I}}{da_\perp},
    \]
    where $da_\perp$ is the differential cross section perpendicular to the flow. The divergence theorem applied to $\vec{J}$ is how the conservation of charge gets written down.
\end{definition}

\subsection{Second Derivatives}
Since we have three kinds of first derivative, we can mix and match to build second derivatives. Taking the divergence of a gradient gives
\[
    \vec{\nabla}\cdot(\vec{\nabla}f) = \frac{\partial^2 f}{\partial x^2} + \frac{\partial^2 f}{\partial y^2} + \frac{\partial^2 f}{\partial z^2} \equiv \nabla^2 f,
\]
where $\nabla^2$ is the Laplacian. Mathematicians often write $\Delta$, but in physics we use $\nabla^2$. Acting on a scalar field it returns a scalar field. It can also act on a vector field, component by component, returning a vector field.

\begin{example}[Sign of the Laplacian]
    Let $f$ have a local maximum at some point. The gradient always points uphill, so from a bird's eye view $\vec{\nabla}f$ points inward toward the maximum from every side. That field has negative divergence there, so $\nabla^2 f < 0$. The one step version: each $\partial^2 f/\partial x_i^2$ is negative at a maximum and the Laplacian is their sum. By the same argument $\nabla^2 f > 0$ at a local minimum.

    At a point where all first derivatives vanish the Laplacian can still have any sign. At a saddle point the second derivatives have opposite signs in different directions and can cancel, giving zero. So a flat point tells you nothing about $\nabla^2 f$ on its own.
\end{example}

The full list of second derivative identities:
\begin{align}
    \vec{\nabla}\cdot(\vec{\nabla}f) &= \nabla^2 f && \text{(generally non-zero)}, \\
    \vec{\nabla}\times(\vec{\nabla}f) &= 0 && \text{(curl of a gradient)}, \\
    \vec{\nabla}\cdot(\vec{\nabla}\times\vec{F}) &= 0 && \text{(divergence of a curl)}, \\
    \vec{\nabla}\times(\vec{\nabla}\times\vec{F}) &= \vec{\nabla}(\vec{\nabla}\cdot\vec{F}) - \nabla^2\vec{F} && \text{(the BAC--CAB of vector calculus)}.
\end{align}
These can be proved by writing out every partial derivative, which is tedious, or with index notation.

\subsection{Index Notation}
Write the $i$th component of $\vec{v}$ as $v_i$, so $v_1 = v_x$, $v_2 = v_y$, $v_3 = v_z$. A matrix carries two indices, $A_{ij}$, where $i$ says which row and $j$ says which column. The transpose swaps them, $(A^T)_{ij} = A_{ji}$. A symmetric matrix has $A_{ij} = A_{ji}$, and an antisymmetric one has $B_{ij} = -B_{ji}$.

\begin{definition}[Kronecker delta]
    The identity matrix is written
    \[
        \delta_{ij} = \begin{cases} +1 & i = j \\ 0 & i \neq j \end{cases}.
    \]
\end{definition}

Matrix multiplication $\vec{w} = A\vec{v}$ becomes $w_i = \sum_j A_{ij}v_j$. Multiplying by the identity gives $\sum_j \delta_{ij}v_j = v_i$, since the delta kills every term except $j = i$. This is the main trick: a delta summed against something sets one index equal to the other.

The dot product is $\vec{v}\cdot\vec{w} = \sum_i v_iw_i$, with no free index because the result is a scalar. The cross product needs a second symbol:
\[
    (\vec{v}\times\vec{w})_i = \sum_{j,k}\varepsilon_{ijk}v_jw_k.
\]

\begin{definition}[Levi-Civita symbol]
    $\varepsilon_{ijk}$ is defined by
    \begin{itemize}
        \item $\varepsilon_{123} = +1$, and cyclic permutations keep the sign: $\varepsilon_{123} = \varepsilon_{231} = \varepsilon_{312}$.
        \item Swapping any two indices flips the sign, so $\varepsilon_{213} = \varepsilon_{132} = \varepsilon_{321} = -1$.
        \item If any two indices are equal, $\varepsilon_{ijk} = 0$.
    \end{itemize}
    A quick way to remember the signs: place $1, 2, 3$ around a circle. Reading them clockwise from any starting point gives $+1$, counterclockwise gives $-1$.
\end{definition}

\begin{example}
    For $i = 1$, only the terms with $(j,k) = (2,3)$ and $(3,2)$ survive:
    \[
        (\vec{v}\times\vec{w})_1 = \varepsilon_{123}v_2w_3 + \varepsilon_{132}v_3w_2 = v_2w_3 - v_3w_2.
    \]
    Likewise $(\vec{v}\times\vec{w})_2 = v_3w_1 - v_1w_3$ and $(\vec{v}\times\vec{w})_3 = v_1w_2 - v_2w_1$, which matches the determinant.
\end{example}

The identity that does all the work is
\begin{equation}
    \sum_i \varepsilon_{ijk}\varepsilon_{imn} = \delta_{jm}\delta_{kn} - \delta_{jn}\delta_{km},
\end{equation}
which can be checked component by component.

Writing $\partial_i \equiv \partial/\partial x_i$, so $\partial_1 = \partial/\partial x$, $\partial_2 = \partial/\partial y$, $\partial_3 = \partial/\partial z$, the derivatives become
\[
    \vec{\nabla}\cdot\vec{F} = \sum_i \partial_i F_i, \qquad (\vec{\nabla}\times\vec{F})_i = \sum_{j,k}\varepsilon_{ijk}\partial_j F_k.
\]
The same bookkeeping as for the cross product gives the curl components,
\[
    (\vec{\nabla}\times\vec{F})_1 = \partial_2F_3 - \partial_3F_2, \qquad
    (\vec{\nabla}\times\vec{F})_2 = \partial_3F_1 - \partial_1F_3, \qquad
    (\vec{\nabla}\times\vec{F})_3 = \partial_1F_2 - \partial_2F_1.
\]

\subsection{Proving the Identities}
Set $\vec{G} \equiv \vec{\nabla}\times(\vec{\nabla}\times\vec{F})$. Then
\begin{align}
    G_i &= \sum_{j,k}\varepsilon_{ijk}\partial_j\left(\sum_{m,n}\varepsilon_{kmn}\partial_m F_n\right) \\
    &= \sum_{j,m,n}\left(\sum_k \varepsilon_{kij}\varepsilon_{kmn}\right)\partial_j\partial_m F_n \\
    &= \sum_{j,m,n}\left(\delta_{im}\delta_{jn} - \delta_{in}\delta_{jm}\right)\partial_j\partial_m F_n.
\end{align}
In the second line we used the cyclic property $\varepsilon_{ijk} = \varepsilon_{kij}$ to move the summed index to the front so the identity applies. The first term gives
\[
    \sum_j \partial_j\partial_i F_j = \partial_i\left(\sum_j \partial_j F_j\right) = \left(\vec{\nabla}(\vec{\nabla}\cdot\vec{F})\right)_i,
\]
and the second gives $-\sum_j\partial_j\partial_j F_i = -(\nabla^2\vec{F})_i$. Together they reproduce the identity claimed above.

For the curl of a gradient,
\begin{equation}
    \left(\vec{\nabla}\times(\vec{\nabla}f)\right)_i = \sum_{j,k}\varepsilon_{ijk}\partial_j\partial_k f. \tag{$*$}
\end{equation}
Now $j$ and $k$ are internal to the sum, so their names do not matter. Swap the labels:
\begin{equation}
    = \sum_{j,k}\varepsilon_{ikj}\partial_k\partial_j f = -\sum_{j,k}\varepsilon_{ijk}\partial_k\partial_j f = -\sum_{j,k}\varepsilon_{ijk}\partial_j\partial_k f. \tag{$**$}
\end{equation}
The second step uses the antisymmetry of $\varepsilon$ and the third uses the fact that partial derivatives commute. Comparing $(*)$ and $(**)$, the expression equals minus itself, and the only such number is zero. So $\vec{\nabla}\times(\vec{\nabla}f) = 0$.

\subsection{Irrotational and Solenoidal Fields}
The two vanishing identities let us classify fields.

\begin{itemize}
    \item Since $\vec{\nabla}\times(\vec{\nabla}\psi) = 0$, writing $\vec{F} = \vec{\nabla}\psi$ guarantees $\vec{\nabla}\times\vec{F} = 0$. Such a field is \textbf{irrotational}.
    \item Since $\vec{\nabla}\cdot(\vec{\nabla}\times\vec{A}) = 0$, writing $\vec{F} = \vec{\nabla}\times\vec{A}$ guarantees $\vec{\nabla}\cdot\vec{F} = 0$. Such a field is \textbf{solenoidal}.
\end{itemize}
Any vector field can be written as $\vec{\nabla}\psi + \vec{\nabla}\times\vec{A}$.

\begin{theorem}[Irrotational fields]
    The following four statements are equivalent:
    \begin{enumerate}[label=(\roman*)]
        \item $\vec{\nabla}\times\vec{F} = 0$ everywhere.
        \item $\int_a^b \vec{F}\cdot d\vec{l}$ is independent of path.
        \item $\oint_C \vec{F}\cdot d\vec{l} = 0$ for any closed loop.
        \item $\vec{F} = \vec{\nabla}\psi$ for some scalar $\psi$.
    \end{enumerate}
\end{theorem}

\begin{theorem}[Solenoidal fields]
    The following four statements are equivalent:
    \begin{enumerate}[label=(\roman*)]
        \item $\vec{\nabla}\cdot\vec{F} = 0$ everywhere.
        \item $\int_S \vec{F}\cdot d\vec{a}$ is independent of the surface, for surfaces sharing a boundary.
        \item $\oint_S \vec{F}\cdot d\vec{a} = 0$ for any closed surface.
        \item $\vec{F} = \vec{\nabla}\times\vec{A}$ for some vector field $\vec{A}$.
    \end{enumerate}
\end{theorem}

\begin{remark}
    In electrostatics we will find $\vec{\nabla}\times\vec{E} = 0$. That is the irrotational case, and it is what licenses us to write $\vec{E} = -\vec{\nabla}V$ and introduce the electric potential. In magnetostatics we will find $\vec{\nabla}\cdot\vec{B} = 0$, which is the solenoidal case and licenses $\vec{B} = \vec{\nabla}\times\vec{A}$, the vector potential.
\end{remark}

\section{Electrostatics}

\subsection{Gauss' Law and the Inverse Square Law}
Coulomb's law for two point charges,
\[
    F = k\frac{q_1q_2}{r^2},
\]
has the same $1/r^2$ shape as Newton's gravity, $F = Gm_1m_2/d^2$. The reason is geometric rather than a coincidence, and Gauss' law makes that explicit.

\begin{law}[Gauss' law]
    The flux of $\vec{E}$ through any closed surface is set by the charge inside it:
    \[
        \oint_S \vec{E}\cdot d\vec{a} = \frac{Q_{\text{enc}}}{\epsilon_0}.
    \]
\end{law}

Surround a point charge with a sphere of radius $r$. The flux is fixed by $Q$, but the surface area over which it is spread grows like $r^2$, so $E \propto 1/r^2$. The exponent is really a count of dimensions.

\begin{example}[Coulomb's law in $d$ dimensions]
    In a universe with four spatial dimensions $(x,y,z,w)$, volumes are four-dimensional and the surfaces enclosing them are three-dimensional, with ``area'' scaling like $r^3$. Gauss' law then gives $E \propto 1/r^3$. In general, in $d$ spatial dimensions a point charge has $E \propto 1/r^{d-1}$. The inverse square law is what three dimensions look like.
\end{example}

\begin{remark}[Compactified dimensions]
    Suppose an extra dimension exists but is curled up, the way a flat sheet with coordinates $(x,y)$ can be rolled into a cylinder whose second coordinate $w$ is periodic with a small radius. At distances much smaller than that radius, field lines spread out into all the dimensions and the field falls off faster than $1/r^2$. At distances much larger than it, there is only room to spread out in the large dimensions and the usual $1/r^2$ is recovered. So a measured deviation from the inverse square law at short range would be evidence for extra dimensions.

    This has been tested for gravity by the E\"ot-Wash torsion balance experiments (2004, 2007, 2020), which checked the $1/r^2$ law down to separations of $52\,\mu$m. Newtonian gravity fit the data, ruling out gravitational strength Yukawa interactions with ranges above $38.6\,\mu$m at 95\% confidence.
\end{remark}

\begin{remark}[Field lines]
    Field lines start on positive charges and end on negative ones, and their density measures the field strength. This is the same statement as Gauss' law: the number of lines through a closed surface counts the enclosed charge. Two like charges have lines that bend away from each other. An infinite sheet of charge has lines that cannot spread out at all, so its field is uniform and does not fall off with distance.
\end{remark}

\subsection{Gauss' Law with Symmetry}
\marginpar{September 14, 2026}

Gauss' law comes in two equivalent forms, related by the divergence theorem:
\[
    \oint_S \vec{E}\cdot d\vec{a} = \frac{Q_{\text{enc}}}{\epsilon_0}
    \qquad\Longleftrightarrow\qquad
    \vec{\nabla}\cdot\vec{E} = \frac{\rho}{\epsilon_0},
\]
where $\rho$ is the charge density. Both are always true, but the integral form is only useful for computing $\vec{E}$ when the symmetry of the charge distribution lets us pull $E$ out of the integral. That requires a closed surface on which $\vec{E}$ has constant magnitude and is either parallel or perpendicular to $d\vec{a}$ everywhere.

\begin{remark}
    The integral form answers the question "what is the flux through some closed surface?" for any surface at all, whatever the field looks like: only the enclosed charge matters. It is turning that flux back into a value of $\vec{E}$ at a point that needs symmetry.
\end{remark}

\subsubsection{Spherical symmetry}
\begin{example}[Point charge]
    Surround a point charge $q$ with a sphere $S$ of radius $r$ centred on it. Two facts reduce the integral to a product:
    \begin{itemize}
        \item By symmetry, $\vec{E} = E(r)\,\hat{r}$: the field is radial and has the same magnitude everywhere on $S$.
        \item By geometry, $d\vec{a} = \hat{n}\,da = \hat{r}\,da$ on a sphere.
    \end{itemize}
    So $\vec{E}\cdot d\vec{a} = E\,da$, and
    \[
        \oint_S \vec{E}\cdot d\vec{a} = E\underbrace{\oint_S da}_{4\pi r^2} = \frac{q}{\epsilon_0}
        \quad\Longrightarrow\quad
        E = \frac{q}{4\pi\epsilon_0 r^2}.
    \]
    This is Coulomb's law, with $k = 1/4\pi\epsilon_0$.
\end{example}

\subsubsection{Cylindrical symmetry}
\begin{example}[Infinite line of charge]
    Take an infinite straight line carrying charge $\lambda$ per unit length. Use as Gaussian surface a cylinder of radius $z$ and length $d$ coaxial with the line. By symmetry $\vec{E}$ points radially outward from the line, perpendicular to the curved surface, and it has no component along the line. It is therefore parallel to $d\vec{a}$ on the curved surface and perpendicular to $d\vec{a}$ on the two end caps, so the end caps contribute no flux. The enclosed charge is $\lambda d$, and
    \[
        \oint_S \vec{E}\cdot d\vec{a} = E\underbrace{\int_{\text{curved}} da}_{2\pi z d} = \frac{\lambda d}{\epsilon_0}
        \quad\Longrightarrow\quad
        E = \frac{\lambda}{2\pi\epsilon_0 z}.
    \]
    The length $d$ drops out, as it must for an infinite line. The field falls off as $1/z$, not $1/z^2$.
\end{example}

\begin{example}[Finite versus infinite line]
    Compare a line of length $2L$ with an infinite line, both with the same $\lambda$, and ask which gives the stronger field at a point $P$ just above the midpoint. The infinite line does. Draw the same cylinder around the middle of each: $Q_{\text{enc}} = \lambda d$ is the same, so the total flux is the same. For the infinite line no flux leaves through the end caps by symmetry. For the finite line the field lines near the ends bend outward, so some flux "leaks" out through the end caps. With the same total, the flux through the curved surface, and hence $E$ at $P$, must be smaller for the finite line.
\end{example}

\subsubsection{Why the fall-off depends on the geometry}
The exponent in the fall-off counts how much charge a fixed cone of field lines can "see" as it grows. Picture a narrow cone opening out from the observation point at distance $r$ toward the source. The cone's cross section grows like $r^2$, which is the $1/r^2$ dilution of a point charge. But for extended sources the enclosed charge grows too:
\begin{itemize}
    \item A line of charge: the cone intercepts a length of line proportional to $r$, so the field goes like $r/r^2 = 1/r$.
    \item A sheet of charge: the cone intercepts an area proportional to $r^2$, so the field goes like $r^2/r^2$, a constant. This is the uniform field of the infinite sheet noted above.
\end{itemize}

\begin{remark}
    For two equal positive charges, the field lines from each are pushed away from the other and none cross the midplane between them. On that plane the field is purely parallel to the plane, and it vanishes at the midpoint. A Gaussian surface straddling the midplane still obeys Gauss' law, but the field on it is not uniform, so the law cannot be used to solve for $\vec{E}$ there: no symmetry, no shortcut.
\end{remark}

\subsection{Coulomb's Law for Continuous Distributions}
\marginpar{September 14, 2026}

Superpose the point charge field over the source:
\[
    \vec{E}(\vec{r}) = \frac{1}{4\pi\epsilon_0}\int \frac{\rho(\vec{r}')\,d^3\vec{r}'}{\mathfrak{r}^2}\,\hat{\mathfrak{r}},
    \qquad \vec{\mathfrak{r}} \equiv \vec{r} - \vec{r}',
\]
where $\vec{r}$ is the field point, $\vec{r}'$ the source point, and $\vec{\mathfrak{r}}$ (Griffiths' script $r$) the separation vector from source to field point. For a line of charge the volume element collapses to $\rho\,d^3\vec{r}' \to \lambda(\vec{r}')\,dl'$.

\begin{example}[Finite line of charge by direct integration]
    A line of length $2L$ lies along the $x$ axis from $-L$ to $L$ with uniform $\lambda$, and we want $\vec{E}$ at $P = (0,0,z)$ on the perpendicular bisector.
        \begin{figure}[H]
        \centering
                \begin{tikzpicture}[scale=1.1, >=stealth]
            % axes
            \draw[->] (-3.2,0) -- (3.4,0) node[right] {$\hat{x}$};
            \draw[->] (0,-0.3) -- (0,3.2) node[above] {$\hat{z}$};
            % line of charge
            \draw[line width=1.6pt] (-2.5,0) -- (2.5,0);
            \foreach \x in {-2.5,2.5} \draw[thick] (\x,0.15) -- (\x,-0.15);
            \node[below=4pt] at (-2.5,0) {$-L$};
            \node[below=4pt] at (2.5,0) {$+L$};
            \node[below left] at (0,0) {$O$};
            % element dx'
            \fill[green!60!black] (1.35,-0.09) rectangle (1.65,0.09);
            \node[above=2pt, green!60!black] at (1.5,0.1) {$dx'$};
            % field point
            \fill (0,2.5) circle (2pt) node[above right] {$P$};
            % r vector, label on the right
            \draw[->, thick, blue] (0,0) -- (0,2.5);
            \node[blue, right=3pt] at (0,1.4) {$\vec{r} = z\hat{z}$};
            % r' vector, drawn just below the axis
            \draw[->, thick, blue] (0,-0.55) -- (1.5,-0.55);
            \node[blue, below=2pt] at (0.75,-0.55) {$\vec{r}' = x'\hat{x}$};
            % separation vector
            \draw[->, thick, red] (1.5,0) -- (0,2.5);
            \node[red, above right=2pt] at (0.85,1.3) {$\vec{\mathfrak{r}} = \vec{r} - \vec{r}'$};
            % brace for z on the left
            \draw[decorate, decoration={brace, amplitude=5pt}] (-0.35,0) -- (-0.35,2.5)
                node[midway, left=7pt] {$z$};
        \end{tikzpicture}
        \caption{Setup for the finite line of charge. The source point $\vec{r}' = x'\hat{x}$ runs from $-L$ to $L$; the field point is $P$ at $\vec{r} = z\hat{z}$; the separation vector is $\vec{\mathfrak{r}} = \vec{r} - \vec{r}'$.}
    \end{figure}
    
    Here $\vec{r} = z\hat{z}$ and $\vec{r}' = x'\hat{x}$, so
    \[
        \vec{\mathfrak{r}} = z\hat{z} - x'\hat{x}, \qquad
        |\vec{\mathfrak{r}}| = \sqrt{z^2 + x'^2}, \qquad
        \hat{\mathfrak{r}} = \frac{z\hat{z} - x'\hat{x}}{\sqrt{z^2 + x'^2}},
    \]
    and the total charge is $Q = \int\lambda\,dx' = 2\lambda L$. Feeding this into the integral,
    \[
        \vec{E}(z\hat{z}) = \frac{\lambda}{4\pi\epsilon_0}\int_{-L}^{L}
        \frac{1}{z^2 + x'^2}\,\frac{z\hat{z} - x'\hat{x}}{\sqrt{z^2 + x'^2}}\,dx'.
    \]
    The $\hat{x}$ term is odd in $x'$ and integrates to zero, as the symmetry of the setup demands. For the $\hat{z}$ term, $\int dx'/(z^2 + x'^2)^{3/2} = x'/\big(z^2\sqrt{z^2 + x'^2}\big)$, so
    \[
        \vec{E} = \frac{\lambda z\hat{z}}{4\pi\epsilon_0}\cdot\frac{2L}{z^2\sqrt{L^2 + z^2}}
        = \underbrace{\frac{\lambda}{2\pi\epsilon_0 z}\hat{z}}_{\text{infinite line}}
        \cdot\underbrace{\frac{L}{\sqrt{L^2 + z^2}}}_{<\,1}.
    \]
    The prefactor is the infinite line result from Gauss' law, and the second factor is the correction for finite length. It is always less than one, which is the ``leaked flux'' argument made quantitative. Two limits:
    \begin{itemize}
        \item $z \ll L$: $L/\sqrt{L^2 + z^2} \approx L/L = 1$, so $\vec{E} \to \dfrac{\lambda}{2\pi\epsilon_0 z}\hat{z}$. Close to the line it looks infinite, and Gauss' law is recovered.
        \item $z \gg L$: $\sqrt{L^2 + z^2} \approx z$, so $\vec{E} \to \dfrac{2\lambda L}{4\pi\epsilon_0 z^2}\hat{z} = \dfrac{Q}{4\pi\epsilon_0 z^2}\hat{z}$. Far away it looks like a point charge $Q = 2\lambda L$.
    \end{itemize}
\end{example}

\begin{remark}
    Always check limits. A result for a finite distribution should collapse to a simpler known field both up close and far away; if it does not, the integral is wrong.
\end{remark}
